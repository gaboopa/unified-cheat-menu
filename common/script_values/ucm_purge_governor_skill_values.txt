# 读取 scope:ucm_grant_target_title 档位；无 scope 时按男爵线
ucm_civil_grant_threshold_for_target_title_value = {
  if = {
    limit = {
      exists = scope:ucm_grant_target_title
      scope:ucm_grant_target_title = {
        tier = tier_empire
      }
    }
    value = ucm_civil_grant_empire_threshold_value
  }
  else_if = {
    limit = {
      exists = scope:ucm_grant_target_title
      scope:ucm_grant_target_title = {
        tier = tier_kingdom
      }
    }
    value = ucm_civil_grant_kingdom_threshold_value
  }
  else_if = {
    limit = {
      exists = scope:ucm_grant_target_title
      scope:ucm_grant_target_title = {
        tier = tier_duchy
      }
    }
    value = ucm_civil_grant_duchy_threshold_value
  }
  else_if = {
    limit = {
      exists = scope:ucm_grant_target_title
      scope:ucm_grant_target_title = {
        tier = tier_county
      }
    }
    value = ucm_civil_grant_county_threshold_value
  }
  else = {
    value = ucm_civil_grant_barony_threshold_value
  }
}

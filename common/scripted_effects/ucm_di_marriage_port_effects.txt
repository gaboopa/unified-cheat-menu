# 自 Divine Intervention #2986538297 移植
# 源文件：common/scripted_guis/DI_diplomacy_sgui.txt
# 变量映射：DI_primary_character → ucm_marriage_slot_primary
#          DI_tertiary_character → ucm_marriage_slot_tertiary
# 使用前须设置 scope:ucm_marriage_pick_char = 被选中角色；scope:actor = 玩家。

ucm_di_select_primary_character_effect = {
  scope:ucm_marriage_pick_char = {
    save_scope_as = select_char
  }
  scope:actor = {
    set_variable = {
      name = ucm_marriage_slot_primary
      value = scope:select_char
    }
    if = {
      limit = {
        scope:select_char = {
          exists = primary_partner
        }
      }
      scope:select_char = {
        primary_partner = {
          save_scope_as = ucm_di_auto_tertiary_pick
        }
      }
      set_variable = {
        name = ucm_marriage_slot_tertiary
        value = scope:ucm_di_auto_tertiary_pick
      }
    }
    else = {
      remove_variable ?= ucm_marriage_slot_tertiary
    }
  }
}

ucm_di_clear_primary_character_effect = {
  scope:actor = {
    remove_variable ?= ucm_marriage_slot_primary
  }
}

ucm_di_select_tertiary_character_effect = {
  scope:ucm_marriage_pick_char = {
    save_scope_as = select_char
  }
  scope:actor = {
    set_variable = {
      name = ucm_marriage_slot_tertiary
      value = scope:select_char
    }
  }
}

ucm_di_clear_tertiary_character_effect = {
  scope:actor = {
    remove_variable ?= ucm_marriage_slot_tertiary
  }
}

# DI_dip_marry
ucm_di_dip_marry_effect = {
  scope:actor = {
    var:ucm_marriage_slot_primary = {
      marry = scope:actor.var:ucm_marriage_slot_tertiary
    }
  }
}

# DI_dip_marry_matrilineal
ucm_di_dip_marry_matrilineal_effect = {
  scope:actor = {
    if = {
      limit = {
        has_variable = ucm_marriage_slot_primary
        has_variable = ucm_marriage_slot_tertiary
      }
      var:ucm_marriage_slot_primary = {
        marry_matrilineal = scope:actor.var:ucm_marriage_slot_tertiary
      }
    }
  }
}

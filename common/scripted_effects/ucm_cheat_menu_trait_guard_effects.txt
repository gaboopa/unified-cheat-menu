# Custom traits only come through the menu: nothing grants them unless a human player is acting.
ucm_cheat_menu_grant_cheat_trait_effect = {
  if = {
    limit = { scope:actor ?= { is_ai = no } ucm_cheat_menu_may_receive_cheat_trait_trigger = yes }
    add_trait = $TRAIT$
    if = { limit = { trait:$TRAIT$ = trait:ucm_trait_baidubuqin } ucm_holder_list_add_effect = { LIST = ucm_baidubuqin_holders } ucm_trait_baidubuqin_upkeep_effect = yes }
    if = { limit = { trait:$TRAIT$ = trait:ucm_trait_xinwuguai } ucm_holder_list_add_effect = { LIST = ucm_xinwuguai_holders } ucm_trait_xinwuguai_upkeep_effect = yes }
    if = { limit = { trait:$TRAIT$ = trait:ucm_trait_stats_100_locked } ucm_holder_list_add_effect = { LIST = ucm_stats100_holders } }
  }
}

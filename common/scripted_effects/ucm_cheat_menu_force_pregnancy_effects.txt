# 隐匿分娩：create_character 用真实生父（玩家）遗传；合法父亲另设；外貌与先天特质从玩家复制（set_father 会冲掉 blend）
ucm_cheat_menu_stealth_birth_child_father_fixup_effect = {
  if = {
    limit = {
      exists = scope:ucm_inst_legal_dad
      exists = scope:ucm_inst_bio_dad
      NOT = { scope:ucm_inst_legal_dad = scope:ucm_inst_bio_dad }
    }
    set_real_father = scope:ucm_inst_bio_dad
    set_father = scope:ucm_inst_legal_dad
    copy_inheritable_appearance_from = scope:ucm_inst_bio_dad
    scope:ucm_inst_bio_dad = {
      save_scope_as = ucm_trait_inherit_donor
    }
    ucm_cheat_menu_copy_inheritable_congenital_traits_from_donor_effect = yes
  }
}

# 自然分娩兜底：隐匿强制怀孕仍确保真实生父=玩家，并复制玩家可遗传外貌与先天特质
ucm_stealth_fp_birth_ensure_player_genes_effect = {
  if = {
    limit = {
      scope:mother = {
        has_variable = ucm_stealth_fp_bio_dad
      }
    }
    scope:mother = {
      var:ucm_stealth_fp_bio_dad = {
        save_scope_as = ucm_fp_bio_dad
      }
    }
    if = {
      limit = {
        exists = scope:ucm_fp_bio_dad
      }
      set_real_father = scope:ucm_fp_bio_dad
      copy_inheritable_appearance_from = scope:ucm_fp_bio_dad
      scope:ucm_fp_bio_dad = {
        save_scope_as = ucm_trait_inherit_donor
      }
      ucm_cheat_menu_copy_inheritable_congenital_traits_from_donor_effect = yes
    }
  }
}

# 当日分娩：登记生父（隐匿时 bio=真实父亲/玩家，legal=假定父亲/配偶等）
ucm_cheat_menu_resolve_delivery_father_effect = {
  clear_saved_scope = ucm_inst_bio_dad
  clear_saved_scope = ucm_inst_legal_dad
  clear_saved_scope = ucm_deliv_real
  clear_saved_scope = ucm_deliv_assumed
  if = {
    limit = {
      pregnancy_real_father = { always = yes }
      pregnancy_assumed_father = { always = yes }
    }
    pregnancy_real_father = { save_scope_as = ucm_deliv_real }
    pregnancy_assumed_father = { save_scope_as = ucm_deliv_assumed }
    scope:ucm_deliv_real = { save_scope_as = ucm_inst_bio_dad }
    if = {
      limit = {
        NOT = { scope:ucm_deliv_real = scope:ucm_deliv_assumed }
      }
      scope:ucm_deliv_assumed = { save_scope_as = ucm_inst_legal_dad }
    }
  }
  else_if = {
    limit = { pregnancy_real_father = { always = yes } }
    pregnancy_real_father = { save_scope_as = ucm_inst_bio_dad }
  }
  else_if = {
    limit = { pregnancy_assumed_father = { always = yes } }
    pregnancy_assumed_father = { save_scope_as = ucm_inst_bio_dad }
  }
  else_if = {
    limit = {
      exists = primary_spouse
      primary_spouse = { is_male = yes }
    }
    primary_spouse = { save_scope_as = ucm_inst_bio_dad }
  }
}

# 当日分娩 create_character：隐匿私生用假定父亲/母亲文化，避免子女并入玩家宗族
ucm_cheat_menu_resolve_delivery_create_scopes_effect = {
  clear_saved_scope = ucm_inst_create_father
  clear_saved_scope = ucm_inst_culture_src
  if = {
    limit = {
      exists = scope:ucm_inst_legal_dad
      exists = scope:ucm_inst_bio_dad
      NOT = { scope:ucm_inst_legal_dad = scope:ucm_inst_bio_dad }
    }
    scope:ucm_inst_legal_dad = { save_scope_as = ucm_inst_create_father }
    scope:mother = { save_scope_as = ucm_inst_culture_src }
  }
  else_if = {
    limit = { exists = scope:ucm_inst_bio_dad }
    scope:ucm_inst_bio_dad = { save_scope_as = ucm_inst_create_father }
    scope:ucm_inst_bio_dad = { save_scope_as = ucm_inst_culture_src }
  }
  else = {
    scope:mother = { save_scope_as = ucm_inst_culture_src }
  }
}

namespace = ucm_debug_actions

# Clean replacement for the old tocmod.2000-2090 debug-event maze.
# The target is restored from the global interaction lock on every page.
# Every submenu has an explicit parent button and a close button; successful actions remain on-page.

ucm_debug_actions.1000 = {
	type = character_event
	title = ucm_debug_actions.1000.t
	desc = ucm_debug_actions.1000.desc
	theme = default
	immediate = { ucm_cheat_menu_sync_target_effect = yes }

	option = { name = ucm_debug_actions.1000.resources trigger_event = ucm_debug_actions.1100 }
	option = { name = ucm_debug_actions.1000.character trigger_event = ucm_debug_actions.1200 }
	option = { name = ucm_debug_actions.1000.perks trigger_event = ucm_debug_actions.2000 }
	option = {
		name = ucm_debug_actions.1000.county
		trigger = { scope:ucm_cheat_menu_target = { is_landed = yes } }
		trigger_event = ucm_debug_actions.3000
	}
	option = {
		name = ucm_debug_actions.1000.traits
		ucm_cheat_menu_open_base_game_traits_effect = yes
	}
	option = {
		name = ucm_debug_actions.1000.lifestyle
		ucm_cheat_menu_open_lifestyle_effect = yes
	}
	option = { name = ucm_debug_actions.close }
}

ucm_debug_actions.2000 = {
	type = character_event
	title = ucm_debug_actions.2000.t
	desc = ucm_debug_actions.2000.desc
	theme = default
	immediate = { ucm_cheat_menu_sync_target_effect = yes }

	option = {
		name = ucm_debug_actions.2000.all_perks
		ucm_debug_actions_grant_base_perks_effect = yes
		trigger_event = ucm_debug_actions.2000
	}
	option = {
		name = ucm_debug_actions.2000.lifestyle
		ucm_cheat_menu_open_lifestyle_effect = yes
	}
	option = {
		name = ucm_debug_actions.2000.seducer
		scope:ucm_cheat_menu_target = { add_perk = seducer_perk }
		trigger_event = ucm_debug_actions.2000
	}
	option = {
		name = ucm_debug_actions.2000.lock_court
		trigger = { scope:ucm_cheat_menu_target = { NOT = { has_character_flag = blocked_from_leaving } } }
		scope:ucm_cheat_menu_target = { add_character_flag = blocked_from_leaving }
		trigger_event = ucm_debug_actions.2000
	}
	option = {
		name = ucm_debug_actions.2000.unlock_court
		trigger = { scope:ucm_cheat_menu_target = { has_character_flag = blocked_from_leaving } }
		scope:ucm_cheat_menu_target = { remove_character_flag = blocked_from_leaving }
		trigger_event = ucm_debug_actions.2000
	}
	option = {
		name = ucm_debug_actions.2000.alliance
		trigger = {
			scope:ucm_cheat_menu_target != root
			root = { is_landed = yes }
			scope:ucm_cheat_menu_target = { is_landed = yes }
		}
		scope:ucm_cheat_menu_target = { create_alliance = root }
		trigger_event = ucm_debug_actions.2000
	}
	option = { name = ucm_debug_actions.2000.punishments trigger_event = ucm_debug_actions.2100 }
	option = {
		name = ucm_debug_actions.2000.clear
		ucm_debug_actions_clear_punishments_effect = yes
		trigger_event = ucm_debug_actions.2000
	}
	option = { name = ucm_debug_actions.back_main trigger_event = ucm_debug_actions.1000 }
	option = { name = ucm_debug_actions.close }
}

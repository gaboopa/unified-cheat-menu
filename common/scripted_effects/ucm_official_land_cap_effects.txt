ucm_land_cap_resolve_grantee_fallback_effect = {
  if = {
    limit = { ucm_land_cap_grantee_needs_fallback_trigger = yes }
    scope:ucm_off = {
      random_vassal_or_below = {
        limit = {
          ucm_land_cap_valid_grantee_character_trigger = yes
          is_landed = yes
        }
        save_scope_as = ucm_land_cap_grantee
      }
    }
  }
  if = {
    limit = { ucm_land_cap_grantee_needs_fallback_trigger = yes }
    scope:ucm_off = {
      random_vassal_or_below = {
        limit = {
          ucm_land_cap_valid_grantee_character_trigger = yes
        }
        save_scope_as = ucm_land_cap_grantee
      }
    }
  }
  if = {
    limit = { ucm_land_cap_grantee_needs_fallback_trigger = yes }
    scope:ucm_off = {
      random_courtier = {
        limit = {
          ucm_land_cap_valid_grantee_character_trigger = yes
          is_adult = yes
        }
        save_scope_as = ucm_land_cap_grantee
      }
    }
  }
  if = {
    limit = { ucm_land_cap_grantee_needs_fallback_trigger = yes }
    scope:ucm_off.top_liege = {
      random_courtier = {
        limit = {
          NOT = { this = scope:ucm_off.top_liege }
          is_adult = yes
          can_be_granted_titles_by = {
            RULER = scope:ucm_off
          }
        }
        save_scope_as = ucm_land_cap_grantee
      }
    }
  }
  if = {
    limit = { ucm_land_cap_grantee_needs_fallback_trigger = yes }
    scope:ucm_off = {
      create_character = {
        employer = scope:ucm_off
        culture = scope:ucm_off.culture
        faith = scope:ucm_off.faith
        gender_female_chance = 50
        age = 28
        dynasty = none
        save_scope_as = ucm_land_cap_grantee
      }
    }
  }
}

# 转封失败时销毁该衔，避免 while 空转卡死
ucm_land_cap_destroy_stuck_strip_title_effect = {
  if = {
    limit = {
      exists = scope:ucm_strip_title
      exists = scope:ucm_off
      scope:ucm_strip_title = {
        holder = scope:ucm_off
      }
    }
    root = {
      destroy_title = scope:ucm_strip_title
    }
  }
}

ucm_land_cap_try_transfer_strip_title_effect = {
  clear_saved_scope = ucm_land_cap_grantee
  ucm_land_cap_resolve_grantee_effect = yes
  if = {
    limit = {
      exists = scope:ucm_land_cap_grantee
      NOT = {
        scope:ucm_land_cap_grantee = scope:ucm_off.top_liege
      }
      exists = scope:ucm_strip_title
    }
    create_title_and_vassal_change = {
      type = granted
      save_scope_as = ucm_land_cap_change
      add_claim_on_loss = no
    }
    scope:ucm_strip_title = {
      change_title_holder = {
        holder = scope:ucm_land_cap_grantee
        change = scope:ucm_land_cap_change
      }
    }
    resolve_title_and_vassal_change = scope:ucm_land_cap_change
  }
  else = {
    ucm_land_cap_destroy_stuck_strip_title_effect = yes
  }
  clear_saved_scope = ucm_land_cap_grantee
}

ucm_land_cap_cleanup_official_vars_effect = {
  if = {
    limit = {
      has_variable = ucm_land_cap_official_only_high_tier
    }
    remove_variable = ucm_land_cap_official_only_high_tier
  }
  remove_character_flag = ucm_land_cap_strip_in_progress
}

ucm_land_cap_setup_kept_county_effect = {
  if = {
    limit = {
      scope:ucm_off = {
        primary_title = {
          tier = tier_county
        }
      }
    }
    scope:ucm_off.primary_title = {
      save_scope_as = ucm_kept_county
    }
  }
  else_if = {
    limit = {
      scope:ucm_off = {
        exists = capital_county
        capital_county.holder = scope:ucm_off
      }
    }
    scope:ucm_off.capital_county = {
      save_scope_as = ucm_kept_county
    }
  }
  else_if = {
    limit = {
      scope:ucm_off = {
        any_held_title = {
          tier = tier_county
          is_noble_family_title = no
          is_leased_out = no
        }
      }
    }
    scope:ucm_off = {
      ordered_held_title = {
        title_tier = county
        limit = {
          is_noble_family_title = no
          is_leased_out = no
        }
        order_by = development_level
        save_scope_as = ucm_kept_county
      }
    }
  }
}

ucm_land_cap_strip_one_excess_county_effect = {
  scope:ucm_off = {
    random_held_title = {
      title_tier = county
      limit = {
        NOT = { this = scope:ucm_kept_county }
        is_noble_family_title = no
        is_leased_out = no
      }
      save_scope_as = ucm_strip_title
    }
  }
  ucm_land_cap_try_transfer_strip_title_effect = yes
  clear_saved_scope = ucm_strip_title
}

ucm_land_cap_strip_one_excess_high_tier_effect = {
  scope:ucm_off = {
    random_held_title = {
      limit = {
        OR = {
          tier = tier_duchy
          tier = tier_kingdom
          tier = tier_empire
        }
        NOT = { this = scope:ucm_off.primary_title }
        is_noble_family_title = no
        is_landless_type_title = no
      }
      save_scope_as = ucm_strip_title
    }
  }
  scope:ucm_off = {
    if = {
      limit = {
        OR = {
          AND = {
            exists = scope:ucm_kept_county
            NOT = {
              any_held_title = {
                title_tier = county
                NOT = {
                  this = scope:ucm_kept_county
                }
                is_noble_family_title = no
                is_leased_out = no
              }
            }
          }
          AND = {
            NOT = {
              any_held_title = {
                title_tier = county
                is_noble_family_title = no
                is_leased_out = no
              }
            }
          }
        }
      }
      set_variable = {
        name = ucm_land_cap_official_only_high_tier
        value = 1
      }
    }
    else = {
      set_variable = {
        name = ucm_land_cap_official_only_high_tier
        value = 0
      }
    }
  }
  ucm_land_cap_try_transfer_strip_title_effect = yes
  clear_saved_scope = ucm_strip_title
}

ucm_enforce_official_land_cap_effect = {
  save_scope_as = ucm_off
  if = {
    limit = {
      exists = scope:ucm_off.top_liege
      scope:ucm_off = {
        is_landed = yes
      }
    }
    ucm_land_cap_setup_kept_county_effect = yes
    while = {
      limit = {
        exists = scope:ucm_kept_county
        scope:ucm_off = {
          any_held_title = {
            title_tier = county
            NOT = { this = scope:ucm_kept_county }
            is_noble_family_title = no
            is_leased_out = no
          }
        }
      }
      ucm_land_cap_strip_one_excess_county_effect = yes
    }
    while = {
      limit = {
        scope:ucm_off = {
          any_held_title = {
            OR = {
              tier = tier_duchy
              tier = tier_kingdom
              tier = tier_empire
            }
            NOT = { this = scope:ucm_off.primary_title }
            is_noble_family_title = no
            is_landless_type_title = no
          }
        }
      }
      ucm_land_cap_strip_one_excess_high_tier_effect = yes
    }
    scope:ucm_off = {
      ucm_land_cap_cleanup_official_vars_effect = yes
    }
    clear_saved_scope = ucm_kept_county
  }
}

# 分批脉冲：每名官员每轮最多剥 8 件头衔
ucm_enforce_official_land_cap_bounded_effect = {
  save_scope_as = ucm_off
  if = {
    limit = {
      exists = scope:ucm_off.top_liege
      scope:ucm_off = {
        is_landed = yes
      }
    }
    add_character_flag = ucm_land_cap_strip_in_progress
    ucm_land_cap_setup_kept_county_effect = yes
    while = {
      count = 8
      limit = {
        exists = scope:ucm_kept_county
        scope:ucm_off = {
          any_held_title = {
            title_tier = county
            NOT = { this = scope:ucm_kept_county }
            is_noble_family_title = no
            is_leased_out = no
          }
        }
      }
      ucm_land_cap_strip_one_excess_county_effect = yes
    }
    while = {
      count = 8
      limit = {
        scope:ucm_off = {
          any_held_title = {
            OR = {
              tier = tier_duchy
              tier = tier_kingdom
              tier = tier_empire
            }
            NOT = { this = scope:ucm_off.primary_title }
            is_noble_family_title = no
            is_landless_type_title = no
          }
        }
      }
      ucm_land_cap_strip_one_excess_high_tier_effect = yes
    }
    scope:ucm_off = {
      ucm_land_cap_cleanup_official_vars_effect = yes
    }
    clear_saved_scope = ucm_kept_county
  }
}

ucm_land_cap_send_complete_toast_effect = {
  send_interface_toast = {
    type = event_toast_effect_good
    title = ucm_land_cap_batch_complete_toast_title
    desc = ucm_land_cap_batch_complete_toast_desc
  }
}

ucm_land_cap_batch_finish_effect = {
  remove_character_flag = ucm_land_cap_batch_active
  clear_variable_list = ucm_land_cap_official_queue
  every_vassal_or_below = {
    limit = {
      has_character_flag = ucm_land_cap_strip_in_progress
    }
    remove_character_flag = ucm_land_cap_strip_in_progress
  }
  ucm_land_cap_send_complete_toast_effect = yes
}

ucm_land_cap_batch_step_effect = {
  while = {
    count = 2
    limit = {
      variable_list_size = {
        name = ucm_land_cap_official_queue
        value > 0
      }
    }
    ordered_in_list = {
      variable = ucm_land_cap_official_queue
      order_by = highest_held_title_tier
      position = 0
      save_scope_as = ucm_land_cap_queue_official
    }
    scope:ucm_land_cap_queue_official = {
      ucm_enforce_official_land_cap_bounded_effect = yes
    }
    if = {
      limit = {
        scope:ucm_land_cap_queue_official = {
          ucm_official_needs_land_cap_enforcement_trigger = yes
        }
      }
      remove_list_variable = {
        name = ucm_land_cap_official_queue
        target = scope:ucm_land_cap_queue_official
      }
      root = {
        add_to_variable_list = {
          name = ucm_land_cap_official_queue
          target = scope:ucm_land_cap_queue_official
        }
      }
    }
    else = {
      remove_list_variable = {
        name = ucm_land_cap_official_queue
        target = scope:ucm_land_cap_queue_official
      }
    }
    clear_saved_scope = ucm_land_cap_queue_official
  }
}

# 整理头衔完整流程：点击一次，全领地同步整理直至无多余头衔（多轮扫尾）
ucm_official_land_cap_full_effect = {
  root = {
    if = {
      limit = {
        has_character_flag = ucm_land_cap_batch_active
      }
      ucm_land_cap_batch_finish_effect = yes
    }
    while = {
      count = 24
      limit = {
        any_vassal_or_below = {
          ucm_official_needs_land_cap_enforcement_trigger = yes
        }
      }
      ucm_official_land_cap_yearly_effect = yes
    }
    ucm_land_cap_send_complete_toast_effect = yes
  }
}

ucm_official_land_cap_yearly_effect = {
  every_vassal_or_below = {
    limit = {
      ucm_official_needs_land_cap_enforcement_trigger = yes
    }
    ucm_enforce_official_land_cap_effect = yes
  }
}

# 一键洗地：多轮 revoke 收归，直至辖内无可收归的伯爵领与公国及以上头衔
ucm_one_click_strip_reclaim_pass_effect = {
  every_vassal_or_below = {
    every_held_title = {
      limit = {
        ucm_realm_title_needs_reclaim_to_root_trigger = yes
      }
      change_title_holder = {
        holder = root
        change = scope:ucm_strip_realm_change
      }
    }
  }
  every_sub_realm_county = {
    limit = {
      ucm_realm_title_needs_reclaim_to_root_trigger = yes
    }
    change_title_holder = {
      holder = root
      change = scope:ucm_strip_realm_change
    }
  }
}

ucm_one_click_strip_realm_titles_full_effect = {
  root = {
    create_title_and_vassal_change = {
      type = revoked
      save_scope_as = ucm_strip_realm_change
      add_claim_on_loss = no
    }
    while = {
      count = 48
      limit = {
        ucm_has_realm_strip_work_remaining_trigger = yes
      }
      ucm_one_click_strip_reclaim_pass_effect = yes
    }
    resolve_title_and_vassal_change = scope:ucm_strip_realm_change
  }
}

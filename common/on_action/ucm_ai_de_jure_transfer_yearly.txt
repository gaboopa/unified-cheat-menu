# 独立、有地 AI：每年 ucm_officials.9200——常备扩张 + 扩张猛攻 + 铁腕改革（无控制力/派系门槛）/（可选）法理转封。
# 玩家版图威胁：`ucm_ai_counter_player_pressure`（反玩家结盟/谍报权重）由本文件年脉冲 + ucm_ai_monthly_strategic_audit.txt 月脉冲共同刷新。
# 新局首年跳过 9200（grace）；总开关见 ucm_ai_yearly_pulse_toggle_decision；子项：法理转封（默认关）、改行政（默认关）。

on_game_start = {
  on_actions = {
    ucm_ai_features_new_campaign_defaults_pulse
  }
}

# Opt-in guard: the toggles are stored as "...disabled" globals (decisions own them), so absence means ON.
# Defaults are written once (marker ucm_ai_defaults_written). on_game_start(_after_lobby) only fire for new
# games, so every AI pulse also calls ucm_ai_write_defaults_once_effect first, and a quarterly player
# pulse writes them on loaded saves. Later loads never overwrite what the player chose.
ucm_ai_features_new_campaign_defaults_pulse = {
  effect = { ucm_ai_write_defaults_once_effect = yes }
}

on_game_start_after_lobby = {
  on_actions = {
    ucm_ai_features_new_campaign_defaults_pulse
    ucm_ai_yearly_pulse_startup_pulse
  }
}

ucm_ai_yearly_pulse_startup_pulse = {
  effect = {
    set_global_variable = {
      name = ucm_ai_yearly_pulse_grace_skip
      value = yes
    }
  }
}

yearly_global_pulse = {
  on_actions = {
    ucm_ai_de_jure_transfer_yearly_pulse
    ucm_ai_counter_player_pulse
  }
}

ucm_ai_counter_player_pulse = {
	trigger = { ucm_single_player_trigger = yes }
  effect = {
    ucm_ai_write_defaults_once_effect = yes
    if = {
      limit = {
        NOT = { has_global_variable = ucm_ai_yearly_pulse_disabled }
      }
      ucm_ai_apply_counter_player_pressure_global_effect = yes
    }
  }
}

ucm_ai_de_jure_transfer_yearly_pulse = {
	trigger = { ucm_single_player_trigger = yes }
  effect = {
    ucm_ai_write_defaults_once_effect = yes
    if = {
      limit = {
        NOT = { has_global_variable = ucm_ai_yearly_pulse_disabled }
      }
      if = {
        limit = { has_global_variable = ucm_ai_yearly_pulse_grace_skip }
        remove_global_variable = ucm_ai_yearly_pulse_grace_skip
      }
      else = {
        every_independent_ruler = {
          limit = {
            is_ai = yes
            is_landed = yes
          }
          trigger_event = {
            id = ucm_officials.9200
            days = 0
          }
        }
      }
    }
  }
}

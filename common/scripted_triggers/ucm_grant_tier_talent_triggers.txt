# Surplus-title grants: is this character skilled enough for a county? Thresholds in ucm_purge_governor_skill_values.txt

ucm_is_talent_for_grant_tier_county_trigger = {
  OR = {
    AND = {
      OR = {
        vassal_contract_has_flag = celestial_military_appointment
        vassal_contract_has_flag = meritocratic_military_appointment
      }
      ucm_military_skill_sum_for_purge_value >= ucm_military_grant_county_threshold_value
    }
    AND = {
      NOR = {
        vassal_contract_has_flag = celestial_military_appointment
        vassal_contract_has_flag = meritocratic_military_appointment
      }
      is_landed = yes
      ucm_civil_skill_sum_for_purge_value >= ucm_civil_grant_county_threshold_value
    }
    AND = {
      NOT = { is_landed = yes }
      OR = {
        ucm_civil_skill_sum_for_purge_value >= ucm_civil_grant_county_threshold_value
        ucm_military_skill_sum_for_purge_value >= ucm_military_grant_county_threshold_value
      }
    }
  }
}

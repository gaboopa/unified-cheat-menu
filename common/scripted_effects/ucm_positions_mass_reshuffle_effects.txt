# 严格宫廷任命：候选须通过引擎 can_be_employed_as（与职位定义一致）。须在 scope:ucm_pos_reshuffle_liege 内由 ucm_positions_appoint_all_named_court_positions_strict_effect 调用。
ucm_positions_try_appoint_court_slot_strict_effect = {
  if = {
    limit = {
      can_employ_court_position_type = $COURT_POS$
      any_courtier = {
        ucm_positions_loose_court_appointment_candidate_for_reshuffle_trigger = yes
        can_be_employed_as = $COURT_POS$
      }
    }
    ordered_courtier = {
      limit = {
        ucm_positions_loose_court_appointment_candidate_for_reshuffle_trigger = yes
        can_be_employed_as = $COURT_POS$
      }
      order_by = aptitude:$COURT_POS$
      max = 1
      check_range_bounds = no
      save_scope_as = ucm_pos_court_pick
    }
    if = {
      limit = { exists = scope:ucm_pos_court_pick }
      ucm_positions_bring_pick_to_court_if_vassal_effect = yes
      appoint_court_position = {
        recipient = scope:ucm_pos_court_pick
        court_position = $COURT_POS$
      }
      clear_saved_scope = ucm_pos_court_pick
    }
  }
  if = {
    limit = {
      can_employ_court_position_type = $COURT_POS$
      any_vassal = {
        ucm_positions_court_reshuffle_vassal_loose_candidate_trigger = yes
        can_be_employed_as = $COURT_POS$
      }
    }
    ordered_vassal = {
      limit = {
        ucm_positions_court_reshuffle_vassal_loose_candidate_trigger = yes
        can_be_employed_as = $COURT_POS$
      }
      order_by = aptitude:$COURT_POS$
      max = 1
      check_range_bounds = no
      save_scope_as = ucm_pos_court_pick
    }
    if = {
      limit = { exists = scope:ucm_pos_court_pick }
      ucm_positions_bring_pick_to_court_if_vassal_effect = yes
      appoint_court_position = {
        recipient = scope:ucm_pos_court_pick
        court_position = $COURT_POS$
      }
      clear_saved_scope = ucm_pos_court_pick
    }
  }
}

# Skonester title succession law overrides.
# CK3 1.20: flat law format (law_group_type + index). Group settings stay in vanilla common/law_groups.
# Laws the mod did not change (princely_elective_succession_law, temporal_head_of_faith_succession_law, law_of_the_land_succession_law, confederation_elective_succession_law) are no longer overridden so the 1.20 vanilla versions apply.
# The mod's theocratic elective is renamed skone_theocratic_elective_succession_law because 1.20 vanilla
# added its own theocratic_elective_succession_law (playable theocracies) with the same key.

oligarchic_elective_succession_law = {
	law_group_type = title_succession_laws
	index = 20
	can_have = {
		OR = {
			government_has_flag = government_is_feudal
			government_has_flag = government_is_clan
		}
		highest_held_title_tier >= tier_duchy
		is_ai = no # cheat law: players only
	}
	can_pass = {
		can_change_title_law_trigger = yes
	}
	can_title_have = {
		can_title_have_law_general_trigger = yes
	}
	succession = {
		order_of_succession = election
		election_type = oligarchic_elective
	}
	flag = elective_succession_law
	title_allegiance_opinion = 10
	revoke_cost = {
		prestige = 1
	}
	pass_cost = {
		prestige = 1
	}
}

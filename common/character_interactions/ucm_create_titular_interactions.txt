ucm_create_titular = {
	category = interaction_category_uncategorized
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	auto_accept = yes
	interface_priority = 11720
	icon = on
	desc = ucm_create_titular_desc
	target_type = title
	target_filter = recipient_domain_titles
	is_shown = {
		ucm_single_player_trigger = yes
		scope:actor = { is_ai = no has_character_flag = ucm_character_interaction_menu_open has_character_flag = bone_cheat_interactions_flag }
		scope:recipient = { is_ruler = yes }
	}
	can_be_picked_title = {
		scope:target = { holder = scope:recipient }
	}
	send_options_exclusive = yes
	send_option = { flag = titular_duchy localization = ucm_create_titular_duchy }
	send_option = { flag = titular_kingdom localization = ucm_create_titular_kingdom }
	send_option = { flag = titular_empire localization = ucm_create_titular_empire }
	send_option = { flag = titular_hegemony localization = ucm_create_titular_hegemony }
	on_accept = {
		if = {
			limit = { always = scope:titular_hegemony }
			ucm_create_titular_effect = { TIER = hegemony }
		}
		else_if = {
			limit = { always = scope:titular_empire }
			ucm_create_titular_effect = { TIER = empire }
		}
		else_if = {
			limit = { always = scope:titular_kingdom }
			ucm_create_titular_effect = { TIER = kingdom }
		}
		else = {
			ucm_create_titular_effect = { TIER = duchy }
		}
	}
	ai_will_do = { base = -999999 }
}

# Pick scripted GUIs for craft modifier slots (one per slot x value). Selected state is a GUI expression in the .gui files.

ucm_acm_gui_pick_m1_v00 = {
  scope = character
  is_valid = { always = yes }
  effect = {
    set_variable = { name = ucm_craft_m1 value = 0 }
    set_variable = { name = ucm_craft_m1_cnt value = 0 }
  }
}

  scope = character
  is_valid = { always = yes }
  effect = {
    set_variable = { name = ucm_craft_m1 value = 1 }
    if = {
      limit = { NOT = { has_variable = ucm_craft_m1_cnt } }

# 厉鬼勾魂：处决波次（手点「开启」可立即触发，或由 yearly 满 5 年时触发）
# 仅你顶层领国（与 is_in_same_realm_as=root 一致）；死因为随机疫病；人类玩家对每名死者发界面提示条（与瘟疫式提示类似）。清全局开关/年计。
ucm_officials.9086 = {
  type = character_event
  hidden = yes

  immediate = {
    root = {
      set_variable = {
        name = ucm_samge_left
        value = 1000
      }
      set_variable = {
        name = ucm_samge_guard
        value = 0
      }
    }
    # 候选名单只扫一次世界；排除玩家本人及其近亲/远亲（配偶不在此列，另行排除）
    every_living_character = {
      limit = {
        is_ai = yes
        NOT = { this = root }
        top_liege = root.top_liege
        NOT = { is_close_or_extended_family_of = root }
        NOT = { is_spouse_of = root }
      }
      add_to_temporary_list = ucm_samge_pool
    }
    while = {
      limit = {
        root = {
          var:ucm_samge_left > 0
          var:ucm_samge_guard < 12000
        }
      }
      root = {
        change_variable = {
          name = ucm_samge_guard
          add = 1
        }
      }
      if = {
        limit = {
          any_in_list = {
            list = ucm_samge_pool
            count >= 1
          }
        }
        random_in_list = {
          list = ucm_samge_pool
          save_scope_as = ucm_samge_victim
          remove_from_list = ucm_samge_pool
          root = {
            if = {
              limit = { is_ai = no }
              send_interface_toast = {
                type = event_toast_effect_neutral
                title = ucm_samge_realm_death_toast_t
                left_icon = scope:ucm_samge_victim
                custom_description_no_bullet = {
                  object = scope:ucm_samge_victim
                  text = ucm_samge_realm_death_toast_desc
                }
              }
            }
          }
          random_list = {
            25 = {
              death = {
                death_reason = death_great_pox
              }
            }
            25 = {
              death = {
                death_reason = death_typhus
              }
            }
            25 = {
              death = {
                death_reason = death_smallpox
              }
            }
            25 = {
              death = {
                death_reason = death_measles
              }
            }
          }
        }
        root = {
          change_variable = {
            name = ucm_samge_left
            add = -1
          }
        }
      }
      else = {
        root = {
          set_variable = {
            name = ucm_samge_left
            value = 0
          }
        }
      }
    }
    root = {
      if = {
        limit = {
          has_variable = ucm_samge_left
        }
        remove_variable = ucm_samge_left
      }
      if = {
        limit = {
          has_variable = ucm_samge_guard
        }
        remove_variable = ucm_samge_guard
      }
    }
    # 一轮结束即关「开」、清年计（新逻辑点按一次一轮；旧存档年循环在开关仍亮时由 yearly 触发的也一并收尾）
    if = {
      limit = {
        has_global_variable = ucm_samge_reaping_active
      }
      remove_global_variable = ucm_samge_reaping_active
    }
    if = {
      limit = {
        has_global_variable = ucm_samge_years_accum
      }
      remove_global_variable = ucm_samge_years_accum
    }
    if = {
      limit = {
        root = {
          has_character_flag = ucm_samge_reopen_pop_9080_after_samge_wave
        }
      }
      root = {
        remove_character_flag = ucm_samge_reopen_pop_9080_after_samge_wave
        trigger_event = {
          id = ucm_officials.9080
          days = 0
        }
      }
    }
  }
}

# Unified Cheat Menu: main menu window (gui/ucm_main_menu.gui). Root = player.

ucm_menu_visible = {
	scope = character
	is_shown = { ucm_single_player_trigger = yes has_character_flag = ucm_menu_open }
}

ucm_menu_close = {
	scope = character
	is_shown = { ucm_single_player_trigger = yes }
	effect = { remove_character_flag = ucm_menu_open }
}

# --- Second character -------------------------------------------------------

ucm_second_set_target = {
	scope = character
	is_shown = { ucm_single_player_trigger = yes }
	is_valid = {
		ucm_menu_target_valid_trigger = yes
		NOT = { global_var:skonester_crozier_cheats_other ?= var:ucm_target }
	}
	effect = { set_global_variable = { name = skonester_crozier_cheats_other value = var:ucm_target } }
}

ucm_second_clear = {
	scope = character
	is_shown = { ucm_single_player_trigger = yes }
	is_valid = { exists = global_var:skonester_crozier_cheats_other }
	effect = { remove_global_variable = skonester_crozier_cheats_other }
}

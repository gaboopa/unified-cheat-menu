# One-click imprison. Uses a temporary scope and limits to alive/not self, to avoid a crash from nested root/scope within one frame.

ucm_imprison_all_direct_vassals_effect = {
  root = {
    every_vassal = {
      limit = {
        is_alive = yes
        is_imprisoned = no
        NOT = { this = root }
      }
      this = { save_temporary_scope_as = ucm_ipr }
      root = {
        imprison = {
          target = scope:ucm_ipr
          type = house_arrest
        }
      }
    }
  }
}

ucm_imprison_all_realm_vassals_effect = {
  root = {
    every_vassal_or_below = {
      limit = {
        is_alive = yes
        is_imprisoned = no
        NOT = { this = root }
      }
      this = { save_temporary_scope_as = ucm_ipr }
      root = {
        imprison = {
          target = scope:ucm_ipr
          type = house_arrest
        }
      }
    }
  }
}

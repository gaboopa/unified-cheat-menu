# Poison Immunity: cures illnesses while held.

ucm_trait_baidubuqin_upkeep_effect = {
  if = { limit = { NOT = { has_character_flag = immune_to_disease } } add_character_flag = immune_to_disease add_character_flag = ucm_baidubuqin_immunity_managed }
  while = {
    count = 24
    limit = { has_trait_with_flag = illness }
    random_character_trait = { limit = { has_trait_flag = illness } save_scope_as = ucm_illness }
    remove_trait = scope:ucm_illness
  }
}

# Leaves the flag alone on characters the game made immune itself.
ucm_trait_baidubuqin_revoke_effect = { if = { limit = { has_character_flag = ucm_baidubuqin_immunity_managed } remove_character_flag = immune_to_disease remove_character_flag = ucm_baidubuqin_immunity_managed } }

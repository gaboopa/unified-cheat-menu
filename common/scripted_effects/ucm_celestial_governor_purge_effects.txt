# Removes mediocre rulers at root.var:ucm_purge_tier (0 = county to empire, 1-4 = one tier), then redistributes titles.
ucm_purge_mediocre_governors_at_selected_tier_effect = {
  every_vassal_or_below = {
    limit = {
      OR = {
        ucm_purge_eligible_governor_base_trigger = yes
        ucm_purge_eligible_non_governor_vassal_base_trigger = yes
      }
      ucm_governor_is_mediocre_by_branch_trigger = yes
      OR = {
        AND = {
          root = { var:ucm_purge_tier = 1 }
          highest_held_title_tier = tier_county
        }
        AND = {
          root = { var:ucm_purge_tier = 2 }
          highest_held_title_tier = tier_duchy
        }
        AND = {
          root = { var:ucm_purge_tier = 3 }
          highest_held_title_tier = tier_kingdom
        }
        AND = {
          root = { var:ucm_purge_tier = 4 }
          highest_held_title_tier = tier_empire
        }
        AND = {
          root = { var:ucm_purge_tier = 0 }
          OR = {
            highest_held_title_tier = tier_county
            highest_held_title_tier = tier_duchy
            highest_held_title_tier = tier_kingdom
            highest_held_title_tier = tier_empire
          }
        }
      }
    }
    save_scope_as = ucm_purge_victim
    scope:ucm_purge_victim = {
      remove_opinion = {
        target = root
        modifier = forced_me_step_down_opinion
        single = no
      }
      add_opinion = {
        target = root
        modifier = forced_me_step_down_opinion
      }
      add_character_flag = ucm_grant_pool_eligible
      if = {
        limit = {
          is_governor = yes
        }
        if = {
          limit = {
            government_allows = merit
          }
          add_character_modifier = {
            modifier = ep3_renounced_governorship_merit
            years = 15
          }
          set_appointment_timeout = {
            years = 3
            desc = appointment_timeout_desc_retired_governor
          }
        }
        else_if = {
          limit = {
            government_is_japanese_trigger = yes
          }
          add_gold = major_gold_value
          change_influence = major_influence_value
          add_character_modifier = {
            modifier = renounced_governorship_japan
            years = 15
          }
        }
        else = {
          add_gold = massive_gold_value
          add_character_modifier = {
            modifier = ep3_renounced_governorship
            years = 15
          }
        }
      }
      force_step_down_landed_titles = yes
      if = {
        limit = { has_tgp_dlc_trigger = yes }
        tgp_step_down_title_recipient_tooltip_effect = yes
      }
    }
  }
  root = {
    ucm_official_land_cap_full_effect = yes
    ucm_one_click_grant_realm_titles_effect = yes
  }
}

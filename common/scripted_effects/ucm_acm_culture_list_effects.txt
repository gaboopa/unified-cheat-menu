# 区域白名单见 common/scripted_triggers/ucm_acm_culture_region_triggers.txt
# 各大区独立 global list；GUI 用 VariableSystem + Concatenate 切换（打开窗时一次性填充）

ucm_acm_heritages_list_refresh_all_effect = {
  clear_global_variable_list = ucm_acm_heritage_list_all
  every_culture_global = {
    if = {
      limit = {
        NOT = {
          any_in_global_list = {
            variable = ucm_acm_heritage_list_all
            has_same_culture_heritage = prev
          }
        }
      }
      add_to_global_variable_list = { name = ucm_acm_heritage_list_all target = this }
    }
  }
}

# Does the realm have a barony convertible to a feudal castle: tribal / nomad / herder (not already a castle)

ucm_has_tribal_holding_in_realm_trigger = {
  any_sub_realm_county = {
    any_county_province = {
      OR = {
        has_holding_type = tribal_holding
        has_holding_type = nomad_holding
        has_holding_type = herder_holding
      }
    }
  }
}

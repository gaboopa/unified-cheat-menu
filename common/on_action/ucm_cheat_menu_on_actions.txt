# Close anything left open by the previous session, on new games and loads.
on_game_start = {
	on_actions = { ucm_new_game_setup }
}

on_game_start_after_lobby = {
	on_actions = { ucm_session_reset }
}

ucm_new_game_setup = {
	effect = { ucm_acm_faith_populate_doctrine_catalog_effect = yes }
	on_actions = { ucm_session_reset }
}

ucm_session_reset = {
	effect = {
		every_player = {
			ucm_acm_gui_startup_cleanup_effect = yes
			remove_character_flag = ucm_cheat_menu_cheat_menu
			remove_character_flag = ucm_character_interaction_menu_open
			remove_character_flag = ucm_craft_gui_open
			remove_character_flag = ucm_craft_edit_gui_open
			remove_character_flag = ucm_trait_editor_open
			remove_character_flag = ucm_lifestyle_open
		}
		remove_global_variable = ucm_cheat_menu_recipient
	}
}

# UCM trait safeguard: who holds them / who may get them (matches the list in the trait files)

# May the current character (root) get a custom trait from the UCM menu: the player, or an AI when scope:actor is a human player
ucm_cheat_menu_may_receive_cheat_trait_trigger = {
  OR = {
    is_ai = no
    AND = {
      exists = scope:actor
      scope:actor = {
        is_ai = no
      }
    }
  }
}

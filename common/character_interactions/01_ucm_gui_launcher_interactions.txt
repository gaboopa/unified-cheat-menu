# Skonester GUI launchers belong in the character interaction menu so the
# clicked character is available as scope:recipient. Ordinary decisions only
# have the decision-taker as root and therefore cannot target the open character.

ucm_open_trait_editor_decision = {
	category = interaction_category_uncategorized
	interface_priority = 11900
	common_interaction = yes
	auto_accept = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	icon = debug_chars
	desc = ucm_open_trait_editor_decision_desc
	is_shown = { ucm_single_player_trigger = yes 
		scope:actor = { is_ai = no has_character_flag = ucm_character_interaction_menu_open }
		scope:recipient = { is_alive = yes }
	}
	is_valid = { always = yes }
	can_send = { always = yes }
	on_accept = { ucm_cheat_menu_open_base_game_traits_effect = yes }
	ai_will_do = 0
}

# The menu flag sits on the player opening it, who is scope:actor in interactions.
ucm_cheat_menu_menu_enabled_trigger = {
  OR = {
    scope:actor ?= { has_character_flag = ucm_cheat_menu_cheat_menu }
    AND = { NOT = { exists = scope:actor } has_character_flag = ucm_cheat_menu_cheat_menu }
  }
}

ucm_steal_picker_enabled_trigger = { OR = { ucm_cheat_menu_menu_enabled_trigger = yes has_character_flag = ucm_steal_picker_active } }

ucm_player_may_use_artifact_craft_trigger = { is_ai = no }

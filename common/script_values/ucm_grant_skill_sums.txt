# One-click grants, best candidate: duchy to county tier uses the civil score (diplomacy + stewardship + learning); marches use the military score.
# Kingdom/empire down to county still uses ucm_grant_empire_to_county_pick_value; baronies use the military score.

ucm_grant_duchy_pick_value = {
  if = {
    limit = {
      vassal_contract_has_flag = march
    }
    add = martial
    add = prowess
  }
  else = {
    add = diplomacy
    add = stewardship
    add = learning
  }
}

ucm_grant_military_sum_value = {
  add = martial
  add = prowess
}

ucm_grant_empire_to_county_pick_value = {
  if = {
    limit = {
      vassal_contract_has_flag = march
    }
    add = martial
    add = prowess
  }
  else = {
    add = diplomacy
    add = stewardship
    add = learning
  }
}

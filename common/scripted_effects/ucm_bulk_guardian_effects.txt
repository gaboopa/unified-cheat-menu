# 批量監護人：童年特質定教育軸 → 教育重心 → 每次執行先徹底拆掉學童側全部監護人（含多條/殘留），滿 6 歲再從「當前無學童」候選擇優掛關係（一人一師）；由決議手動執行

# 學童 scope:ucm_bulk_ward：while + every_relation 多輪拆除（避免單輪迭代漏拆）；再拆 education 主監護人鏈接
# 僅用 hidden_effect：不顯示專用提示行，且隱去每條 remove_relation_* 的明細
ucm_bulk_guardian_strip_all_guardians_from_ward_effect = {
  hidden_effect = {
    scope:ucm_bulk_ward = {
      while = {
        count = 30
        limit = {
          any_relation = {
            type = guardian
          }
        }
        every_relation = {
          type = guardian
          remove_relation_ward = scope:ucm_bulk_ward
          scope:ucm_bulk_ward = {
            remove_relation_guardian = prev
          }
        }
      }
    }
  }
}

# 監護人 scope:ucm_chosen_guardian：拆掉其名下全部學童，避免一人多童殘留
ucm_bulk_guardian_strip_all_wards_from_tutor_effect = {
  hidden_effect = {
    scope:ucm_chosen_guardian = {
      while = {
        count = 30
        limit = {
          any_relation = {
            type = ward
          }
        }
        every_relation = {
          type = ward
          remove_relation_guardian = scope:ucm_chosen_guardian
          scope:ucm_chosen_guardian = {
            remove_relation_ward = prev
          }
        }
      }
    }
  }
}

ucm_bulk_guardian_set_axis_for_ward_effect = {
  scope:ucm_bulk_ward = {
    if = {
      limit = {
        has_trait = rowdy
      }
      set_variable = {
        name = ucm_guar_axis
        value = 1
      }
    }
    else_if = {
      limit = {
        has_trait = bossy
      }
      set_variable = {
        name = ucm_guar_axis
        value = 1
      }
    }
    else_if = {
      limit = {
        has_trait = charming
      }
      set_variable = {
        name = ucm_guar_axis
        value = 4
      }
    }
    else_if = {
      limit = {
        has_trait = curious
      }
      set_variable = {
        name = ucm_guar_axis
        value = 5
      }
    }
    else_if = {
      limit = {
        has_trait = pensive
      }
      set_variable = {
        name = ucm_guar_axis
        value = 3
      }
    }
    else = {
      set_variable = {
        name = ucm_guar_axis
        value = 3
      }
    }
  }
}

ucm_bulk_guardian_pick_and_assign_for_ward_effect = {
  # 再清一次學童側（流水線開頭已清；此處雙保險）
  ucm_bulk_guardian_strip_all_guardians_from_ward_effect = yes
  if = {
    limit = {
      root = {
        any_courtier = {
          ucm_bulk_guardian_candidate_trigger = yes
        }
      }
    }
    root = {
      ordered_courtier = {
        limit = {
          ucm_bulk_guardian_candidate_trigger = yes
        }
        order_by = ucm_guardian_pick_score_value
        max = 1
        check_range_bounds = no
        save_scope_as = ucm_chosen_guardian
      }
    }
    ucm_bulk_guardian_strip_all_wards_from_tutor_effect = yes
    scope:ucm_bulk_ward = {
      set_relation_guardian = {
        target = scope:ucm_chosen_guardian
      }
    }
  }
}

ucm_bulk_guardian_run_ward_pipeline_effect = {
  save_scope_as = ucm_bulk_ward
  ucm_bulk_guardian_strip_all_guardians_from_ward_effect = yes
  ucm_bulk_guardian_set_axis_for_ward_effect = yes
  ucm_bulk_guardian_apply_focus_for_ward_effect = yes
  if = {
    limit = {
      scope:ucm_bulk_ward = {
        ucm_bulk_guardian_ward_needs_guardian_relation_trigger = yes
      }
    }
    ucm_bulk_guardian_pick_and_assign_for_ward_effect = yes
  }
  if = {
    limit = {
      scope:ucm_bulk_ward = {
        has_variable = ucm_guar_axis
      }
    }
    scope:ucm_bulk_ward = {
      remove_variable = ucm_guar_axis
    }
  }
}

  root = {
    every_courtier = {
      limit = {
        ucm_bulk_guardian_ward_at_court_trigger = yes
      }
      ucm_bulk_guardian_run_ward_pipeline_effect = yes
    }
    every_child = {
      limit = {
        ucm_bulk_guardian_ward_player_off_court_child_trigger = yes
      }
      ucm_bulk_guardian_run_ward_pipeline_effect = yes
    }
  }
}

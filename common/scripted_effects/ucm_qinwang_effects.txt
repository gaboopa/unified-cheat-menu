# Know-secrets interaction window; while Chamber Watch is on, gives spouses and concubines the fidelity traits

ucm_seduce_shield_grant_chastity_traits_to_consorts_effect = {
  root = {
    every_spouse = {
      limit = {
        is_alive = yes
      }
      if = {
        limit = {
          NOT = { has_trait = chaste }
        }
        add_trait = chaste
      }
      if = {
        limit = {
          NOT = { has_trait = loyal }
        }
        add_trait = loyal
      }
    }
    every_concubine = {
      limit = {
        is_alive = yes
      }
      if = {
        limit = {
          NOT = { has_trait = chaste }
        }
        add_trait = chaste
      }
      if = {
        limit = {
          NOT = { has_trait = loyal }
        }
        add_trait = loyal
      }
    }
  }
}

# Chamber Watch: end lover relations between spouses/concubines and anyone but the head of household (root), including AI, so events and secrets can't get around the scheme block
ucm_seduce_shield_strip_non_patron_lovers_from_consorts_effect = {
  root = {
    every_spouse = {
      limit = {
        is_alive = yes
      }
      save_scope_as = ucm_lover_strip_ward
      every_relation = {
        type = lover
        limit = {
          NOT = {
            this = root
          }
        }
        scope:ucm_lover_strip_ward = {
          remove_relation_lover = this
        }
      }
    }
    every_concubine = {
      limit = {
        is_alive = yes
      }
      save_scope_as = ucm_lover_strip_ward
      every_relation = {
        type = lover
        limit = {
          NOT = {
            this = root
          }
        }
        scope:ucm_lover_strip_ward = {
          remove_relation_lover = this
        }
      }
    }
  }
}

# Chamber Watch global rule: all AI end seduce/courting/elope schemes, and married characters lose extramarital lovers
ucm_seduce_shield_enforce_global_ai_fidelity_rules_effect = {
  if = {
    limit = { ucm_seduce_shield_global_ai_rules_active_trigger = yes }
    every_living_character = {
      limit = {
        is_ai = yes
        is_adult = yes
      }
      save_scope_as = ucm_global_fid_ai
      every_scheme = {
        limit = {
          scheme_owner = scope:ucm_global_fid_ai
          OR = {
            scheme_type = seduce
            scheme_type = courting
            scheme_type = elope
          }
        }
        end_scheme = yes
      }
      every_targeting_scheme = {
        limit = {
          OR = {
            scheme_type = seduce
            scheme_type = courting
            scheme_type = elope
          }
          scheme_owner = { is_ai = yes }
        }
        end_scheme = yes
      }
      if = {
        limit = { is_married = yes }
        every_relation = {
          type = lover
          limit = { NOT = { is_consort_of = scope:ucm_global_fid_ai } }
          scope:ucm_global_fid_ai = { remove_relation_lover = this }
        }
      }
    }
  }
}

ucm_seduce_shield_upkeep_effect = {
  ucm_seduce_shield_grant_chastity_traits_to_consorts_effect = yes
  ucm_seduce_shield_strip_non_patron_lovers_from_consorts_effect = yes
  ucm_seduce_shield_enforce_global_ai_fidelity_rules_effect = yes
}

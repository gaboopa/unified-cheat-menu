# UCM / 文化编辑：advcm 式动态文化选择窗

ucm_restore_ucm_selected_culture_scope_effect = {
  if = {
    limit = { has_variable = ucm_selected_culture_var }
    var:ucm_selected_culture_var = {
      save_scope_as = ucm_selected_culture
    }
  }
}

ucm_save_ucm_selected_culture_effect = {
  scope:selected_culture = {
    save_scope_as = ucm_selected_culture
  }
  set_variable = {
    name = ucm_selected_culture_var
    value = scope:selected_culture
  }
}

# GUI callbacks run on the player. Apply the selected culture to the character
# whose interaction menu opened the picker (a direct set_culture call),
# while keeping the persistent target lock.
ucm_acm_apply_selected_culture_to_target_effect = {
  if = {
    limit = { NOT = { exists = scope:selected_culture } }
    ucm_restore_ucm_selected_culture_scope_effect = yes
  }
  if = {
    limit = {
      exists = scope:selected_culture
      exists = global_var:ucm_cheat_menu_recipient
    }
    global_var:ucm_cheat_menu_recipient = {
      if = {
        limit = { NOT = { culture = scope:selected_culture } }
        set_culture = scope:selected_culture
      }
    }
  }
  else_if = {
    limit = { exists = scope:selected_culture }
    set_culture = scope:selected_culture
  }
}

ucm_open_acm_culture_picker_effect = {
  # The picker is always owned by the player/root character.  The character
  # being edited is kept separately by ucm_cheat_menu_recipient.
  ucm_cheat_menu_sync_target_effect = yes
  if = {
    limit = {
      exists = scope:ucm_cheat_menu_target
      scope:ucm_cheat_menu_target = { exists = culture }
    }
    scope:ucm_cheat_menu_target = {
      culture = { save_scope_as = selected_culture }
    }
    ucm_save_ucm_selected_culture_effect = yes
  }
  ucm_acm_culture_picker_refresh_all_region_lists_effect = yes
  add_character_flag = ucm_acm_culture_picker_open
}

ucm_close_acm_culture_picker_effect = {
  remove_character_flag = ucm_acm_culture_picker_open
}

# 新开局/读档：清文化编辑挂起，避免桥接窗误开空 culture_window
ucm_acm_culture_gui_startup_cleanup_effect = {
  remove_character_flag = ucm_acm_culture_picker_open
  remove_character_flag = ucm_culture_vanilla_edit_toolbar
  remove_character_flag = ucm_cw_tradition_picker_open
  if = {
    limit = { has_variable = ucm_selected_culture_var }
    remove_variable = ucm_selected_culture_var
  }
}

ucm_acm_open_culture_vanilla_editor_effect = {
  ucm_restore_ucm_selected_culture_scope_effect = yes
  if = {
    limit = { exists = scope:ucm_selected_culture }
    add_character_flag = ucm_culture_vanilla_edit_toolbar
    scope:ucm_selected_culture = {
      open_view_data = {
        view = culture_window
      }
    }
  }
}

ucm_acm_apply_selected_culture_and_open_vanilla_effect = {
  scope:selected_culture = {
    save_scope_as = ucm_selected_culture
  }
  set_variable = {
    name = ucm_selected_culture_var
    value = scope:selected_culture
  }
  ucm_close_acm_culture_picker_effect = yes
  ucm_acm_open_culture_vanilla_editor_effect = yes
}

# One-click exam: at least one landed vassal can be processed (any non-meritocratic landed vassal, or meritocratic below max rank); not the Emperor of China

ucm_has_any_merit_official_to_promote_trigger = {
  any_vassal_or_below = {
    is_adult = yes
    NOT = { has_trait = incapable }
    is_imprisoned = no
    NOT = { is_at_war_with_liege = yes }
    is_landed = yes
    ucm_is_not_h_china_emperor_title_holder_trigger = yes
    OR = {
      NOT = { government_has_flag = government_has_merit }
      AND = {
        government_has_flag = government_has_merit
        merit_level < maximum_merit_level
      }
    }
  }
}

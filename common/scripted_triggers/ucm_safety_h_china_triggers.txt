# Emperor of China title: when not registered through the DLC or load, a direct has_title = title:h_china can fail in native code in some setups; the right-click menu prechecks this a lot.
# Check exists = title:h_china first, then who holds it.

ucm_is_h_china_emperor_title_holder_trigger = {
  exists = title:h_china
  has_title = title:h_china
}

ucm_is_not_h_china_emperor_title_holder_trigger = {
  OR = {
    NOT = { exists = title:h_china }
    AND = {
      exists = title:h_china
      NOT = { has_title = title:h_china }
    }
  }
}

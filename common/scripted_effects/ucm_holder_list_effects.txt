# Character scope. Records the character in a global holders list (once) so monthly pulses
# only visit characters that actually have the cheat instead of scanning the whole world.
ucm_holder_list_add_effect = {
  if = {
    limit = {
      NOT = { is_target_in_global_variable_list = { name = $LIST$ target = this } }
    }
    add_to_global_variable_list = { name = $LIST$ target = this }
  }
}

# Run $UPKEEP$ on the list's living holders of $TRAIT$; collect and then remove the others.
ucm_holder_upkeep_effect = {
  every_in_global_list = {
    variable = $LIST$
    if = {
      limit = { is_alive = yes has_trait = $TRAIT$ }
      $UPKEEP$ = yes
    }
    else = { add_to_temporary_list = ucm_stale_$LIST$ }
  }
  every_in_list = {
    list = ucm_stale_$LIST$
    remove_list_global_variable = { name = $LIST$ target = this }
  }
}

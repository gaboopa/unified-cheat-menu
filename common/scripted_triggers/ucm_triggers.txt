# Unified Cheat Menu: shared triggers.

# The mod is single-player only. Every interaction, decision, scripted GUI and
# cheat pulse checks this. GUI-only elements use [Not( GameHasMultiplePlayers )].
ucm_single_player_trigger = {
	has_multiple_players = no
}

# Group actions: is this member (a county or character) one the action would change?
# Root = player, see ucm_effects.txt. Character groups never include the target itself.
ucm_group_member_eligible_trigger = {
	trigger_if = {
		limit = { root.var:ucm_group_action = flag:max_development }
		development_level < 100
	}
	trigger_else_if = {
		limit = { root.var:ucm_group_action = flag:max_control }
		county_control < 100
	}
	trigger_else_if = {
		limit = { root.var:ucm_group_action = flag:county_faith }
		NOT = { faith = root.var:ucm_target.faith }
	}
	trigger_else_if = {
		limit = { root.var:ucm_group_action = flag:county_culture }
		NOT = { culture = root.var:ucm_target.culture }
	}
	trigger_else_if = {
		limit = { root.var:ucm_group_action = flag:clear_stress }
		is_alive = yes
		NOT = { this = root.var:ucm_target }
		stress > 0
	}
	trigger_else_if = {
		limit = { root.var:ucm_group_action = flag:convert_faith }
		is_alive = yes
		NOT = { this = root }
		NOT = { this = root.var:ucm_target }
		NOT = { faith = root.var:ucm_target.faith }
	}
	trigger_else_if = {
		limit = { root.var:ucm_group_action = flag:convert_culture }
		is_alive = yes
		NOT = { this = root }
		NOT = { this = root.var:ucm_target }
		NOT = { culture = root.var:ucm_target.culture }
	}
	trigger_else_if = {
		limit = {
			OR = {
				root.var:ucm_group_action = flag:skills_up
				root.var:ucm_group_action = flag:gold
				root.var:ucm_group_action = flag:max_opinion
			}
		}
		is_alive = yes
		NOT = { this = root.var:ucm_target }
	}
	trigger_else = { always = no }
}

# Does the picked group hold counties (rather than characters)?
ucm_group_pick_is_county_trigger = {
	OR = {
		var:ucm_group_pick = flag:sub_realm_counties
		var:ucm_group_pick = flag:dejure_counties
		var:ucm_group_pick = flag:culture_counties
		var:ucm_group_pick = flag:faith_counties
	}
}

# Main menu buttons that act on the clicked target.
ucm_menu_target_valid_trigger = {
	exists = var:ucm_target
	var:ucm_target = { is_alive = yes }
}

# Buttons that act on the target and the second character together.
ucm_menu_pair_valid_trigger = {
	ucm_menu_target_valid_trigger = yes
	exists = global_var:ucm_crozier_cheats_other
	global_var:ucm_crozier_cheats_other = { is_alive = yes }
	NOT = { global_var:ucm_crozier_cheats_other = var:ucm_target }
}

# Making the second character a parent of the target must not create a loop:
# the target can't already be one of the second character's ancestors.
# ponytail: any_ancestor only looks 5 generations up; deeper loops aren't caught.
ucm_second_can_be_parent_trigger = {
	ucm_menu_pair_valid_trigger = yes
	NOT = {
		global_var:ucm_crozier_cheats_other = {
			any_ancestor = { this = root.var:ucm_target }
		}
	}
}

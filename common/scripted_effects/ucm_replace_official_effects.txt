# Skonester菜单 · 更换官员：granter=玩家；统一由旧官领主 resolve 交接，先郡后主衔，新官授衔剥余

ucm_replace_official_begin_handover_title_change_effect = {
  if = {
    limit = { exists = scope:ucm_ro_handover_granter }
    scope:ucm_ro_handover_granter = {
      create_title_and_vassal_change = {
        type = granted
        save_scope_as = ucm_ro_title_change
        add_claim_on_loss = no
      }
    }
  }
  else = {
    scope:ucm_ro_granter = {
      create_title_and_vassal_change = {
        type = granted
        save_scope_as = ucm_ro_title_change
        add_claim_on_loss = no
      }
    }
  }
}

ucm_replace_official_resolve_handover_title_change_effect = {
  if = {
    limit = { exists = scope:ucm_ro_handover_granter }
    scope:ucm_ro_handover_granter = {
      resolve_title_and_vassal_change = scope:ucm_ro_title_change
    }
  }
  else = {
    scope:ucm_ro_granter = {
      resolve_title_and_vassal_change = scope:ucm_ro_title_change
    }
  }
}

ucm_replace_official_revoke_xfer_to_handover_granter_effect = {
  if = {
    limit = {
      exists = scope:ucm_ro_xfer_title
      scope:ucm_ro_xfer_title = {
        exists = holder
      }
      OR = {
        exists = scope:ucm_ro_handover_granter
        exists = scope:ucm_ro_granter
      }
    }
    if = {
      limit = {
        exists = scope:ucm_ro_handover_granter
      }
      scope:ucm_ro_handover_granter = {
        create_title_and_vassal_change = {
          type = revoked
          save_scope_as = ucm_ro_title_change
          add_claim_on_loss = no
        }
      }
      scope:ucm_ro_xfer_title = {
        change_title_holder = {
          holder = scope:ucm_ro_handover_granter
          change = scope:ucm_ro_title_change
        }
      }
      scope:ucm_ro_handover_granter = {
        resolve_title_and_vassal_change = scope:ucm_ro_title_change
      }
    }
    else = {
      scope:ucm_ro_granter = {
        create_title_and_vassal_change = {
          type = revoked
          save_scope_as = ucm_ro_title_change
          add_claim_on_loss = no
        }
      }
      scope:ucm_ro_xfer_title = {
        change_title_holder = {
          holder = scope:ucm_ro_granter
          change = scope:ucm_ro_title_change
        }
      }
      scope:ucm_ro_granter = {
        resolve_title_and_vassal_change = scope:ucm_ro_title_change
      }
    }
  }
}

ucm_replace_official_absorb_strip_title_to_granter_effect = {
  if = {
    limit = {
      exists = scope:ucm_ro_xfer_title
      exists = scope:ucm_ro_strip_holder
      scope:ucm_ro_xfer_title = {
        holder = scope:ucm_ro_strip_holder
        ucm_replace_official_transferable_title_trigger = yes
      }
      NOR = {
        AND = {
          exists = scope:ucm_ro_evt_victim
          scope:ucm_ro_strip_holder = scope:ucm_ro_evt_victim
          exists = scope:ucm_ro_evt_victim.primary_title
          scope:ucm_ro_xfer_title = scope:ucm_ro_evt_victim.primary_title
        }
        AND = {
          exists = scope:ucm_ro_evt_victim
          scope:ucm_ro_strip_holder = scope:ucm_ro_evt_victim
          exists = scope:ucm_ro_victim_main_title
          scope:ucm_ro_xfer_title = scope:ucm_ro_victim_main_title
        }
        AND = {
          exists = scope:ucm_ro_evt_victim
          scope:ucm_ro_strip_holder = scope:ucm_ro_evt_victim
          exists = scope:ucm_ro_handover_county
          scope:ucm_ro_xfer_title = scope:ucm_ro_handover_county
        }
      }
      OR = {
        scope:ucm_ro_xfer_title = {
          OR = {
            tier = tier_county
            tier = tier_barony
          }
        }
        AND = {
          exists = scope:ucm_land_cap_grantee
          exists = scope:ucm_strip_title
          ucm_replace_official_grant_safe_for_realm_trigger = yes
        }
      }
    }
    if = {
      limit = {
        exists = scope:ucm_ro_replacement
        scope:ucm_ro_replacement = {
          is_landed = yes
        }
        NOT = {
          scope:ucm_ro_strip_holder = scope:ucm_ro_replacement
        }
        scope:ucm_ro_xfer_title = {
          OR = {
            tier = tier_county
            tier = tier_barony
          }
        }
      }
      scope:ucm_ro_replacement = {
        create_title_and_vassal_change = {
          type = revoked
          save_scope_as = ucm_ro_title_change
          add_claim_on_loss = no
        }
      }
      scope:ucm_ro_xfer_title = {
        change_title_holder = {
          holder = scope:ucm_ro_replacement
          change = scope:ucm_ro_title_change
        }
      }
      scope:ucm_ro_replacement = {
        resolve_title_and_vassal_change = scope:ucm_ro_title_change
      }
    }
    else_if = {
      limit = {
        exists = scope:ucm_ro_regrant_authority
        NOT = {
          scope:ucm_ro_strip_holder = scope:ucm_ro_regrant_authority
        }
      }
      scope:ucm_ro_regrant_authority = {
        create_title_and_vassal_change = {
          type = revoked
          save_scope_as = ucm_ro_title_change
          add_claim_on_loss = no
        }
      }
      scope:ucm_ro_xfer_title = {
        change_title_holder = {
          holder = scope:ucm_ro_regrant_authority
          change = scope:ucm_ro_title_change
        }
      }
      scope:ucm_ro_regrant_authority = {
        resolve_title_and_vassal_change = scope:ucm_ro_title_change
      }
    }
    else = {
      if = {
        limit = {
          scope:ucm_ro_xfer_title = {
            OR = {
              tier = tier_county
              tier = tier_barony
            }
          }
        }
        root = {
          destroy_title = scope:ucm_ro_xfer_title
        }
      }
    }
  }
}

# 受封人与郡/男爵法理属新官，却仍是玩家或旧官直属 → 改宣誓新官
ucm_replace_official_attach_grantee_under_replacement_effect = {
  if = {
    limit = {
      exists = scope:ucm_ro_grant_target
      scope:ucm_ro_grant_target = {
        ucm_replace_official_grantee_should_serve_replacement_trigger = yes
      }
    }
    scope:ucm_ro_replacement = {
      create_title_and_vassal_change = {
        type = swear_fealty
        save_scope_as = ucm_ro_attach_change
        add_claim_on_loss = no
      }
    }
    scope:ucm_ro_grant_target = {
      change_liege = {
        liege = scope:ucm_ro_replacement
        change = scope:ucm_ro_attach_change
      }
    }
    scope:ucm_ro_replacement = {
      resolve_title_and_vassal_change = scope:ucm_ro_attach_change
    }
    clear_saved_scope = ucm_ro_attach_change
  }
}

# 将误从封臣（非旧官）手中剥夺的郡归还快照持有人；scope:ucm_ro_xfer_title = 目标郡
ucm_replace_official_restore_one_county_to_snapshot_holder_effect = {
  if = {
    limit = {
      exists = scope:ucm_ro_xfer_title
      exists = scope:ucm_ro_granter
      scope:ucm_ro_xfer_title = {
        tier = tier_county
        has_variable = ucm_ro_snap_holder
        exists = var:ucm_ro_snap_holder
        exists = holder
        var:ucm_ro_snap_holder = {
          is_alive = yes
          NOT = { this = scope:ucm_ro_evt_victim }
        }
        NOT = { holder = var:ucm_ro_snap_holder }
        trigger_if = {
          limit = { exists = scope:ucm_ro_handover_county }
          NOT = { this = scope:ucm_ro_handover_county }
        }
      }
    }
    scope:ucm_ro_xfer_title = {
      var:ucm_ro_snap_holder = {
        save_scope_as = ucm_ro_restore_holder
      }
    }
    ucm_replace_official_begin_regrant_title_change_effect = yes
    scope:ucm_ro_xfer_title = {

    scope:ucm_ro_granter = {
      random_vassal_or_below = {
        limit = {
          is_landed = yes
          NOT = { this = scope:ucm_off }
          NOT = { this = scope:ucm_ro_replacement }
          ucm_replace_official_can_be_granted_by_regrant_authority_trigger = yes
        }
        save_scope_as = ucm_land_cap_grantee
      }
    }
  }
  if = {
    limit = {
      NOT = { exists = scope:ucm_land_cap_grantee }
      exists = scope:ucm_ro_replacement
      scope:ucm_ro_replacement = {
        is_landed = yes
      }
      exists = scope:ucm_strip_title
      scope:ucm_strip_title = {
        OR = {
          tier = tier_county
          tier = tier_barony
        }
      }
      exists = scope:ucm_off
      scope:ucm_off = {
        exists = top_liege
        top_liege = {
          exists = primary_title
          primary_title.tier > scope:ucm_strip_title.tier
        }
      }
    }
    scope:ucm_ro_replacement = {
      create_character = {
        employer = scope:ucm_ro_replacement
        culture = scope:ucm_off.culture
        faith = scope:ucm_off.faith
        gender_female_chance = 50
        age = 28
        dynasty = none
        save_scope_as = ucm_land_cap_grantee
      }
    }
  }
  else_if = {
    limit = {
      NOT = { exists = scope:ucm_land_cap_grantee }
      exists = scope:ucm_ro_regrant_authority
      exists = scope:ucm_strip_title
      scope:ucm_strip_title = {
        OR = {
          tier = tier_county
          tier = tier_barony
        }
      }
      exists = scope:ucm_off
      scope:ucm_off = {
        exists = top_liege
        top_liege = {
          exists = primary_title
          primary_title.tier > scope:ucm_strip_title.tier
        }
      }
    }
    scope:ucm_ro_regrant_authority = {
      create_character = {
        employer = scope:ucm_ro_regrant_authority
        culture = scope:ucm_off.culture
        faith = scope:ucm_off.faith
        gender_female_chance = 50
        age = 28
        dynasty = none
        save_scope_as = ucm_land_cap_grantee
      }
    }
  }
}

# 新官须先挂到旧官直属领主（无领主时挂 granter/玩家）；无地者先入廷再宣誓
ucm_replace_official_prepare_replacement_under_granter_effect = {
  if = {
    limit = {
      exists = scope:ucm_ro_replacement
      scope:ucm_ro_replacement = {
        is_alive = yes
        is_landed = no
      }
    }
    if = {
      limit = {
        exists = scope:ucm_ro_victim_liege
        NOT = {
          scope:ucm_ro_replacement = {
            is_courtier_of = scope:ucm_ro_victim_liege
          }
        }
      }
      scope:ucm_ro_replacement = {
        set_employer = scope:ucm_ro_victim_liege
      }
    }
    else_if = {
      limit = {
        exists = scope:ucm_ro_granter
        NOT = {
          scope:ucm_ro_replacement = {
            is_courtier_of = scope:ucm_ro_granter
          }
        }
      }
      scope:ucm_ro_replacement = {
        set_employer = scope:ucm_ro_granter
      }
    }
  }
  if = {
    limit = {
      exists = scope:ucm_ro_replacement
      scope:ucm_ro_replacement = {
        is_alive = yes
        OR = {
          NOT = { exists = liege }
          AND = {
            exists = scope:ucm_ro_victim_liege
            NOT = { liege = scope:ucm_ro_victim_liege }
          }
          AND = {
            NOT = { exists = scope:ucm_ro_victim_liege }
            exists = scope:ucm_ro_granter
            NOT = { liege = scope:ucm_ro_granter }
          }
        }
      }
    }
    if = {
      limit = {
        exists = scope:ucm_ro_victim_liege
      }
      scope:ucm_ro_victim_liege = {
        create_title_and_vassal_change = {
          type = swear_fealty
          save_scope_as = ucm_ro_vassal_change
        }
      }
      scope:ucm_ro_replacement = {
        change_liege = {
          liege = scope:ucm_ro_victim_liege
          change = scope:ucm_ro_vassal_change
        }
      }
      scope:ucm_ro_victim_liege = {
        resolve_title_and_vassal_change = scope:ucm_ro_vassal_change
      }
    }
    else = {
      scope:ucm_ro_granter = {
        create_title_and_vassal_change = {
          type = swear_fealty
          save_scope_as = ucm_ro_vassal_change
        }
      }
      scope:ucm_ro_replacement = {
        change_liege = {
          liege = scope:ucm_ro_granter
          change = scope:ucm_ro_vassal_change
        }
      }
      scope:ucm_ro_granter = {
        resolve_title_and_vassal_change = scope:ucm_ro_vassal_change
      }
    }
  }
}

# 统一路径 + 方案B：先交接→迁封臣→锁定新官授衔→剥余/限地→子树法理→兜底
ucm_replace_official_execute_core_effect = {
  if = {
    limit = {
      exists = scope:ucm_ro_replacement
      exists = scope:ucm_ro_evt_victim
      scope:ucm_ro_replacement != scope:ucm_ro_evt_victim
    }
    ucm_replace_official_resolve_granter_scope_effect = yes
    scope:ucm_ro_evt_victim = {
      if = {
        limit = { exists = liege }
        liege = {
          save_scope_as = ucm_ro_victim_liege
        }
      }
      if = {
        limit = { exists = primary_title }
        primary_title = {
          save_scope_as = ucm_ro_victim_primary
        }
      }
    }
    ucm_replace_official_resolve_handover_granter_scope_effect = yes
    ucm_replace_official_cache_victim_direct_vassals_effect = yes
    scope:ucm_ro_evt_victim = {
      add_character_flag = ucm_ro_grant_excluded
    }
    scope:ucm_ro_replacement = {
      add_character_flag = ucm_ro_grant_excluded
    }
    ucm_replace_official_revoke_victim_leases_effect = yes
    ucm_replace_official_prepare_replacement_under_granter_effect = yes
    ucm_replace_official_save_victim_high_title_effect = yes
    ucm_replace_official_save_victim_main_title_effect = yes
    ucm_replace_official_select_handover_county_effect = yes
    ucm_replace_official_snapshot_county_holders_effect = yes
    # 阶段1：主衔 + 指定交接郡
    ucm_replace_official_execute_handover_sequence_effect = yes
    scope:ucm_ro_replacement = {
      if = {
        limit = {
          exists = scope:ucm_ro_victim_main_title
          scope:ucm_ro_victim_main_title = {
            holder = scope:ucm_ro_replacement
          }
        }
        set_primary_title_to = scope:ucm_ro_victim_main_title
      }
      else_if = {
        limit = {
          exists = scope:ucm_ro_victim_primary
          scope:ucm_ro_victim_primary = {
            holder = scope:ucm_ro_replacement
          }
        }
        set_primary_title_to = scope:ucm_ro_victim_primary
      }
      else_if = {
        limit = {
          exists = scope:ucm_ro_handover_county
          scope:ucm_ro_handover_county = {
            holder = scope:ucm_ro_replacement
          }
        }
        set_primary_title_to = scope:ucm_ro_handover_county
      }
    }
    ucm_replace_official_resolve_replacement_kept_county_effect = yes
    # 阶段2：缓存的旧官直属封臣 → 新官
    ucm_replace_official_preserve_vassal_liege_effect = yes
    ucm_replace_official_transfer_victim_vassals_to_replacement_effect = yes
    # 阶段3：转封授权 = 新官
    ucm_replace_official_lock_regrant_authority_to_replacement_effect = yes
    # 阶段4：剥旧官余衔 + 新官限地（由新官授衔，受封人挂新官）
    ucm_replace_official_restore_stolen_counties_from_snapshot_effect = yes
    ucm_replace_official_trim_victim_extra_personal_counties_effect = yes
    ucm_replace_official_revoke_victim_remaining_to_granter_effect = yes
    ucm_replace_official_enforce_replacement_land_cap_effect = yes
    ucm_replace_official_redistribute_granter_stray_counties_effect = yes
    # 阶段5：新官子树法理转封 + 误挂扫回
    ucm_replace_official_post_handover_de_jure_transfer_effect = yes
    # 阶段6：轻量兜底
    ucm_replace_official_salvage_execute_failures_effect = yes
    ucm_replace_official_post_cleanup_effect = yes
    ucm_replace_official_mark_office_holder_flags_effect = yes
    ucm_replace_official_clear_county_snapshot_effect = yes
    ucm_replace_official_clear_victim_vassal_queue_effect = yes
    scope:ucm_ro_evt_victim = {
      remove_character_flag = ucm_ro_grant_excluded
    }
    scope:ucm_ro_replacement = {
      remove_character_flag = ucm_ro_grant_excluded
    }
    clear_saved_scope = ucm_ro_granter
    clear_saved_scope = ucm_ro_handover_granter
    clear_saved_scope = ucm_ro_regrant_authority
    clear_saved_scope = ucm_ro_victim_liege
    clear_saved_scope = ucm_ro_victim_primary
    clear_saved_scope = ucm_ro_victim_main_title
    clear_saved_scope = ucm_ro_victim_high_title
    clear_saved_scope = ucm_ro_handover_county
    clear_saved_scope = ucm_ro_kept_county
    clear_saved_scope = ucm_ro_xfer_title

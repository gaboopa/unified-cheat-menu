ucm_clr_any_secrets_hint = {
  type = character
  random_valid = no
  text = {
    trigger = {
      OR = {
        any_secret = { always = yes }
        any_targeting_secret = { always = yes }
        any_known_secret = { always = yes }
      }
    }
    localization_key = ucm_zw_empty_loc
  }
  text = {
    trigger = { always = yes }
    localization_key = ucm_clear_secrets_no_secrets_hint_loc
    fallback = yes
  }
}

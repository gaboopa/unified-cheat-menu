# 无限补给·零损耗：战争与每月为已开启者拉满全部军团补给（不改编制人数）

ucm_monthly_tick = {
  on_actions = { ucm_military_supply_monthly_pulse }
}

ucm_military_supply_monthly_pulse = {
	trigger = { ucm_single_player_trigger = yes }
  effect = {
    # One-time migration for saves made before the holders list existed.
    if = {
      limit = { NOT = { has_global_variable = ucm_supply_list_built } }
      set_global_variable = { name = ucm_supply_list_built value = 1 }
      every_living_character = {
        limit = { has_character_modifier = ucm_military_instant_replenish_modifier }
        ucm_holder_list_add_effect = { LIST = ucm_supply_holders }
      }
    }
    every_in_global_list = {
      variable = ucm_supply_holders
      limit = {
        is_alive = yes
        has_character_modifier = ucm_military_instant_replenish_modifier
        has_character_flag = ucm_military_cheat_player_authorized
      }
      ucm_cheat_menu_refill_all_army_supply_on_character_effect = yes
      ucm_cheat_menu_apply_supply_overflow_stacks_effect = yes
    }
    # Collect stale entries first, remove after iterating (never edit a list while iterating it).
    every_in_global_list = {
      variable = ucm_supply_holders
      limit = {
        OR = {
          is_alive = no
          NOT = { has_character_modifier = ucm_military_instant_replenish_modifier }
        }
      }
      add_to_temporary_list = ucm_stale_ucm_supply_holders
    }
    every_in_list = {
      list = ucm_stale_ucm_supply_holders
      remove_list_global_variable = { name = ucm_supply_holders target = this }
    }
  }
}

on_war_started = {
  on_actions = { ucm_military_supply_war_started_pulse }
}

ucm_military_supply_war_started_pulse = {
	trigger = { ucm_single_player_trigger = yes }
  effect = {
    scope:attacker ?= {
      if = {
        limit = {
          is_alive = yes
          has_character_modifier = ucm_military_instant_replenish_modifier
          has_character_flag = ucm_military_cheat_player_authorized
        }
        ucm_cheat_menu_refill_all_army_supply_on_character_effect = yes
        ucm_cheat_menu_start_supply_war_loop_effect = yes
      }
    }
    scope:defender ?= {
      if = {
        limit = {
          is_alive = yes
          has_character_modifier = ucm_military_instant_replenish_modifier
          has_character_flag = ucm_military_cheat_player_authorized
        }
        ucm_cheat_menu_refill_all_army_supply_on_character_effect = yes
        ucm_cheat_menu_start_supply_war_loop_effect = yes
      }
    }
  }
}

# UCM菜单 · 强牵制：5 年内禁止入派系、禁止对宗主（玩家）宣战

ucm_cheat_menu_strong_deterrence_leave_factions_vs_liege_effect = {
  if = {
    limit = {
      exists = joined_faction
      joined_faction = {
        faction_target = scope:ucm_deterrence_liege
      }
    }
    if = {
      limit = {
        joined_faction = { exists = faction_war }
      }
      joined_faction = {
        faction_war = { end_war = white_peace }
      }
    }
    leave_faction = joined_faction
  }
  every_targeting_faction = {
    limit = { faction_target = scope:ucm_deterrence_liege }
    if = {
      limit = { exists = faction_war }
      faction_war = { end_war = white_peace }
    }
    destroy_faction = yes
  }
}

ucm_cheat_menu_strong_deterrence_end_wars_vs_liege_effect = {
  every_character_war = {
    limit = {
      OR = {
        AND = {
          primary_attacker = scope:ucm_deterrence_vassal
          primary_defender = scope:ucm_deterrence_liege
        }
        AND = {
          primary_attacker = scope:ucm_deterrence_liege
          primary_defender = scope:ucm_deterrence_vassal
        }
      }
    }
    end_war = white_peace
  }
}

ucm_cheat_menu_strong_deterrence_give_liege_hook_effect = {
  if = {
    limit = {
      can_add_hook = {
        target = scope:ucm_deterrence_vassal
        type = strong_obligation_hook
      }
    }
    add_hook = {
      target = scope:ucm_deterrence_vassal
      type = strong_obligation_hook
    }
  }
}

ucm_cheat_menu_strong_deterrence_remove_liege_hook_effect = {
  if = {
    limit = { has_strong_hook = scope:ucm_deterrence_vassal }
    remove_hook = {
      target = scope:ucm_deterrence_vassal
      type = strong_obligation_hook
    }
  }
}

# 维护 / 到期清理（root = 宗主，在 monthly_playable_pulse 中执行）
ucm_cheat_menu_strong_deterrence_monthly_on_liege_effect = {
  root = { save_scope_as = ucm_deterrence_liege }
  every_vassal_or_below = {
    limit = { is_alive = yes }
    save_scope_as = ucm_deterrence_vassal
    if = {
      limit = {
        has_character_flag = ucm_strong_deterrence_active
        var:ucm_strong_deterrence_liege ?= scope:ucm_deterrence_liege
      }
      ucm_cheat_menu_strong_deterrence_leave_factions_vs_liege_effect = yes
      ucm_cheat_menu_strong_deterrence_end_wars_vs_liege_effect = yes
      scope:ucm_deterrence_liege = {
        ucm_cheat_menu_strong_deterrence_give_liege_hook_effect = yes
      }
    }
    else_if = {
      limit = {
        var:ucm_strong_deterrence_liege ?= scope:ucm_deterrence_liege
        scope:ucm_deterrence_liege = { has_strong_hook = scope:ucm_deterrence_vassal }
      }
      scope:ucm_deterrence_liege = {
        ucm_cheat_menu_strong_deterrence_remove_liege_hook_effect = yes
      }
      if = {
        limit = { has_variable = ucm_strong_deterrence_liege }
        remove_variable = ucm_strong_deterrence_liege
      }
    }
  }
}

# Records optional mods once at game start, so triggers and windows only read a variable.
on_game_start_after_lobby = {
	on_actions = { ucm_compat_probe }
}

ucm_compat_probe = {
	effect = {
		remove_global_variable = ucm_compat_ek2_loaded
		if = {
			limit = { exists = title:h_tamriel }
			set_global_variable = { name = ucm_compat_ek2_loaded value = 1 }
		}
	}
}

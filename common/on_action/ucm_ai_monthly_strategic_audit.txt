# 每月一检：同步反玩家压力；独立有地 AI 内政/宗教文化/军事派系分项修饰（详见 ucm_ai_monthly_audit_effects）
# 仅追加 ucm_monthly_tick（见 ucm_monthly_tick.txt）；总开关 ucm_ai_yearly_pulse_disabled 关闭时全部停止

ucm_monthly_tick = {
  on_actions = {
    ucm_ai_monthly_independent_ai_audit_pulse
    ucm_ai_world_ai_boost_spread_pulse
    ucm_ai_counter_player_quarterly_refresh_pulse
    ucm_ai_player_vassal_opposition_quarterly_pulse
  }
}

ucm_ai_monthly_independent_ai_audit_pulse = {
	trigger = {
		ucm_single_player_trigger = yes
		NOT = { has_global_variable = ucm_ai_yearly_pulse_disabled }
	}
  effect = {
    # 性能：原逻辑每月对所有独立 AI 逐个触发 9210，后期国家多时会造成明显月初卡顿。
    # 改为「约每季度执行一次」的整批审计（平均降低 ~66% 触发与遍历成本）。
    random = {
      chance = 34
      every_independent_ruler = {
        limit = {
          is_ai = yes
          is_landed = yes
        }
        trigger_event = {
          id = ucm_officials.9210
          days = 0
        }
      }
    }
  }
}

# 反玩家压力：维持“更勤外交结盟/更积极评估对抗”的加压修正。
# 为避免月初固定卡顿，这里用约季度概率刷新；年度脉冲仍会兜底刷新一次。
ucm_ai_counter_player_quarterly_refresh_pulse = {
	trigger = {
		ucm_single_player_trigger = yes
		NOT = { has_global_variable = ucm_ai_yearly_pulse_disabled }
	}
  effect = {
    random = {
      chance = 34
      ucm_ai_apply_counter_player_pressure_global_effect = yes
    }
  }
}

# 玩家领国内：AI 封臣“更反对玩家”（覆盖全部直辖封臣，但用分批抽样避免一次性全扫）
ucm_ai_player_vassal_opposition_quarterly_pulse = {
	trigger = {
		ucm_single_player_trigger = yes
		NOT = { has_global_variable = ucm_ai_yearly_pulse_disabled }
	}
  effect = {
    random = {
      chance = 34
      every_independent_ruler = {
        limit = {
          is_ai = no
          is_landed = yes
          realm_size > 10
        }
        # 每个玩家领主：每轮随机抽样若干直辖封臣加压；已加压者跳过
        while = {
          count = 150
          limit = {
            any_vassal = {
              is_ai = yes
              is_landed = yes
              NOT = { has_character_modifier = ucm_ai_anti_player_vassal_pressure }
            }
          }
          random_vassal = {
            limit = {
              is_ai = yes
              is_landed = yes
              NOT = { has_character_modifier = ucm_ai_anti_player_vassal_pressure }
            }
            weight = { base = 1 }
            add_character_modifier = {
              modifier = ucm_ai_anti_player_vassal_pressure
              months = 12
            }
          }
        }
      }
    }
  }
}

# 强有地AI强化：分批发放（避免决议一次性全图遍历造成卡顿）
# 开启后每月补发一批给尚未获得 modifier 的有地 AI（含封臣 AI），直至覆盖完毕；优先从玩家领地向外扩散。
ucm_ai_world_ai_boost_spread_pulse = {
	trigger = { ucm_single_player_trigger = yes }
  effect = {
    if = {
      limit = {
        NOT = { has_global_variable = ucm_ai_boost_disabled }
        NOT = { has_global_variable = ucm_world_ai_boost_disabled }
      }
      ucm_ai_world_ai_boost_spread_batch_effect = yes
    }
  }
}

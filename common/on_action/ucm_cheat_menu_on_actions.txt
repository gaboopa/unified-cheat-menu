# CK3 1.20: chained via on_actions instead of a second `effect` block (only one effect per on-action survives;
# the collision was dropping vanilla 1.20 logic or this mod logic).
on_pregnancy_mother = {
  on_actions = { ucm_on_pregnancy_mother_clear_bgp }
}

ucm_on_pregnancy_mother_clear_bgp = {
  effect = {
    # 清理“UCM胎儿性别”操作的挂起旗，避免跨胎残留
    remove_character_flag = ucm_bgp_pending
  }
}

# 只挂接 on_actions 列表（可叠加），不直接写 effect，避免覆盖原版/AGOT 的开局 effect
on_game_start = {
  on_actions = { ucm_cheat_menu_game_start }
}

on_game_start_after_lobby = {
  on_actions = { ucm_cheat_menu_after_lobby }
}

ucm_cheat_menu_game_start = {
  effect = {
    ucm_acm_faith_populate_doctrine_catalog_effect = yes
    # 回到主菜单/新开一局时清宝物制作 GUI 挂起（scripted_widget 在部分流程仍会挂载）。玩家=非 AI，故只需 every_player（无需全图扫描）
    every_player = {
      ucm_acm_gui_startup_cleanup_effect = yes
      remove_character_flag = ucm_cheat_menu_cheat_menu
      remove_character_flag = ucm_character_interaction_menu_open
      remove_character_flag = toc_showdebsflag
      remove_character_flag = toc_showspecsflag
      remove_character_flag = toc_showtextrasflag
      remove_character_flag = ucm_acm_gui_open
      remove_character_flag = ucm_acm_repair_gui_open
      remove_character_flag = ucm_trait_editor_open
      remove_character_flag = ucm_lifestyle_open
    }
    if = {
      limit = { has_global_variable = ucm_acm_repair_target }
      remove_global_variable = ucm_acm_repair_target
    }
    if = {
      limit = { has_global_variable = ucm_acm_recipient }
      remove_global_variable = ucm_acm_recipient
    }
    if = {
      limit = { has_global_variable = ucm_cheat_menu_recipient }
      remove_global_variable = ucm_cheat_menu_recipient
    }
  }
}

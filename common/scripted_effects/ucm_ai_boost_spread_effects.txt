# The AI Realm Bonuses toggle: root is the player, global variable ucm_ai_boost_enabled holds the state.
ucm_ai_boost_toggle_effect = {
	if = {
		limit = { has_global_variable = ucm_ai_boost_enabled }
		remove_global_variable = ucm_ai_boost_enabled
		every_living_character = {
			limit = { has_character_modifier = ucm_ai_realm_governance_support }
			remove_character_modifier = ucm_ai_realm_governance_support
		}
	}
	else = {
		set_global_variable = { name = ucm_ai_boost_enabled value = yes }
		every_living_character = {
			limit = {
				is_ai = yes
				is_landed = yes
				NOT = { has_character_modifier = ucm_ai_realm_governance_support }
			}
			add_character_modifier = ucm_ai_realm_governance_support
		}
	}
}

# 新官交接后允许保留：主头衔、唯一指定伯爵领（ucm_ro_kept_county）、该领下男爵
ucm_replace_official_replacement_may_keep_title_trigger = {
  OR = {
    AND = {
      exists = scope:ucm_ro_replacement.primary_title
      this = scope:ucm_ro_replacement.primary_title
    }
    AND = {
      tier = tier_county
      exists = scope:ucm_ro_kept_county
      this = scope:ucm_ro_kept_county
      scope:ucm_ro_kept_county = {
        holder = scope:ucm_ro_replacement
      }
    }
    AND = {
      tier = tier_barony
      exists = scope:ucm_ro_kept_county
      de_jure_liege = scope:ucm_ro_kept_county
      scope:ucm_ro_kept_county = {
        holder = scope:ucm_ro_replacement
      }
    }
  }
}

    }
  }
}

      exists = scope:ucm_ro_granter

        is_ai = no
      }
      ucm_replace_official_cheat_grantee_base_trigger = yes
    }
  }

# 转封后应挂到新官旗下：法理属新官主衔，且当前误挂在玩家/旧官直属
ucm_replace_official_grantee_should_serve_replacement_trigger = {
  exists = scope:ucm_ro_replacement
  scope:ucm_ro_replacement = {
    is_landed = yes
    exists = primary_title
  }
  NOT = { this = scope:ucm_ro_replacement }
  NOT = { liege = scope:ucm_ro_replacement }
  top_liege = scope:ucm_ro_replacement.top_liege
  OR = {
    AND = {
      exists = scope:ucm_ro_granter
      liege = scope:ucm_ro_granter
    }
    AND = {
      exists = scope:ucm_ro_evt_victim
      liege = scope:ucm_ro_evt_victim
    }
    AND = {
      exists = scope:ucm_ro_victim_liege
      liege = scope:ucm_ro_victim_liege
    }
  }
  OR = {
    AND = {
      exists = primary_title
      primary_title = {
        tier = tier_county
        target_is_de_jure_liege_or_above = scope:ucm_ro_replacement.primary_title
      }
    }
    AND = {
      highest_held_title_tier = tier_barony
      any_held_title = {
        tier = tier_barony
        de_jure_liege = {
          holder = scope:ucm_ro_replacement
        }
      }
    }
    any_held_title = {
      tier = tier_county
      target_is_de_jure_liege_or_above = scope:ucm_ro_replacement.primary_title
    }
    any_held_title = {
      tier = tier_county
      de_jure_liege = {
        holder = scope:ucm_ro_replacement
      }
    }
  }
}

# 转封对象须能承接 strip 衔级（limit 内不可用 trigger_else_if）
ucm_replace_official_strip_title_grantee_tier_ok_trigger = {
  OR = {
    AND = {
      exists = scope:ucm_strip_title
      scope:ucm_strip_title = {
        tier = tier_kingdom
      }
      highest_held_title_tier >= tier_kingdom
    }
    AND = {
      exists = scope:ucm_strip_title
      scope:ucm_strip_title = {
        tier = tier_duchy
      }
      highest_held_title_tier >= tier_duchy
    }
    AND = {
      OR = {
        NOT = { exists = scope:ucm_strip_title }
        scope:ucm_strip_title = {
          NOR = {
            tier = tier_kingdom
            tier = tier_duchy
          }
        }
      }
    }
  }
}

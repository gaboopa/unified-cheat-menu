# CK3 1.20: chained via on_actions instead of a second `effect` block (only one effect per on-action survives;
# the collision was dropping vanilla 1.20 logic or this mod logic).
on_marriage = {
  on_actions = { ucm_on_marriage_ban_check }
}

ucm_on_marriage_ban_check = {
	trigger = { ucm_single_player_trigger = yes }
  effect = {
    # 任一方带禁婚或旧版不婚誓约：立即拆散本段婚姻（多配偶则循环离婚）
    if = {
      limit = {
        NOT = { has_character_flag = ucm_free_marriage_ui_prepared }
        OR = {
          has_trait = ucm_trait_marriage_ban
          has_character_flag = ucm_no_marry_vow
        }
      }
      ucm_cheat_menu_marriage_ban_strip_wedlocks_effect = yes
    }
    else_if = {
      limit = {
        exists = primary_spouse
        primary_spouse = {
          NOT = { has_character_flag = ucm_free_marriage_ui_prepared }
          OR = {
            has_trait = ucm_trait_marriage_ban
            has_character_flag = ucm_no_marry_vow
          }
        }
      }
      divorce = primary_spouse
    }
  }
}

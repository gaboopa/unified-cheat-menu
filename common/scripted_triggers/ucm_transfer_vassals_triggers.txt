# Character scope: someone in the realm a vassal (scope:ucm_dj_vassal) can move under. Root = the ruler.
ucm_de_jure_transfer_holder_trigger = {
	NOT = { this = root }
	NOT = { this = scope:ucm_dj_vassal }
	is_vassal_or_below_of = root
	NOT = { is_vassal_or_below_of = scope:ucm_dj_vassal }
	NOT = { is_at_war_with = root }
}

# Vassal scope, root = the ruler. True when a title above the vassal's primary title in the de jure
# chain is held by someone else in the realm, and their liege doesn't already hold one (the ruler doesn't count).
ucm_vassal_needs_de_jure_transfer_trigger = {
	is_landed = yes
	exists = primary_title
	NOT = { is_at_war_with = root }
	save_temporary_scope_as = ucm_dj_vassal
	primary_title = {
		any_this_title_or_de_jure_above = {
			exists = holder
			holder = { ucm_de_jure_transfer_holder_trigger = yes }
		}
		NOT = {
			any_this_title_or_de_jure_above = {
				exists = holder
				holder = scope:ucm_dj_vassal.liege
				NOT = { holder = root }
			}
		}
	}
}

ucm_has_de_jure_transferable_direct_vassals_trigger = {
	any_vassal_or_below = { ucm_vassal_needs_de_jure_transfer_trigger = yes }
}

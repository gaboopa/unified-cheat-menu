# Compatibility: vanilla / Orient Empires / AGOT / EK2
# An on_action probes at game start and writes global variables; features should check the matching trigger before using another mod's IDs.

# Requires OE to be loaded this session (OE's on_game_start writes orient_empires_loaded)
ucm_compat_orient_loaded_trigger = {
  exists = global_var:orient_empires_loaded
}

# Elder Kings 2 (2887120253); written at game start by zzz_ucm_compat_probe
ucm_compat_ek2_loaded_trigger = {
  has_global_variable = ucm_compat_ek2_loaded
}

# TGP ministers: vanilla tgp_is_any_minister is in 10_tgp_triggers.txt; AGOT overrides that file and doesn't have this trigger.
# Called directly, AGOT drops the token and the NOR/NOT/OR structure changes (candidates become always true or always false).
# So check the celestial government flag first (a plain string, parses in both): AGOT has no celestial government, so this is always false and the dropped line doesn't matter;
# in vanilla it's equivalent to tgp_is_any_minister (which already requires a celestial liege). AGOT logs 1 load error, only here.
ucm_is_tgp_minister_trigger = {
  liege ?= { government_has_flag = government_is_celestial }
  tgp_is_any_minister = yes
}

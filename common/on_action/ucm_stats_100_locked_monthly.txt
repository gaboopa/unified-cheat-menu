# 属性100：每月校正持有特质的角色六维（仅追加 ucm_monthly_tick）

ucm_monthly_tick = {
  on_actions = {
    ucm_stats_100_locked_monthly_pulse
  }
}

ucm_stats_100_locked_monthly_pulse = {
	trigger = { ucm_single_player_trigger = yes }
  effect = {
    every_living_character = {
      limit = {
        has_trait = ucm_trait_stats_100_locked
      }
      ucm_stats_100_clamp_all_skills_effect = yes
    }
  }
}

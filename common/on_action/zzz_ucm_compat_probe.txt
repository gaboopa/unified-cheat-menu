# 开局探测可选依赖 mod，供宫廷任免 / 罢免庸官等使用（不修改对方变量）。
# 只挂接 on_actions 列表（可叠加），不直接写 effect，避免覆盖原版/AGOT 的开局 effect
on_game_start_after_lobby = {
  on_actions = { ucm_compat_probe_after_lobby }
}

ucm_compat_probe_after_lobby = {
  effect = {
    if = {
      limit = {
        exists = global_var:orient_empires_loaded
      }
      set_global_variable = {
        name = ucm_compat_orient_loaded
        value = 1
      }
    }
    else = {
      remove_global_variable = ucm_compat_orient_loaded
    }
    if = {
      limit = {
        exists = decision:dfbz_zfz_001_decision
      }
      set_global_variable = {
        name = ucm_compat_orient_expansion_loaded
        value = 1
      }
    }
    else = {
      remove_global_variable = ucm_compat_orient_expansion_loaded
    }
    # A Game of Thrones（2962333032）：e_the_iron_islands 仅存在于 AGOT；GUI 只读此变量，避免逐帧查头衔
    if = {
      limit = {
        exists = title:e_the_iron_islands
      }
      set_global_variable = {
        name = ucm_compat_agot_loaded
        value = 1
      }
    }
    else = {
      remove_global_variable = ucm_compat_agot_loaded
    }
    # Elder Kings 2（2887120253）：h_tamriel 仅存在于 EK2
    if = {
      limit = {
        exists = title:h_tamriel
      }
      set_global_variable = {
        name = ucm_compat_ek2_loaded
        value = 1
      }
    }
    else = {
      remove_global_variable = ucm_compat_ek2_loaded
    }
  }
}

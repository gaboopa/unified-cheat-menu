# 一键任免：在「严格择优」内阁与宫廷之后，对仍空缺的席位先宽松直属封臣择优，再廷臣择优，再 create_character 入廷并任命。
# 须在 scope:ucm_pos_reshuffle_liege 已存在时由 ucm_positions_mass_reshuffle_liege_effect 调用。

ucm_positions_spawn_one_generic_courtier_for_pos_reshuffle_effect = {
  scope:ucm_pos_reshuffle_liege = {
    create_character = {
      employer = scope:ucm_pos_reshuffle_liege
      culture = scope:ucm_pos_reshuffle_liege.culture
      faith = scope:ucm_pos_reshuffle_liege.faith
      gender_female_chance = 50
      age = 28
      random_traits = no
      dynasty = none
      diplomacy = { min_template_average_skill max_template_high_skill }
      martial = { min_template_average_skill max_template_high_skill }
      stewardship = { min_template_average_skill max_template_high_skill }
      intrigue = { min_template_average_skill max_template_high_skill }
      learning = { min_template_average_skill max_template_high_skill }
      prowess = { min_template_average_skill max_template_high_skill }
      save_scope_as = ucm_pos_spawned
    }
  }
}

# 单一宫廷职：严格 can_be_employed_as → 宽松 aptitude → 按职种生成廷臣；多槽循环至满
ucm_positions_try_appoint_court_slot_strict_loose_spawn_effect = {
  scope:ucm_pos_reshuffle_liege = {
    set_local_variable = {
      name = ucm_pos_slot_fill_iter
      value = 0
    }
    while = {
      limit = {
        can_employ_court_position_type = $COURT_POS$
        local_var:ucm_pos_slot_fill_iter < 8
      }
      ucm_positions_try_appoint_court_slot_strict_effect = { COURT_POS = $COURT_POS$ }
      ucm_positions_try_appoint_court_slot_best_vacant_effect = { COURT_POS = $COURT_POS$ }
      if = {
        limit = {
          can_employ_court_position_type = $COURT_POS$
        }
        ucm_positions_spawn_one_generic_courtier_for_pos_reshuffle_effect = yes
        if = {
          limit = {
            exists = scope:ucm_pos_spawned
            scope:ucm_pos_spawned = {
              can_be_employed_as = $COURT_POS$
            }
          }
          appoint_court_position = {
            recipient = scope:ucm_pos_spawned
            court_position = $COURT_POS$
          }
          clear_saved_scope = ucm_pos_spawned
        }
        else_if = {
          limit = { exists = scope:ucm_pos_spawned }
          clear_saved_scope = ucm_pos_spawned
        }
      }
      change_local_variable = {
        name = ucm_pos_slot_fill_iter
        add = 1
      }
    }
  }
}

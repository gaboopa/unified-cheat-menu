# 征兵×100 + 零维护费：每月重同步叠层（征召规模与维护费减免分开叠）

ucm_monthly_tick = {
  on_actions = { ucm_military_levy_overflow_monthly_pulse }
}

ucm_military_levy_overflow_monthly_pulse = {
	trigger = { ucm_single_player_trigger = yes }
  effect = {
    # One-time migration for saves made before the holders list existed.
    if = {
      limit = { NOT = { has_global_variable = ucm_levy_list_built } }
      set_global_variable = { name = ucm_levy_list_built value = 1 }
      every_living_character = {
        limit = { has_character_modifier = ucm_military_levy_overflow_modifier }
        ucm_holder_list_add_effect = { LIST = ucm_levy_holders }
      }
    }
    every_in_global_list = {
      variable = ucm_levy_holders
      limit = {
        is_alive = yes
        has_character_modifier = ucm_military_levy_overflow_modifier
        has_character_flag = ucm_military_cheat_player_authorized
      }
      ucm_cheat_menu_apply_levy_overflow_full_effect = yes
    }
    # Collect stale entries first, remove after iterating (never edit a list while iterating it).
    every_in_global_list = {
      variable = ucm_levy_holders
      limit = {
        OR = {
          is_alive = no
          NOT = { has_character_modifier = ucm_military_levy_overflow_modifier }
        }
      }
      add_to_temporary_list = ucm_stale_ucm_levy_holders
    }
    every_in_list = {
      list = ucm_stale_ucm_levy_holders
      remove_list_global_variable = { name = ucm_levy_holders target = this }
    }
  }
}

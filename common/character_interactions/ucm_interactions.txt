# Unified Cheat Menu: Layer 1 right-click entries (spec Section 2).

ucm_open_cheat_menu = {
	category = interaction_category_uncategorized
	interface_priority = 12100
	common_interaction = yes
	auto_accept = yes
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	icon = debug_chars
	desc = ucm_open_cheat_menu_desc
	is_shown = {
		ucm_single_player_trigger = yes
		scope:actor = { is_ai = no }
		scope:recipient = { is_alive = yes }
	}
	on_accept = {
		ucm_cheat_menu_lock_recipient_effect = yes
		scope:actor = {
			set_variable = { name = ucm_target value = scope:recipient }
			ucm_menu_open_effect = yes
		}
	}
	ai_will_do = 0
}

# Picks one of the recipient's titles; the actor takes it, with or without its vassals.
ucm_take_title = {
	category = interaction_category_uncategorized
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	auto_accept = yes
	interface_priority = 12080
	icon = debug_take_title
	desc = ucm_take_title_desc
	target_type = title
	target_filter = recipient_domain_titles
	is_shown = {
		ucm_single_player_trigger = yes
		scope:actor = { is_ai = no }
		scope:recipient = { is_alive = yes is_landed_or_landless_administrative = yes }
		NOT = { scope:recipient = scope:actor }
	}
	can_be_picked_title = {
		scope:target = { holder = scope:recipient }
	}
	send_option = {
		flag = with_vassals
		localization = ucm_take_title_with_vassals
	}
	on_accept = {
		create_title_and_vassal_change = {
			type = conquest
			save_scope_as = change
			add_claim_on_loss = no
		}
		scope:target = {
			if = {
				limit = { always = scope:with_vassals }
				change_title_holder_include_vassals = { holder = scope:actor change = scope:change }
			}
			else = {
				change_title_holder = { holder = scope:actor change = scope:change }
			}
		}
		resolve_title_and_vassal_change = scope:change
	}
	ai_will_do = 0
}

# Two picks: the first marks a title, the second merges it. Into a title of the same tier
# its de jure vassals move across; into the tier above the title itself moves.
ucm_merge_title = {
	category = interaction_category_uncategorized
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	auto_accept = yes
	interface_priority = 12060
	icon = debug_take_title
	desc = ucm_merge_title_desc
	target_type = title
	target_filter = recipient_domain_titles
	is_shown = {
		ucm_single_player_trigger = yes
		scope:actor = { is_ai = no }
		scope:recipient = { is_alive = yes is_landed = yes }
	}
	can_be_picked_title = {
		scope:target = { tier >= tier_county is_landless_type_title = no }
		trigger_if = {
			limit = { scope:actor = { has_variable = ucm_merge_title } }
			custom_tooltip = {
				text = ucm_merge_title_invalid_pair
				NOT = { scope:target = scope:actor.var:ucm_merge_title }
				NOT = { scope:actor.var:ucm_merge_title.de_jure_liege ?= scope:target }
				OR = {
					AND = {
						scope:target.tier = scope:actor.var:ucm_merge_title.tier
						scope:target = { tier > tier_county }
					}
					AND = {
						scope:actor.var:ucm_merge_title = { tier = tier_county }
						scope:target = { tier = tier_duchy }
					}
					AND = {
						scope:actor.var:ucm_merge_title = { tier = tier_duchy }
						scope:target = { tier = tier_kingdom }
					}
					AND = {
						scope:actor.var:ucm_merge_title = { tier = tier_kingdom }
						scope:target = { tier = tier_empire }
					}
				}
			}
		}
	}
	on_accept = {
		scope:actor = {
			if = {
				limit = { NOT = { has_variable = ucm_merge_title } }
				set_variable = { name = ucm_merge_title value = scope:target }
			}
			else = {
				if = {
					limit = { scope:target.tier = var:ucm_merge_title.tier }
					var:ucm_merge_title = {
						every_direct_de_jure_vassal_title = { set_de_jure_liege_title = scope:target }
					}
				}
				else = {
					var:ucm_merge_title = { set_de_jure_liege_title = scope:target }
				}
				remove_variable = ucm_merge_title
			}
		}
	}
	ai_will_do = 0
}

# The game's contract window, with no cost or limit on changes.
ucm_modify_vassal_contract = {
	category = interaction_category_uncategorized
	use_diplomatic_range = no
	ignores_pending_interaction_block = yes
	auto_accept = yes
	interface_priority = 12050
	special_interaction = liege_modify_vassal_contract
	interface = modify_vassal_contract
	icon = icon_contract_modification_single
	desc = ucm_modify_vassal_contract_desc
	send_name = "modify_vassal_contract_interaction_SEND"
	is_shown = {
		ucm_single_player_trigger = yes
		scope:actor = { is_ai = no }
		scope:recipient = { is_ruler = yes liege ?= scope:actor }
		NOT = { scope:recipient = scope:actor }
	}
	on_accept = {
		every_in_list = {
			list = changed_obligations
			save_scope_as = obligation
			scope:recipient = {
				vassal_contract_set_obligation_level = {
					type = scope:obligation.vassal_contract_type
					level = scope:obligation
				}
			}
		}
	}
	ai_will_do = 0
}

# Unified Cheat Menu: shared effects.

# Main menu buttons run with root = player. The baseline's editor launchers expect
# scope:actor and scope:recipient, as set by a character interaction.
ucm_menu_scopes_effect = {
	save_scope_as = actor
	if = {
		limit = { exists = var:ucm_target }
		var:ucm_target = { save_scope_as = recipient }
	}
}

# Open/close the main menu. Scope = player. The baseline's interactions and decisions
# only show while their sub-menu toggles (character flags) are on, so the menu turns
# them on while it's open (the right-click menu and decision list show them too),
# then turns off only the ones it turned on.
ucm_menu_flag_on_effect = {
	if = {
		limit = { NOT = { has_character_flag = $FLAG$ } }
		add_character_flag = $FLAG$
		add_character_flag = ucm_added_$FLAG$
	}
}

ucm_menu_flag_off_effect = {
	if = {
		limit = { has_character_flag = ucm_added_$FLAG$ }
		remove_character_flag = $FLAG$
		remove_character_flag = ucm_added_$FLAG$
	}
}

# The sub-menu toggles, each passed to DO (ucm_menu_flag_on_effect / _off_effect).
ucm_menu_sub_flags_effect = {
	$DO$ = { FLAG = ucm_character_interaction_menu_open }
	$DO$ = { FLAG = bone_cheat_interactions_flag }
	$DO$ = { FLAG = open_optroop_menu }
}

ucm_menu_open_effect = {
	add_character_flag = ucm_menu_open
	if = {
		limit = { NOT = { exists = var:ucm_step } }
		set_variable = { name = ucm_step value = 1 }
	}
	ucm_menu_sub_flags_effect = { DO = ucm_menu_flag_on_effect }
}

ucm_menu_close_effect = {
	remove_character_flag = ucm_menu_open
	remove_variable = ucm_group_pick
	ucm_menu_sub_flags_effect = { DO = ucm_menu_flag_off_effect }
}

# Picker launchers, shared by the right-click interactions and the main menu.
ucm_open_culture_picker_effect = {
	ucm_cheat_menu_lock_recipient_effect = yes
	scope:actor = {
		ucm_cheat_menu_sync_target_effect = yes
		ucm_open_acm_culture_picker_effect = yes
	}
}

ucm_open_religion_picker_effect = {
	ucm_cheat_menu_lock_recipient_effect = yes
	scope:recipient = { faith = { save_scope_as = selected_faith } }
	scope:actor = {
		ucm_save_ucm_selected_faith_effect = yes
		ucm_open_acm_religion_picker_effect = yes
	}
}

ucm_open_tradition_picker_effect = {
	ucm_cheat_menu_lock_recipient_effect = yes
	scope:recipient = { culture = { save_scope_as = culture } }
	scope:actor = {
		ucm_cw_open_tradition_picker_effect = yes
		add_character_flag = ucm_cw_tradition_picker_open
	}
}

# ---------------------------------------------------------------------------
# Group actions (spec Section 5). Root = player throughout.
#   var:ucm_target       the clicked character the group comes from
#   var:ucm_group        one of the groups in ucm_group_for_each_effect
#   var:ucm_group_pick   the group picked in the menu's group-action picker
#   var:ucm_group_action one of the actions in ucm_group_apply_member_effect
# Adding a group: a branch below, ucm_group_is_county_trigger if it holds counties,
# pick scripted GUIs (common/scripted_guis/ucm_group_sguis.txt), loc.
# Adding an action: a branch in ucm_group_member_eligible_trigger and in
# ucm_group_apply_member_effect, a run scripted GUI, loc.
# ---------------------------------------------------------------------------

# Runs ACTION on the group picked in the menu.
ucm_group_start_picked_effect = {
	set_variable = { name = ucm_group value = var:ucm_group_pick }
	set_variable = { name = ucm_group_action value = flag:$ACTION$ }
	trigger_event = ucm_group.1
}

ucm_group_for_each_effect = {
	if = {
		limit = { var:ucm_group = flag:courtiers }
		var:ucm_target = { every_courtier = { $DO$ = yes } }
	}
	else_if = {
		limit = { var:ucm_group = flag:vassals }
		var:ucm_target = { every_vassal = { $DO$ = yes } }
	}
	else_if = {
		limit = { var:ucm_group = flag:all_vassals }
		var:ucm_target = { every_vassal_or_below = { $DO$ = yes } }
	}
	else_if = {
		limit = { var:ucm_group = flag:councillors }
		var:ucm_target = { every_councillor = { $DO$ = yes } }
	}
	else_if = {
		limit = { var:ucm_group = flag:knights }
		var:ucm_target = { every_knight = { $DO$ = yes } }
	}
	else_if = {
		limit = { var:ucm_group = flag:children }
		var:ucm_target = { every_child = { $DO$ = yes } }
	}
	else_if = {
		limit = { var:ucm_group = flag:close_family }
		var:ucm_target = { every_close_family_member = { $DO$ = yes } }
	}
	else_if = {
		limit = { var:ucm_group = flag:spouses }
		var:ucm_target = { every_spouse = { $DO$ = yes } }
	}
	else_if = {
		limit = { var:ucm_group = flag:house }
		var:ucm_target = { house ?= { every_house_member = { $DO$ = yes } } }
	}
	else_if = {
		limit = { var:ucm_group = flag:dynasty }
		var:ucm_target = { dynasty ?= { every_dynasty_member = { $DO$ = yes } } }
	}
	else_if = {
		limit = { var:ucm_group = flag:lovers }
		var:ucm_target = { every_relation = { type = lover $DO$ = yes } }
	}
	else_if = {
		limit = { var:ucm_group = flag:friends }
		var:ucm_target = { every_relation = { type = friend $DO$ = yes } }
	}
	else_if = {
		limit = { var:ucm_group = flag:rivals }
		var:ucm_target = { every_relation = { type = rival $DO$ = yes } }
	}
	else_if = {
		limit = { var:ucm_group = flag:extended_family }
		var:ucm_target = { every_extended_family_member = { $DO$ = yes } }
	}
	else_if = {
		limit = { var:ucm_group = flag:house_bloc }
		var:ucm_target = { house ?= { confederation ?= { every_confederation_member = { $DO$ = yes } } } }
	}
	else_if = {
		limit = { var:ucm_group = flag:faith_members }
		var:ucm_target = { faith ?= { every_faith_character = { $DO$ = yes } } }
	}
	else_if = {
		limit = { var:ucm_group = flag:culture_members }
		every_living_character = { limit = { culture = root.var:ucm_target.culture } $DO$ = yes }
	}
	else_if = {
		limit = { var:ucm_group = flag:sub_realm_counties }
		var:ucm_target = { every_sub_realm_county = { $DO$ = yes } }
	}
	else_if = {
		limit = { var:ucm_group = flag:dejure_counties }
		var:ucm_target = { primary_title ?= { every_de_jure_county = { $DO$ = yes } } }
	}
	else_if = {
		limit = { var:ucm_group = flag:culture_counties }
		var:ucm_target = { culture ?= { every_culture_county = { $DO$ = yes } } }
	}
	else_if = {
		limit = { var:ucm_group = flag:faith_counties }
		var:ucm_target = { faith ?= { every_faith_county = { $DO$ = yes } } }
	}
}

ucm_group_count_effect = {
	set_variable = { name = ucm_group_total value = 0 }
	set_variable = { name = ucm_group_eligible value = 0 }
	ucm_group_for_each_effect = { DO = ucm_group_count_member_effect }
}

ucm_group_count_member_effect = {
	root = { change_variable = { name = ucm_group_total add = 1 } }
	if = {
		limit = { ucm_group_member_eligible_trigger = yes }
		root = { change_variable = { name = ucm_group_eligible add = 1 } }
	}
}

# Every member is visited; ineligible ones are counted as skipped, never stop the run.
ucm_group_apply_effect = {
	set_variable = { name = ucm_group_changed value = 0 }
	set_variable = { name = ucm_group_skipped value = 0 }
	ucm_group_for_each_effect = { DO = ucm_group_apply_member_effect }
}

ucm_group_apply_member_effect = {
	if = {
		limit = { ucm_group_member_eligible_trigger = yes }
		# Counties
		if = {
			limit = { root.var:ucm_group_action = flag:max_development }
			change_development_level = 100
		}
		else_if = {
			limit = { root.var:ucm_group_action = flag:max_control }
			change_county_control = 100
		}
		else_if = {
			limit = { root.var:ucm_group_action = flag:county_faith }
			set_county_faith = root.var:ucm_target.faith
		}
		else_if = {
			limit = { root.var:ucm_group_action = flag:county_culture }
			set_county_culture = root.var:ucm_target.culture
		}
		# Characters
		else_if = {
			limit = { root.var:ucm_group_action = flag:clear_stress }
			add_stress = -1000
		}
		else_if = {
			limit = { root.var:ucm_group_action = flag:skills_up }
			add_diplomacy_skill = ucm_step_value
			add_martial_skill = ucm_step_value
			add_stewardship_skill = ucm_step_value
			add_intrigue_skill = ucm_step_value
			add_learning_skill = ucm_step_value
			add_prowess_skill = ucm_step_value
			force_character_skill_recalculation = yes
		}
		else_if = {
			limit = { root.var:ucm_group_action = flag:gold }
			add_gold = 1000
		}
		else_if = {
			limit = { root.var:ucm_group_action = flag:prestige }
			add_prestige = 1000
		}
		else_if = {
			limit = { root.var:ucm_group_action = flag:piety }
			add_piety = 1000
		}
		else_if = {
			limit = { root.var:ucm_group_action = flag:convert_faith }
			set_character_faith = root.var:ucm_target.faith
		}
		else_if = {
			limit = { root.var:ucm_group_action = flag:convert_culture }
			set_culture_same_as = root.var:ucm_target
		}
		else_if = {
			limit = { root.var:ucm_group_action = flag:max_opinion }
			add_opinion = { target = root.var:ucm_target modifier = ucm_loyal_vassal_opinion years = 100 }
		}
		root = { change_variable = { name = ucm_group_changed add = 1 } }
	}
	else = {
		root = { change_variable = { name = ucm_group_skipped add = 1 } }
	}
}

ucm_group_clear_effect = {
	remove_variable = ucm_group
	remove_variable = ucm_group_action
	remove_variable = ucm_group_total
	remove_variable = ucm_group_eligible
	remove_variable = ucm_group_changed
	remove_variable = ucm_group_skipped
}

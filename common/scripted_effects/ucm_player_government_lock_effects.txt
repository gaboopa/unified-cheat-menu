# A player the menu made meritocratic stays meritocratic; ucm_monthly_tick_events reverts other changes.
ucm_player_government_lock_apply_effect = {
	if = {
		limit = { is_ai = no }
		add_character_flag = ucm_player_government_locked
	}
}

ucm_player_government_lock_maintain_effect = {
	if = {
		limit = { is_ai = no has_character_flag = ucm_player_government_locked NOT = { has_government = meritocratic_government } }
		ucm_convert_character_to_meritocratic_government_effect = yes
		send_interface_toast = { title = ucm_government_lock_reverted_toast_title desc = ucm_government_lock_reverted_toast_desc }
	}
}

# At most once a month, since mass conversions can repeat.
ucm_player_government_feudal_blocked_notify_effect = {
	if = {
		limit = { NOT = { has_character_flag = ucm_government_feudal_block_toast_recent } }
		send_interface_toast = { title = ucm_government_feudal_blocked_toast_title desc = ucm_government_feudal_blocked_toast_desc }
		add_character_flag = { flag = ucm_government_feudal_block_toast_recent days = 30 }
	}
}

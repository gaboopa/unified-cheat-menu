# Culture innovation page return routing.
ucm_cheat_menu_culture_innovation_era_menu_back_effect = {
  if = {
    limit = { has_character_flag = ucm_quick_modifications_innovation_entry }
    trigger_event = ucm_cheat_menu.0130
  }
  else_if = {
    limit = { has_character_flag = ucm_culture_innov_from_0120 }
    trigger_event = ucm_cheat_menu.0130
  }
  else_if = {
    limit = { has_character_flag = ucm_culture_innov_from_global_selected }
    trigger_event = ucm_cheat_menu.0130
  }
  else = {
    trigger_event = ucm_cheat_menu.0130
  }
}

ucm_quick_modifications_innovation_close_effect = {
  remove_character_flag = ucm_quick_modifications_innovation_entry
}

# 打开当前角色的原版文化窗口
ucm_open_selected_culture_vanilla_window_effect = {
  if = {
    limit = { exists = scope:ucm_selected_culture }
    set_variable = {
      name = ucm_selected_culture_var
      value = scope:ucm_selected_culture
    }
    ucm_acm_open_culture_vanilla_editor_effect = yes
  }
}

# 与 HUD 左下角点击「文化」相同：打开当前角色文化的原版侧栏（须在角色 scope 内调用）
# 同步 save_scope_as = ucm_selected_culture，便于支柱/传统分页对「当前侧栏文化」生效
ucm_open_actor_culture_vanilla_window_effect = {
  culture = {
    save_scope_as = ucm_selected_culture
    open_view_data = {
      view = culture_window
    }
  }
}

# 子类默认名：制作旗标在 ucm_acm_craft_creator（或 scope:actor）上
ucm_acm_craft_subtype_has_flag_effect = {
  if = {
    limit = {
      OR = {
        AND = {
          exists = scope:ucm_acm_craft_creator
          scope:ucm_acm_craft_creator = { has_character_flag = $FLAG$ }
        }
        AND = {
          exists = scope:actor
          scope:actor = { has_character_flag = $FLAG$ }
        }
      }
    }
    set_artifact_name = $NAME$
    set_artifact_description = ucm_acm_ades_craft_subtype
  }
}

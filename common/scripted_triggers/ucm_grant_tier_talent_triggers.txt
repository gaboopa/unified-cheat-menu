# 多余头衔授予：拟授头衔越高，文/武能力线越高（见 ucm_purge_governor_skill_values.txt）

ucm_is_talent_for_grant_tier_barony_trigger = {
  OR = {
    AND = {
      ucm_is_gm_yan_military_government_trigger = yes
      ucm_military_skill_sum_for_purge_effective_value >= ucm_military_grant_barony_threshold_value
    }
    AND = {
      ucm_is_gm_yan_civil_government_trigger = yes
      ucm_civil_skill_sum_for_purge_effective_value >= ucm_civil_grant_barony_threshold_value
    }
    AND = {
      NOT = { ucm_is_gm_yan_government_trigger = yes }
      OR = {
        vassal_contract_has_flag = celestial_military_appointment
        vassal_contract_has_flag = meritocratic_military_appointment
      }
      ucm_military_skill_sum_for_purge_value >= ucm_military_grant_barony_threshold_value
    }
    AND = {
      NOT = { ucm_is_gm_yan_government_trigger = yes }
      NOR = {
        vassal_contract_has_flag = celestial_military_appointment
        vassal_contract_has_flag = meritocratic_military_appointment
      }
      is_landed = yes
      ucm_civil_skill_sum_for_purge_effective_value >= ucm_civil_grant_barony_threshold_value
    }
    AND = {
      NOT = { is_landed = yes }
      OR = {
        ucm_civil_skill_sum_for_purge_value >= ucm_civil_grant_barony_threshold_value
        ucm_military_skill_sum_for_purge_value >= ucm_military_grant_barony_threshold_value
      }
    }
  }
}

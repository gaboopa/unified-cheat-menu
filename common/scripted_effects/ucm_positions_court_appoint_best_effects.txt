# 廷臣 + 直属封臣合一候选池，按 aptitude 择优任命/替换（不依赖 can_be_employed_as 严格筛选）

ucm_positions_clear_court_candidate_list_effect = {
  scope:ucm_pos_reshuffle_liege = {
    if = {
      limit = { has_variable_list = ucm_pos_court_candidate_list }
      clear_variable_list = ucm_pos_court_candidate_list
    }
  }
}

ucm_positions_build_court_candidate_list_effect = {
  ucm_positions_clear_court_candidate_list_effect = yes
  scope:ucm_pos_reshuffle_liege = {
    every_courtier = {
      limit = {
        ucm_positions_loose_court_appointment_candidate_for_reshuffle_trigger = yes
        NOT = { has_court_position = $COURT_POS$ }
      }
      save_scope_as = ucm_pos_cand
      scope:ucm_pos_reshuffle_liege = {
        add_to_variable_list = {
          name = ucm_pos_court_candidate_list
          target = scope:ucm_pos_cand
        }
      }
      clear_saved_scope = ucm_pos_cand
    }
    every_vassal = {
      limit = {
        ucm_positions_court_reshuffle_vassal_loose_candidate_trigger = yes
        NOT = { has_court_position = $COURT_POS$ }
      }
      save_scope_as = ucm_pos_cand
      scope:ucm_pos_reshuffle_liege = {
        add_to_variable_list = {
          name = ucm_pos_court_candidate_list
          target = scope:ucm_pos_cand
        }
      }
      clear_saved_scope = ucm_pos_cand
    }
  }
}

ucm_positions_bring_pick_to_court_if_vassal_effect = {
  scope:ucm_pos_court_pick = {
    if = {
      limit = {
        is_vassal_of = scope:ucm_pos_reshuffle_liege
        NOT = { is_courtier_of = scope:ucm_pos_reshuffle_liege }
      }
      set_employer = scope:ucm_pos_reshuffle_liege
    }
  }
}

ucm_positions_try_appoint_court_slot_best_vacant_effect = {
  scope:ucm_pos_reshuffle_liege = {
    if = {
      limit = {
        can_employ_court_position_type = $COURT_POS$
      }
      # 先 can_be_employed_as 择优（拜占庭都长、冠军战车手、游牧将军/颉于伽思等）
      if = {
        limit = {
          any_courtier = {
            ucm_positions_loose_court_appointment_candidate_for_reshuffle_trigger = yes
            can_be_employed_as = $COURT_POS$
            NOT = { has_court_position = $COURT_POS$ }
          }
        }
        ordered_courtier = {
          limit = {
            ucm_positions_loose_court_appointment_candidate_for_reshuffle_trigger = yes
            can_be_employed_as = $COURT_POS$
            NOT = { has_court_position = $COURT_POS$ }
          }
          order_by = aptitude:$COURT_POS$
          max = 1
          check_range_bounds = no
          save_scope_as = ucm_pos_court_pick
        }
        if = {
          limit = { exists = scope:ucm_pos_court_pick }
          ucm_positions_bring_pick_to_court_if_vassal_effect = yes
          appoint_court_position = {
            recipient = scope:ucm_pos_court_pick
            court_position = $COURT_POS$
          }
          clear_saved_scope = ucm_pos_court_pick
        }
      }
      else_if = {
        limit = {
          any_vassal = {
            ucm_positions_court_reshuffle_vassal_loose_candidate_trigger = yes
            can_be_employed_as = $COURT_POS$
            NOT = { has_court_position = $COURT_POS$ }
          }
        }
        ordered_vassal = {
          limit = {
            ucm_positions_court_reshuffle_vassal_loose_candidate_trigger = yes
            can_be_employed_as = $COURT_POS$
            NOT = { has_court_position = $COURT_POS$ }
          }
          order_by = aptitude:$COURT_POS$
          max = 1
          check_range_bounds = no
          save_scope_as = ucm_pos_court_pick
        }
        if = {
          limit = { exists = scope:ucm_pos_court_pick }
          ucm_positions_bring_pick_to_court_if_vassal_effect = yes
          appoint_court_position = {
            recipient = scope:ucm_pos_court_pick
            court_position = $COURT_POS$
          }
          clear_saved_scope = ucm_pos_court_pick
        }
      }
      else = {
        ucm_positions_build_court_candidate_list_effect = { COURT_POS = $COURT_POS$ }
        if = {
          limit = {
            has_variable_list = ucm_pos_court_candidate_list
            variable_list_size = {
              name = ucm_pos_court_candidate_list
              value > 0
            }
          }
          ordered_in_list = {
            variable = ucm_pos_court_candidate_list
            order_by = aptitude:$COURT_POS$
            max = 1
            check_range_bounds = no
            save_scope_as = ucm_pos_court_pick
          }
          if = {
            limit = { exists = scope:ucm_pos_court_pick }
            ucm_positions_bring_pick_to_court_if_vassal_effect = yes
            appoint_court_position = {
              recipient = scope:ucm_pos_court_pick
              court_position = $COURT_POS$
            }
            clear_saved_scope = ucm_pos_court_pick
          }
        }
        ucm_positions_clear_court_candidate_list_effect = yes
      }
    }
  }
}

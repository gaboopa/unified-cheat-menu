ucm_char_design_hidden_01 = {
	if = {
		limit = {
			has_game_rule = sarc_initial_five
		}
		create_character = {
			employer = root
			gender_female_chance = $CHANCE$
			random_traits = no
			age = 16
			diplomacy = 5
			martial = 5
			stewardship = 5
			intrigue = 5
			learning = 5
			prowess = 5
			health = 5.0
			fertility = 0.6
			save_scope_as = custom_char
			dynasty = root
			faith = root.faith
			culture = root.culture
		}
		set_variable = { name = sarc_custom_char_var value = scope:custom_char }
	}
	else = {
		create_character = {
			employer = root
			gender_female_chance = $CHANCE$
			random_traits = no
			age = 16
			diplomacy = 10
			martial = 10
			stewardship = 10
			intrigue = 10
			learning = 10
			prowess = 10
			health = 5.0
			fertility = 0.6
			save_scope_as = custom_char
			dynasty = root
			faith = root.faith
			culture = root.culture
		}
		set_variable = { name = sarc_custom_char_var value = scope:custom_char }
	}
	set_designated_heir = var:sarc_custom_char_var
}

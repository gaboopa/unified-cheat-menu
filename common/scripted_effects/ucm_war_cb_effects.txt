# Victory for the menu's Declare wars: the attacker takes the defender's titles of $TIER$ and every county it occupies.
ucm_war_victory_effect = {
	create_title_and_vassal_change = {
		type = conquest
		save_scope_as = change
		add_claim_on_loss = no
	}
	scope:defender = {
		every_held_title = {
			limit = { tier = $TIER$ }
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
				take_baronies = yes
			}
		}
		every_sub_realm_county = {
			limit = { county_controller = scope:attacker }
			change_title_holder = {
				holder = scope:attacker
				change = scope:change
				take_baronies = yes
			}
		}
	}
	resolve_title_and_vassal_change = scope:change
}

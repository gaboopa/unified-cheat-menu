# The marriage ban trait undoes any wedding involving its holder.
on_marriage = {
	on_actions = { ucm_marriage_ban_check }
}

ucm_marriage_ban_check = {
	trigger = { ucm_single_player_trigger = yes }
	effect = {
		if = {
			limit = { has_trait = ucm_trait_marriage_ban }
			ucm_cheat_menu_marriage_ban_strip_wedlocks_effect = yes
		}
		else_if = {
			limit = { primary_spouse ?= { has_trait = ucm_trait_marriage_ban } }
			divorce = primary_spouse
		}
	}
}

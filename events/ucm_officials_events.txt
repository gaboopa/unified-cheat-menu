# Spreads the landless purge over several days so one click doesn't freeze the game.
ucm_officials.9088 = {
	type = character_event
	hidden = yes
	trigger = {
		has_character_flag = ucm_world_purge_active
		has_variable = ucm_world_purge_remaining
		var:ucm_world_purge_remaining > 0
	}
	immediate = { ucm_world_purge_landless_ai_effect = yes }
}

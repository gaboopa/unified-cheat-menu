# this = a spouse or concubine of the shielded player.
ucm_fidelity_end_ward_schemes_effect = {
  save_scope_as = ucm_fid_ward
  # Incoming
  every_targeting_scheme = {
    if = {
      limit = {
        ucm_fidelity_scheme_type_trigger = yes
        scope:ucm_fid_ward = {
          ucm_seduce_shield_active_for_target_trigger = yes
        }
        scheme_owner = {
          NOR = {
            is_spouse_of = scope:ucm_fid_ward
            AND = {
              scope:ucm_fid_ward = { is_concubine = yes }
              this = scope:ucm_fid_ward.concubinist
            }
          }
        }
      }
      end_scheme = yes
    }
  }
  # Outgoing, same rule as ucm_seduce_scheme_blocked_outgoing: shielded, and not aimed at a spouse or concubinist
  every_scheme = {
    scheme_target_character = { save_scope_as = ucm_fid_paramour }
    if = {
      limit = {
        ucm_fidelity_scheme_type_trigger = yes
        scheme_owner = { this = scope:ucm_fid_ward }
        scope:ucm_fid_ward = { ucm_seduce_shield_active_for_target_trigger = yes }
        NOR = {
          scope:ucm_fid_ward = { is_spouse_of = scope:ucm_fid_paramour }
          scope:ucm_fid_ward = { is_concubine_of = scope:ucm_fid_paramour }
        }
      }
      end_scheme = yes
    }
  }
}

# Called monthly from ucm_monthly_tick.0001 (events/ucm_monthly_tick_events.txt). root/this = the player with ucm_seduce_shield_active.
ucm_fidelity_affair_sweep_effect = {
  every_spouse = {
    limit = { is_alive = yes }
    ucm_fidelity_end_ward_schemes_effect = yes
  }
  every_concubine = {
    limit = { is_alive = yes }
    ucm_fidelity_end_ward_schemes_effect = yes
  }
  ucm_seduce_shield_upkeep_effect = yes
}

# this = a character whose betrothal ends if either side has the marriage ban.
ucm_marriage_ban_break_betrothal_effect = {
  if = {
    limit = {
      is_betrothed = yes
      OR = {
        has_trait = ucm_trait_marriage_ban
        betrothed = { has_trait = ucm_trait_marriage_ban }
      }
    }
    betrothed = {
      break_betrothal = prev
    }
  }
}

# Marriage ban: break betrothals of the player/courtiers that involve a banned character. root/this = the player.
ucm_marriage_ban_betrothal_sweep_effect = {
  ucm_marriage_ban_break_betrothal_effect = yes
  every_courtier = {
    ucm_marriage_ban_break_betrothal_effect = yes
  }
}

# World landless cleanup: base pool (without the remaining-quota check)
# Spared: the player, human players, the player's dynasty, children, grandchildren, spouses and concubines

ucm_world_purge_landless_base_eligible_trigger = {
  is_alive = yes
  is_landed = no
  is_ai = yes
  is_pool_character = yes
  is_lowborn = yes
  age >= 16
  NOT = { this = root }
  NOR = {
    AND = {
      is_playable_character = yes
      is_ai = no
    }
  }
  NOR = {
    AND = {
      root = {
        exists = dynasty
      }
      exists = dynasty
      dynasty = root.dynasty
    }
  }
  NOR = {
    father = root
    mother = root
  }
  NOR = {
    father = {
      OR = {
        father = root
        mother = root
      }
    }
  }
  NOR = {
    mother = {
      OR = {
        father = root
        mother = root
      }
    }
  }
  NOT = {
    is_spouse_of = root
  }
  NOT = {
    is_concubine_of = root
  }
}

# Skonester · 军事管理（效果一律作用在右击目标 ucm_cheat_menu_target，含 AI/玩家/廷臣）

# 同步右击目标；优先 global_var，其次玩家身上的 ucm_military_menu_victim
ucm_military_ensure_target_effect = {
  ucm_cheat_menu_sync_target_effect = yes
  if = {
    limit = { NOT = { exists = scope:ucm_cheat_menu_target } }
    root = {
      if = {
        limit = { has_variable = ucm_military_menu_victim }
        var:ucm_military_menu_victim = {
          save_scope_as = ucm_cheat_menu_target
          set_global_variable = {
            name = ucm_cheat_menu_recipient
            value = this
          }
        }
      }
    }
  }
}

ucm_military_event_immediate_effect = {

}

}

ucm_military_send_supply_toggle_toast_effect = {
  if = {
    limit = {
      exists = scope:ucm_military_toggle_char
      scope:ucm_military_toggle_char = { var:ucm_military_toggle_on = 1 }
    }
    root = {
      send_interface_toast = {
        type = event_toast_effect_good
        title = ucm_military_supply_on_toast
        desc = ucm_military_supply_on_toast_desc
        left_icon = scope:ucm_military_toggle_char
      }
    }
  }
  else_if = {
    limit = { exists = scope:ucm_military_toggle_char }
    root = {
      send_interface_toast = {
        type = event_toast_effect_neutral
        title = ucm_military_supply_off_toast
        left_icon = scope:ucm_military_toggle_char
      }
    }
  }
}

ucm_cheat_menu_raise_all_levies_effect = {
  ucm_military_ensure_target_effect = yes
  if = {
    limit = { exists = scope:ucm_cheat_menu_target }
    scope:ucm_cheat_menu_target = {
      ucm_cheat_menu_raise_all_levies_on_character_effect = yes
      save_scope_as = ucm_military_toast_char
    }
    root = {
      send_interface_toast = {
        type = event_toast_effect_good
        title = ucm_military_raise_levies_toast
        desc = ucm_military_raise_levies_toast_desc
        left_icon = scope:ucm_military_toast_char
      }
    }
  }
}

ucm_cheat_menu_raise_all_levies_on_character_effect = {
  if = {
    limit = {
      is_alive = yes
      exists = capital_province
    }
    capital_province = { save_scope_as = ucm_military_spawn_loc }
  }
  else_if = {
    limit = {
      is_alive = yes
      exists = location
    }
    location = { save_scope_as = ucm_military_spawn_loc }
  }
  if = {
    limit = { exists = scope:ucm_military_spawn_loc }
    spawn_army = {
      name = ucm_military_raise_levies_army_name
      levies = {
        add = max_military_strength
        min = 1000
      }
      location = scope:ucm_military_spawn_loc
      inheritable = no
    }
  }
}

  every_army = {
    add_supply = 999999
  }
}

ucm_cheat_menu_apply_supply_overflow_stacks_effect = {
  remove_all_character_modifier_instances = ucm_military_supply_overflow_modifier
  while = {
    count = 11
    add_character_modifier = ucm_military_supply_overflow_modifier
  }
}

ucm_cheat_menu_start_supply_war_loop_effect = {
  if = {
    limit = {
      is_alive = yes
      is_at_war = yes
      has_character_modifier = ucm_military_instant_replenish_modifier
    }
    trigger_event = { id = ucm_cheat_menu.3099 days = 1 }
  }
}

ucm_cheat_menu_instant_army_replenish_on_character_effect = {
  # 不修改士兵/常备人数；切换军需修正 + stacking，并拉满当前军团补给
  if = {
    limit = { NOT = { has_character_modifier = ucm_military_instant_replenish_modifier } }
    ucm_cheat_menu_refill_all_army_supply_on_character_effect = yes
    ucm_cheat_menu_apply_supply_overflow_stacks_effect = yes
    add_character_modifier = ucm_military_instant_replenish_modifier
    ucm_holder_list_add_effect = { LIST = ucm_supply_holders }
    ucm_cheat_menu_start_supply_war_loop_effect = yes
    ucm_military_cheat_authorize_target_if_player_actor_effect = yes
    save_scope_as = ucm_military_toggle_char
    set_variable = { name = ucm_military_toggle_on value = 1 }
  }
  else = {
    remove_character_modifier = ucm_military_instant_replenish_modifier
    remove_all_character_modifier_instances = ucm_military_supply_overflow_modifier
    save_scope_as = ucm_military_toggle_char
    set_variable = { name = ucm_military_toggle_on value = 0 }
  }
}

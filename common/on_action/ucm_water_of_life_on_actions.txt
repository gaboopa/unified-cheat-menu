# Monthly upkeep for characters holding the water_of_life trait (initial pass is in ucm_startup_sweep.txt).
ucm_monthly_tick = {
    on_actions = { ucm_water_of_life_restore_all }
}
ucm_water_of_life_restore_all = {
	trigger = { ucm_single_player_trigger = yes }
    effect = {
        # One-time migration for saves made before the holders list existed. Grant sites
        # (special gifts, water_of_life_effect, the hooded old man) add new holders.
        if = {
            limit = { NOT = { has_global_variable = ucm_water_of_life_list_built } }
            set_global_variable = { name = ucm_water_of_life_list_built value = 1 }
            every_living_character = {
                limit = { has_trait = water_of_life }
                ucm_holder_list_add_effect = { LIST = ucm_water_of_life_holders }
            }
        }
        every_in_global_list = {
            variable = ucm_water_of_life_holders
            limit = {
                is_alive = yes
                has_trait = water_of_life
            }
            ucm_water_of_life_restore_effect = yes
        }
        # Collect stale entries first, remove after iterating (never edit a list while iterating it).
        every_in_global_list = {
            variable = ucm_water_of_life_holders
            limit = {
                OR = {
                    is_alive = no
                    NOT = { has_trait = water_of_life }
                }
            }
            add_to_temporary_list = ucm_stale_ucm_water_of_life_holders
        }
        every_in_list = {
            list = ucm_stale_ucm_water_of_life_holders
            remove_list_global_variable = { name = ucm_water_of_life_holders target = this }
        }
    }
}

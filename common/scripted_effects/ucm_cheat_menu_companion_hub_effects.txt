# 生成帅哥美女 hub：按目标角色上保存的文化变量与性别变量调用原有生成效果
ucm_cheat_menu_spawn_companion_from_hub_prefs_effect = {
  ucm_char_gen_sync_recipient_to_scope_effect = yes
  scope:ucm_char_gen_target = {
    save_scope_as = ucm_spawn_talent_employer
    if = {
      limit = {
        OR = {
          NOT = { has_variable = ucm_companion_female_chance }
          var:ucm_companion_female_chance >= 50
        }
      }
      var:ucm_companion_stored_culture = {
        save_scope_as = ucm_beauty_spawn_culture
      }
    }
    else = {
      var:ucm_companion_stored_culture = {
        save_scope_as = ucm_handsome_spawn_culture
      }
    }
  }
  if = {
    limit = {
      scope:ucm_char_gen_target = {
        OR = {
          NOT = { has_variable = ucm_companion_female_chance }
          var:ucm_companion_female_chance >= 50
        }
      }
    }
    ucm_cheat_menu_spawn_beauty_companion_effect = yes
  }
  else = {
    ucm_cheat_menu_spawn_handsome_companion_effect = yes
  }
}

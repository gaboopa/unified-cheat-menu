# Weapon crafting dispatcher. Type-specific effects create and finalize the artifact
# in the current recipient scope.

ucm_acm_create_weapon_artifact_effect = {
  if = {
    limit = {
      OR = {
        scope:ucm_acm_craft_creator = {
          OR = {
            has_character_flag = ucm_acm_wpn_battleaxe
            has_character_flag = ucm_acm_wpn_greataxe
          }
        }
        AND = {
          NOT = { exists = scope:ucm_acm_craft_creator }
          root = {
            OR = {
              has_character_flag = ucm_acm_wpn_battleaxe
              has_character_flag = ucm_acm_wpn_greataxe
            }
          }
        }
      }
    }
    ucm_acm_create_type_axe_effect = yes
  }
  else_if = {
    limit = {
      OR = {
        scope:ucm_acm_craft_creator = {
          OR = {
            has_character_flag = ucm_acm_wpn_mace
            has_character_flag = ucm_acm_wpn_flail
          }
        }
        AND = {
          NOT = { exists = scope:ucm_acm_craft_creator }
          root = {
            OR = {
              has_character_flag = ucm_acm_wpn_mace
              has_character_flag = ucm_acm_wpn_flail
            }
          }
        }
      }
    }
    ucm_acm_create_type_hammer_effect = yes
  }
  else_if = {
    limit = {
      OR = {
        scope:ucm_acm_craft_creator = {
          OR = {
            has_character_flag = ucm_acm_wpn_spear
            has_character_flag = ucm_acm_wpn_halberd
          }
        }
        AND = {
          NOT = { exists = scope:ucm_acm_craft_creator }
          root = {
            OR = {
              has_character_flag = ucm_acm_wpn_spear
              has_character_flag = ucm_acm_wpn_halberd
            }
          }
        }
      }
    }
    ucm_acm_create_type_spear_effect = yes
  }
  else_if = {
    limit = {
      OR = {
        scope:ucm_acm_craft_creator = {
          OR = {
            has_character_flag = ucm_acm_wpn_shortbow
            has_character_flag = ucm_acm_wpn_longbow
            has_character_flag = ucm_acm_wpn_crossbow
          }
        }
        AND = {
          NOT = { exists = scope:ucm_acm_craft_creator }
          root = {
            OR = {
              has_character_flag = ucm_acm_wpn_shortbow
              has_character_flag = ucm_acm_wpn_longbow
              has_character_flag = ucm_acm_wpn_crossbow
            }
          }
        }
      }
    }
    ucm_acm_create_type_bow_effect = yes
  }
  else = {
    ucm_acm_create_type_sword_effect = yes
  }
}

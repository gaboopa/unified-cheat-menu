# UCM「庄园编辑」：须在 scope:recipient 或 scope:ucm_cheat_menu_target 上调用。
# 外槽/内建沿用原版 fill_external_* 与 add_random_internal_*；主建筑逐级 add_domicile_building。

# 行政/贤能等：庄园多在世族头衔或宗族族长名下；帝国/公国/伯爵持有者本人常无 domicile。
ucm_cheat_menu_resolve_domicile_holder_effect = {
  if = {
    limit = {
      is_alive = yes
      has_domicile = yes
    }
    save_scope_as = ucm_domicile_holder
  }
  else_if = {
    limit = {
      is_alive = yes
      exists = house.house_head
      house.house_head = {
        is_alive = yes
        has_domicile = yes
      }
    }
    house.house_head = {
      save_scope_as = ucm_domicile_holder
    }
  }
  else_if = {
    limit = {
      is_alive = yes
      exists = house
    }
    house = {
      ordered_house_member = {
        limit = {
          is_alive = yes
          has_domicile = yes
          is_house_head = yes
        }
        order_by = age
        save_scope_as = ucm_domicile_holder
      }
    }
  }
  else_if = {
    limit = {
      is_alive = yes
      exists = house
    }
    house = {
      ordered_house_member = {
        limit = {
          is_alive = yes
          has_domicile = yes
          any_held_title = { is_noble_family_title = yes }
        }
        order_by = age
        save_scope_as = ucm_domicile_holder
      }
    }
  }
  else_if = {
    limit = {
      is_alive = yes
      exists = house
    }
    house = {
      random_house_member = {
        limit = {
          is_alive = yes
          has_domicile = yes
        }
        save_scope_as = ucm_domicile_holder
      }
    }
  }
}

# 绕过 can_construct 革新门槛与建造费用（行政制国库 / 西式金币）。
ucm_domicile_cheat_prereqs_effect = {
  owner = {
    if = {
      limit = { has_treasury = yes }
      add_treasury = 50000
    }
    else = {
      add_gold = 50000
    }
    culture = {
      if = {
        limit = { NOT = { has_innovation = innovation_city_planning } }
        add_innovation = innovation_city_planning
      }
      if = {
        limit = { NOT = { has_innovation = innovation_manorialism } }
        add_innovation = innovation_manorialism
      }
      if = {
        limit = { NOT = { has_innovation = innovation_development_03 } }
        add_innovation = innovation_development_03
      }
      if = {
        limit = { NOT = { has_innovation = innovation_cranes } }
        add_innovation = innovation_cranes
      }
    }
  }
  switch = {
    trigger = is_domicile_type
    estate = {
      if = {
        limit = { NOT = { has_domicile_building_or_higher = estate_main_01 } }
        add_domicile_building = estate_main_01
      }
    }
    east_asian_estate = {
      if = {
        limit = { NOT = { has_domicile_building_or_higher = east_asian_estate_main_01 } }
        add_domicile_building = east_asian_estate_main_01
      }
    }
    japanese_manor = {
      if = {
        limit = { NOT = { has_domicile_building_or_higher = japanese_manor_main_01 } }
        add_domicile_building = japanese_manor_main_01
      }
    }
  }
}

ucm_cheat_menu_estate_edit_full_effect = {
  if = {
    limit = {
      is_alive = yes
      domicile ?= {
        OR = {
          is_domicile_type = estate
          is_domicile_type = japanese_manor
          is_domicile_type = east_asian_estate
        }
      }
    }
    domicile = {
      ucm_domicile_cheat_prereqs_effect = yes
      switch = {
        trigger = is_domicile_type
        estate = {
          ucm_cheat_menu_maximize_estate_domicile_inner_effect = yes
        }
        japanese_manor = {
          ucm_cheat_menu_maximize_japanese_manor_domicile_inner_effect = yes
        }
        east_asian_estate = {
          ucm_cheat_menu_maximize_east_asian_estate_domicile_inner_effect = yes
        }
      }
    }
  }
}

ucm_cheat_menu_maximize_estate_domicile_inner_effect = {
  ucm_domicile_residence_full_max_effect = yes
}

ucm_cheat_menu_maximize_japanese_manor_domicile_inner_effect = {
  ucm_domicile_residence_full_max_effect = yes
}

ucm_cheat_menu_maximize_east_asian_estate_domicile_inner_effect = {

}

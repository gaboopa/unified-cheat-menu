ucm_open_trait_editor_gui_effect = {
  ucm_cheat_menu_sync_trait_edit_target_effect = yes
  if = {
    limit = { exists = scope:actor }
    scope:actor = {
      if = {
        limit = { exists = global_var:ucm_cheat_menu_recipient }
        set_variable = {
          name = ucm_trait_gui_target
          value = global_var:ucm_cheat_menu_recipient
        }
      }
      remove_character_flag = ucm_trait_gui_open
      add_character_flag = ucm_trait_gui_open
    }
  }
  else = {
    if = {
      limit = { exists = global_var:ucm_cheat_menu_recipient }
      set_variable = {
        name = ucm_trait_gui_target
        value = global_var:ucm_cheat_menu_recipient
      }
    }
    remove_character_flag = ucm_trait_gui_open
    add_character_flag = ucm_trait_gui_open
  }
}

ucm_close_trait_editor_gui_effect = {
  remove_character_flag = ucm_trait_gui_open
}

# Palace exam phase: vanilla 2000 + extra merit scaled by this exam's type and the character's skills (military: martial + prowess; otherwise diplomacy + stewardship + learning)
ucm_imperial_exam_activity_scaled_bonus_merit_value = {
  value = 0
  if = {
    limit = {
      scope:activity = {
        has_activity_option = {
          category = special_type
          option = imperial_examination_focus_martial
        }
      }
    }
    add = martial
    add = prowess
  }
  else = {
    add = diplomacy
    add = stewardship
    add = learning
  }
  multiply = 18
  min = 0
  max = 1500
}

ucm_imperial_exam_palace_total_merit_value = {
  value = 2000
  add = ucm_imperial_exam_activity_scaled_bonus_merit_value
}

# One-click exam: matches the contract branches (military: military appointment; civil: civil appointment; otherwise the three civil skills)
ucm_one_click_exam_bonus_merit_value = {
  value = 0
  if = {
    limit = {
      OR = {
        vassal_contract_has_flag = celestial_military_appointment
        vassal_contract_has_flag = meritocratic_military_appointment
      }
    }
    add = martial
    add = prowess
  }
  else_if = {
    limit = {
      OR = {
        vassal_contract_has_flag = celestial_civil_appointment
        vassal_contract_has_flag = meritocratic_civil_appointment
      }
    }
    add = diplomacy
    add = stewardship
    add = learning
  }
  else = {
    add = diplomacy
    add = stewardship
    add = learning
  }
  multiply = 20
  min = 0
  max = 1200
}

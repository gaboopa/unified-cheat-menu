# 文化侧栏快捷条(0181)：事件 root 为玩家/actor 时，判断其文化是否与已保存的 ucm_selected_culture 相同

ucm_root_culture_matches_saved_selected_culture_trigger = {
  exists = scope:ucm_selected_culture
  culture = scope:ucm_selected_culture
}

ucm_root_culture_differs_from_saved_selected_culture_trigger = {
  exists = scope:ucm_selected_culture
  NOT = { culture = scope:ucm_selected_culture }
}

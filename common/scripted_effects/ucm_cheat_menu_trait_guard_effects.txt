# UCM特质保险丝：birth/random=0；仅玩家通过UCM菜单授予时打 ucm_traits_player_authorized。
# AI 仅在被玩家授予且带授权标时可保留；脚本/随机/未授权一律脉冲剥除。

# 授予UCM作弊特质：仅人类玩家为 actor 时可赋予（目标可为 AI 或玩家本人）
ucm_cheat_menu_grant_cheat_trait_effect = {
  if = {
    limit = {
      exists = scope:actor
      scope:actor = {
        is_ai = no
      }
      ucm_cheat_menu_may_receive_cheat_trait_trigger = yes
    }
    add_trait = $TRAIT$
    if = {
      limit = { trait:$TRAIT$ = trait:ucm_trait_baidubuqin }
      ucm_holder_list_add_effect = { LIST = ucm_baidubuqin_holders }
    }
    if = {
      limit = { trait:$TRAIT$ = trait:ucm_trait_xinwuguai }
      ucm_holder_list_add_effect = { LIST = ucm_xinwuguai_holders }
    }
    if = {
      limit = { trait:$TRAIT$ = trait:ucm_trait_stats_100_locked }
      ucm_holder_list_add_effect = { LIST = ucm_stats100_holders }
    }
    if = {
      limit = {
        trait:$TRAIT$ = trait:ucm_trait_baidubuqin
      }
      ucm_trait_baidubuqin_apply_effect = yes
    }
    if = {
      limit = {
        trait:$TRAIT$ = trait:ucm_trait_xinwuguai
      }
      ucm_trait_xinwuguai_apply_effect = yes
    }
    # 仅自制UCM特质授予时打授权标；原版/其它特质编辑不调用本 effect
    ucm_traits_authorize_recipient_if_player_actor_effect = yes
  }
}

# 玩家通过UCM菜单授予自制特质时给目标打标（人类或 AI 均可，防止脉冲误剥）
ucm_traits_authorize_recipient_if_player_actor_effect = {
  ucm_cheat_menu_sync_trait_edit_target_effect = yes
  if = {
    limit = {
      scope:actor = {
        is_ai = no
      }
      exists = scope:ucm_trait_edit_target
    }
    scope:ucm_trait_edit_target = {
      add_character_flag = ucm_traits_player_authorized
    }
  }
  else_if = {
    limit = {
      scope:actor = {
        is_ai = no
      }
      exists = scope:ucm_cheat_menu_target
    }
    scope:ucm_cheat_menu_target = {
      add_character_flag = ucm_traits_player_authorized
    }
  }
  else_if = {
    limit = {
      scope:actor = {
        is_ai = no
      }
      exists = scope:recipient
    }
    scope:recipient = {
      add_character_flag = ucm_traits_player_authorized
    }
  }
}

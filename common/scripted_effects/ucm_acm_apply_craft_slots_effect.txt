  if = {
    limit = {
      NOT = { exists = scope:ucm_acm_craft_creator }
      exists = scope:actor

  }
  if = {
    limit = { exists = scope:ucm_new_crafted_artifact }
    scope:ucm_new_crafted_artifact = {
      ucm_acm_add_artifact_modifier_slot_1 = yes
      ucm_acm_add_artifact_modifier_slot_2 = yes
      ucm_acm_add_artifact_modifier_slot_3 = yes
      ucm_acm_add_artifact_modifier_slot_4 = yes
      ucm_acm_add_artifact_modifier_slot_5 = yes
      ucm_acm_persist_craft_slots_on_artifact_effect = yes
    }
  }
}

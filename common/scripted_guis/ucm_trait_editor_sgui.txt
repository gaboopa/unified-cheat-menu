# Trait editor window (gui/ucm_trait_editor.gui). Root = player.

ucm_trait_editor_visible = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes has_character_flag = ucm_trait_editor_open }
}

ucm_trait_editor_close = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes }
  effect = { ucm_close_trait_editor_gui_effect = yes }
}

ucm_trait_editor_toggle = {
  scope = character
  saved_scopes = { trait_flag }
  is_shown = { ucm_single_player_trigger = yes }
  is_valid = { has_variable = ucm_trait_editor_target }
  effect = { ucm_trait_editor_dispatch_effect = yes }
}

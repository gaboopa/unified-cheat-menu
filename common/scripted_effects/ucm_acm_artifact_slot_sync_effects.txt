    limit = { scope:ucm_acm_sync_art = { has_artifact_modifier = $MOD$ } }
    root = { ucm_acm_artifact_slot_sync_put_effect = { VAL = $VAL$ CNT = $CNT$ } }
  }
}
# advcm 叠层修饰：POS/NEG 成对，先统计层数（负向为负数）再追加一个 GUI 槽。须在 global_var:ucm_acm_repair_target 作用域调用。
ucm_acm_artifact_slot_sync_stacked_effect = {
  root = {
    set_variable = { name = ucm_acm_tmp_stack value = 0 }
    set_variable = { name = ucm_acm_tmp_neg value = 0 }
  }
  if = {
    limit = {
      scope:ucm_acm_sync_art = {
        OR = {
          has_artifact_modifier = $POS$
          has_artifact_modifier = $NEG$
        }
      }
    }
    scope:ucm_acm_sync_art = {
      if = {
        limit = { has_artifact_modifier = $NEG$ }
        while = {
          limit = { has_artifact_modifier = $NEG$ }
          remove_artifact_modifier = $NEG$
          root = {
            change_variable = { name = ucm_acm_tmp_stack add = 1 }
            set_variable = { name = ucm_acm_tmp_neg value = 1 }
          }
        }
        while = {
          count = root.var:ucm_acm_tmp_stack
          add_artifact_modifier = $NEG$
        }
      }
      else = {
        while = {
          limit = { has_artifact_modifier = $POS$ }
          remove_artifact_modifier = $POS$
          root = { change_variable = { name = ucm_acm_tmp_stack add = 1 } }
        }
        while = {
          count = root.var:ucm_acm_tmp_stack
          add_artifact_modifier = $POS$
        }
      }
    }
    root = {
      if = {
        limit = { var:ucm_acm_tmp_neg = 1 }
        set_variable = {
          name = ucm_acm_tmp_stack
          value = {
            value = var:ucm_acm_tmp_stack
            multiply = -1
          }
        }
      }
      ucm_acm_artifact_slot_sync_put_effect = { VAL = $VAL$ CNT = var:ucm_acm_tmp_stack }
    }
  }
}

ucm_acm_load_modifier_slots_from_artifact_effect = {
  if = {
    limit = { exists = global_var:ucm_acm_repair_target }

    if = {
      limit = {
        global_var:ucm_acm_repair_target = { has_variable = ucm_acm_from_gui_craft }
      }

        limit = {
          global_var:ucm_acm_repair_target = { has_variable = ucm_acm_saved_m1 }
        }
        ucm_acm_load_craft_slots_from_artifact_saved_effect = yes
      }
      else = {
        ucm_acm_detect_craft_slots_from_artifact_modifiers_effect = yes
      }
    }
    root = {
      ucm_acm_sync_craft_layer_quality_effect = yes
      ucm_acm_clamp_craft_slot_layers_to_quality_effect = yes
    }
  }
}

# 官吏统辖「宝物制作管理」独立窗口（gui/ucm_artifact_craft.gui）
# 须同时写 is_valid，否则部分版本下 GetScriptedGui(...).Execute 不生效（关窗、按钮无反应）。
# 窗口是否显示请用 ucm_acm_gui_open_visible + IsShown，勿直接用 GetPlayer.HasCharacterFlag（部分版本下不可靠，会误显且关不掉）。

ucm_acm_gui_open_visible = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes 
    has_character_flag = ucm_acm_gui_open
    NOT = { has_character_flag = ucm_acm_gui_text_edit_active }
  }
  is_valid = { always = yes }
}

ucm_acm_gui_close = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes  always = yes }
  is_valid = { always = yes }
  effect = {
    remove_character_flag = ucm_acm_gui_open
    remove_character_flag = ucm_acm_modify_session
    remove_character_flag = ucm_acm_repair_gui_open
    remove_character_flag = ucm_acm_edit_from_craft_gui
    remove_character_flag = ucm_acm_edit_from_repair_gui
    remove_character_flag = ucm_acm_gui_text_edit_active
    if = {
      limit = { has_global_variable = ucm_acm_recipient }
      remove_global_variable = ucm_acm_recipient
    }
    if = {
      limit = { has_global_variable = ucm_acm_repair_target }
      remove_global_variable = ucm_acm_repair_target
    }
    if = {
      limit = { has_variable = ucm_acm_repair_artifact_var }
      remove_variable = ucm_acm_repair_artifact_var
    }
  }
}

    ucm_acm_sync_gui_kind_to_artifact_type_effect = yes
    trigger_event = { id = ucm_artifact_craft_gui.9001 days = 0 }
  }
}

# 君权固化：月度/年度维护、读档、玩家死亡传位给继承人

monthly_playable_pulse = {
  on_actions = {
    ucm_realm_authority_lock_monthly_pulse
    ucm_player_government_lock_monthly_pulse
  }
}

yearly_playable_pulse = {
  on_actions = {
    ucm_realm_authority_lock_yearly_pulse
  }
}

ucm_realm_authority_lock_monthly_pulse = {
	trigger = { ucm_single_player_trigger = yes }
  effect = {
    root = {
      if = {
        limit = { has_character_flag = ucm_realm_authority_locked }
        ucm_cheat_menu_maintain_locked_realm_authority_effect = yes
        ucm_dissolve_liberty_factions_targeting_root_effect = yes
        ucm_clear_authority_rollback_variables_effect = yes
      }
      else_if = {
        limit = {
          has_variable = ucm_realm_authority_desired_level
          is_independent_ruler = yes
        }
        ucm_cheat_menu_maintain_desired_realm_authority_effect = yes
      }
    }
  }
}

ucm_realm_authority_lock_yearly_pulse = {
	trigger = { ucm_single_player_trigger = yes }
  effect = {
    root = {
      if = {
        limit = { has_character_flag = ucm_realm_authority_locked }
        ucm_cheat_menu_maintain_locked_realm_authority_effect = yes
      }
      else_if = {
        limit = {
          has_variable = ucm_realm_authority_desired_level
          is_independent_ruler = yes
        }
        ucm_cheat_menu_maintain_desired_realm_authority_effect = yes
      }
    }
  }
}

# 只挂接 on_actions 列表（可叠加），不直接写 effect，避免覆盖原版/AGOT 的开局 effect
on_game_start_after_lobby = {
  on_actions = { ucm_realm_authority_lock_after_lobby }
}

ucm_realm_authority_lock_after_lobby = {
  effect = {
    every_player = {
      if = {
        limit = { has_character_flag = ucm_realm_authority_locked }
        ucm_cheat_menu_maintain_locked_realm_authority_effect = yes
      }
      else_if = {
        limit = {
          has_variable = ucm_realm_authority_desired_level
          is_independent_ruler = yes
        }
        ucm_cheat_menu_maintain_desired_realm_authority_effect = yes
      }
    }
  }
}

# CK3 1.20: chained via on_actions instead of a second `effect` block (only one effect per on-action survives;
# the collision was dropping vanilla 1.20 logic or this mod logic).
on_death = {
  on_actions = { ucm_realm_authority_on_death }
}

ucm_realm_authority_on_death = {
  effect = {
    if = {
      limit = {
        OR = {
          has_character_flag = ucm_realm_authority_locked
          has_variable = ucm_realm_authority_desired_level
        }
        exists = player_heir
        player_heir = {
          is_ai = no
          is_alive = yes
        }
      }
      player_heir = {
        if = {
          limit = { prev = { has_variable = ucm_realm_authority_desired_level } }
          set_variable = {
            name = ucm_realm_authority_desired_level
            value = prev.var:ucm_realm_authority_desired_level
          }
        }
        if = {
          limit = { prev = { has_character_flag = ucm_realm_authority_locked } }
          add_character_flag = ucm_realm_authority_locked
          ucm_cheat_menu_maintain_locked_realm_authority_effect = yes
        }
        else_if = {
          limit = { has_variable = ucm_realm_authority_desired_level }
          ucm_cheat_menu_maintain_desired_realm_authority_effect = yes
        }
      }
    }
  }
}

# 首都部落/游牧/牧民地产改为城堡，再换东亚宅邸并改政府（对齐原版行政转换）

ucm_convert_character_to_celestial_government_effect = {
  if = {
    limit = {
      has_tgp_dlc_trigger = yes
      NOT = { has_government = celestial_government }
    }
    if = {
      limit = { is_landed = yes }
      capital_province = {
        if = {
          limit = {
            OR = {
              has_holding_type = tribal_holding
              has_holding_type = herder_holding
              has_holding_type = nomad_holding
            }
          }
          set_holding_type = castle_holding
          ucm_ensure_castle_holding_base_castle_building = yes
        }
      }
    }
    if = {
      limit = { has_tgp_dlc_trigger = yes }
      tgp_domicile_conversion_when_changing_government_type = {
        NEW_DOMICILE_TYPE = east_asian_estate
        NEW_GOVERNMENT_TYPE = celestial_government
      }
    }
    change_government = celestial_government
  }
}

      }
      remove_character_flag = new_government_is_byz_admin
    }
    change_government = administrative_government
  }
}

ucm_convert_character_to_feudal_government_effect = {
  if = {
    limit = {
      NOT = { has_government = feudal_government }
      OR = {
        has_character_flag = ucm_chenzhi_force_feudal_conversion
        NOT = { ucm_protected_from_forced_feudal_conversion_trigger = yes }
      }
      # 修复：开局日本律令（japan_administrative_government）不应被通用“转封建”误改
      NOT = { has_government = japan_administrative_government }
      NOT = { has_government = japan_feudal_government }
      # 额外保护：日本文化角色不应被强制转封建
      NOT = {
        AND = {
          culture = japanese
          OR = {
            has_government = japan_administrative_government
            has_government = japan_feudal_government
          }
        }
      }
      # 额外保护：日本帝国头衔持有者不应被强制转封建
      NOT = {
        AND = {
          highest_held_title_tier >= tier_empire
          culture = japanese
          OR = {
            has_government = japan_administrative_government
            has_government = japan_feudal_government
          }
        }
      }
    }
    if = {
      limit = { is_landed = yes }
      capital_province = {
        if = {
          limit = {
            OR = {
              has_holding_type = tribal_holding
              has_holding_type = herder_holding
              has_holding_type = nomad_holding
            }
          }
          set_holding_type = castle_holding
          ucm_ensure_castle_holding_base_castle_building = yes
        }
      }
    }
    change_government = feudal_government
    if = {
      limit = { is_ai = no }
      send_interface_toast = {
        title = ucm_government_changed_feudal_toast_title
        desc = ucm_government_changed_feudal_toast_desc
      }
    }
  }
  else_if = {
    limit = {
      is_ai = no
      is_playable_character = yes
      NOT = { has_government = feudal_government }
      ucm_protected_from_forced_feudal_conversion_trigger = yes
      NOT = { has_character_flag = ucm_chenzhi_force_feudal_conversion }
    }
    ucm_player_government_feudal_blocked_notify_effect = yes
  }
}

ucm_convert_character_to_nomadic_government_effect = {

# CK3 1.20: use the faith's own theocracy type (Christians get ecclesiastical_government with By God Alone),
# the same way vanilla's pam_clergy_events do; plain theocracy_government is the fallback.
ucm_convert_character_to_theocracy_government_effect = {
  if = {
    limit = { NOT = { government_has_flag = government_is_theocracy } }
    if = {
      limit = { exists = faith.faith_theocratic_goverment_type }
      change_government = faith.faith_theocratic_goverment_type
    }
    else = {
      change_government = theocracy_government
    }
  }
}

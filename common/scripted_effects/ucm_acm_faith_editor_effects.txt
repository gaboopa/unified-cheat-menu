# 宗教编辑：信仰详情窗与教义库

# GUI 传入的 pick_doctrine（MakeScopeFlag）写入变量，供 resolve / add_doctrine 使用
ucm_acm_faith_capture_pick_doctrine_effect = {
  if = {
    limit = { exists = scope:pick_doctrine }
    set_variable = {
      name = ucm_acm_pick_doctrine_key
      value = scope:pick_doctrine
    }
  }
}

ucm_acm_faith_clear_pick_category_flags_effect = {
  remove_character_flag = ucm_acm_faith_pick_cat_core_tenets
  remove_character_flag = ucm_acm_faith_pick_cat_main_group
  remove_character_flag = ucm_acm_faith_pick_cat_marriage
  remove_character_flag = ucm_acm_faith_pick_cat_crimes
  remove_character_flag = ucm_acm_faith_pick_cat_clergy
}

ucm_open_acm_faith_editor_effect = {
  remove_character_flag = ucm_acm_faith_gui_locked
  remove_character_flag = ucm_acm_faith_pick_sidebar_open
  remove_character_flag = ucm_acm_faith_doctrine_picker_open
  ucm_acm_faith_clear_pick_category_flags_effect = yes
  ucm_acm_faith_populate_doctrine_catalog_effect = yes
  ucm_acm_faith_refresh_display_lists_effect = yes
  add_character_flag = ucm_acm_faith_editor_open
  ucm_restore_ucm_selected_faith_scope_effect = yes
  # 仅打开 ucm_acm_faith_editor 独立窗；勿 open_view_data=faith，避免劫持左下角宗教详情
}

ucm_close_acm_faith_editor_effect = {

}

ucm_acm_faith_begin_pick_doctrine_effect = {
  remove_character_flag = ucm_acm_faith_doctrine_picker_open
  add_character_flag = ucm_acm_faith_pick_sidebar_open
  remove_variable = ucm_acm_pick_group_name
  ucm_acm_faith_capture_pick_doctrine_effect = yes

  ucm_acm_faith_populate_pick_current_effect = yes
}

# 核心信条：记录被点击槽位，填充全部核心信条供替换
ucm_acm_faith_begin_pick_core_tenets_effect = {
  remove_character_flag = ucm_acm_faith_doctrine_picker_open
  add_character_flag = ucm_acm_faith_pick_sidebar_open
  ucm_acm_faith_capture_pick_doctrine_effect = yes
  if = {
    limit = { has_variable = ucm_acm_pick_doctrine_key }
    set_variable = {
      name = ucm_acm_pick_slot
      value = var:ucm_acm_pick_doctrine_key
    }
  }
  else = {
    remove_variable = ucm_acm_pick_slot
  }
  set_variable = {
    name = ucm_acm_pick_group_name
    value = flag:doctrine_core_tenets
  }
  ucm_acm_faith_populate_pick_current_effect = yes
}

ucm_acm_faith_apply_pick_doctrine_effect = {
  ucm_acm_faith_capture_pick_doctrine_effect = yes
  # 始终以「新选教义」解析组名，避免 begin_pick 时旧组名导致 remove 未命中
  remove_variable = ucm_acm_pick_group_name
  ucm_acm_faith_resolve_pick_group_effect = yes
  if = {
    limit = {
      has_variable = ucm_acm_pick_group_name
      var:ucm_acm_pick_group_name = flag:doctrine_core_tenets
    }
    ucm_acm_faith_remove_pick_slot_doctrine_effect = yes
  }
  ucm_acm_faith_add_doctrine_to_selected_effect = yes
  remove_variable = ucm_acm_pick_slot
  remove_variable = ucm_acm_pick_group_name
  remove_variable = ucm_acm_pick_doctrine_key
  ucm_acm_faith_refresh_display_lists_effect = yes
}

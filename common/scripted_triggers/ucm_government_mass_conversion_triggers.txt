# 不可被「自动/批量转封建」降级的政体（玩家在手改政体菜单中显式确认时仍可通过 force 旗标转换）
ucm_protected_from_forced_feudal_conversion_trigger = {
  OR = {
    has_government = meritocratic_government
    has_government = celestial_government
    has_government = administrative_government
    has_government = steppe_admin_government
    has_government = japan_administrative_government
    has_government = japan_feudal_government
  }
}

# 一键改封臣政府：可转化的有地封臣（含下级封臣），主头衔须伯爵及以上；不含玩家本人与内阁大臣职位持有者

ucm_mass_gov_conversion_target_trigger = {
  is_adult = yes
  NOT = { has_trait = incapable }
  is_imprisoned = no
  NOT = { is_at_war_with_liege = yes }
  is_landed = yes
  highest_held_title_tier >= tier_county
  NOT = { this = root }
  NOR = {
    AND = {
      has_tgp_dlc_trigger = yes
      ucm_is_tgp_minister_trigger = yes
    }
  }
}

ucm_has_any_vassal_for_celestial_conversion_trigger = {
  has_tgp_dlc_trigger = yes
  any_vassal_or_below = {
    ucm_mass_gov_conversion_target_trigger = yes
    NOT = { has_government = celestial_government }
  }
}

ucm_has_any_vassal_for_administrative_conversion_trigger = {
  any_vassal_or_below = {
    ucm_mass_gov_conversion_target_trigger = yes
    NOT = { has_government = administrative_government }
  }
}

ucm_has_any_vassal_for_meritocratic_conversion_trigger = {
  has_tgp_dlc_trigger = yes
  any_vassal_or_below = {
    ucm_mass_gov_conversion_target_trigger = yes
    NOT = { has_government = meritocratic_government }
  }
}

ucm_has_any_vassal_for_japan_administrative_conversion_trigger = {
  has_tgp_dlc_trigger = yes
  any_vassal_or_below = {
    ucm_mass_gov_conversion_target_trigger = yes
    NOT = { has_government = japan_administrative_government }
    tgp_can_become_japan_administrative_trigger = yes
  }
}

ucm_has_any_vassal_for_steppe_admin_conversion_trigger = {
  has_tgp_dlc_trigger = yes
  any_vassal_or_below = {
    ucm_mass_gov_conversion_target_trigger = yes
    NOT = { has_government = steppe_admin_government }
  }
}

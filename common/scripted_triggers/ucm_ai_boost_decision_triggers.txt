# 相对某一邻国明显劣势（max 兵力 < 对方 × 0.4，仅极端弱势才大幅扣分）
ucm_ai_boost_critically_weaker_than_neighbor_trigger = {
  any_neighboring_top_liege_realm_owner = {
    root.max_military_strength < {
      value = this.max_military_strength
      multiply = 0.4
    }
  }
}

# 持有伯爵及以上宣称（近似「有法理/宣称线」）
ucm_ai_boost_has_notable_claim_trigger = {
  any_claim = {
    tier >= tier_county
  }
}

# 邻国弱邦（realm_size ≤ minor_realm_size，原版 12）
ucm_ai_boost_neighbor_small_realm_trigger = {
  any_neighboring_top_liege_realm_owner = {
    realm_size <= minor_realm_size
  }
}

# 同时至少三场战争（略收敛）
ucm_ai_boost_multiple_character_wars_trigger = {
  any_character_war = {
    count >= 3
  }
}

# 任意针对玩家的派系处于「高危」不满（原版 is_dangerous_faction_trigger：农奴 ≤12 个月满条；其它派系 discontent_per_month > 0）
ucm_ai_boost_high_faction_discontent_trigger = {
  any_targeting_faction = {
    is_dangerous_faction_trigger = yes
  }
}

ucm_ai_boost_neighbor_ratio_ge_065_trigger = {
  any_neighboring_top_liege_realm_owner = {
    root.max_military_strength >= {
      value = this.max_military_strength
      multiply = 0.65
    }
  }
}

ucm_ai_boost_human_neighbor_any_trigger = {
  any_neighboring_top_liege_realm_owner = {

  }
}

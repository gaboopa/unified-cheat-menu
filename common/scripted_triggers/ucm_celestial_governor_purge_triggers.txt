# 分级罢免：主头衔为伯爵/公国/王国/帝国四档（可选单档，或事件内「全部主衔档」一次处理四档）；庸碌按文官三项或武官两项
# 目标可为总督，或境内有地的非总督封臣（无总督制时走同一套庸碌阈值与分档）

# 变身大明：布政司制等为廷推(generate)继承，常无 primary_heir / any_heir；须查主衔继承人队列或廷推法
ucm_purge_has_succession_candidate_trigger = {
  OR = {
    exists = primary_heir
    exists = player_heir
    any_heir = {
      is_alive = yes
    }
    any_held_title = {
      is_noble_family_title = no
      any_title_heir = {
        is_alive = yes
      }
    }
  }
}

ucm_purge_eligible_governor_base_trigger = {
  is_ai = yes
  is_adult = yes
  NOT = { has_trait = incapable }
  is_imprisoned = no
  NOT = { is_at_war_with_liege = yes }
  NOR = {
    AND = {
      has_tgp_dlc_trigger = yes
      ucm_is_tgp_minister_trigger = yes
    }
  }
  ucm_is_not_h_china_emperor_title_holder_trigger = yes
  is_governor = yes
  ucm_purge_has_succession_candidate_trigger = yes
}

ucm_purge_eligible_non_governor_vassal_base_trigger = {
  is_ai = yes
  is_adult = yes
  NOT = { has_trait = incapable }
  is_imprisoned = no
  NOT = { is_at_war_with_liege = yes }
  NOR = {
    AND = {
      has_tgp_dlc_trigger = yes
      ucm_is_tgp_minister_trigger = yes
    }
  }
  ucm_is_not_h_china_emperor_title_holder_trigger = yes
  is_landed = yes
  is_governor = no
  ucm_purge_has_succession_candidate_trigger = yes
}

# Military-appointment contracts: martial + prowess; otherwise civil: diplomacy + stewardship + learning

ucm_governor_is_mediocre_by_branch_trigger = {
  OR = {
    AND = {
      OR = {
        vassal_contract_has_flag = celestial_military_appointment
        vassal_contract_has_flag = meritocratic_military_appointment
      }
      ucm_military_skill_sum_for_purge_value < ucm_military_mediocre_threshold_value
    }
    AND = {
      NOR = {
        vassal_contract_has_flag = celestial_military_appointment
        vassal_contract_has_flag = meritocratic_military_appointment
      }
      ucm_civil_skill_sum_for_purge_value < ucm_civil_mediocre_threshold_value
    }
  }
}

ucm_has_purge_target_county_tier_trigger = {
  any_vassal_or_below = {
    OR = {
      ucm_purge_eligible_governor_base_trigger = yes
      ucm_purge_eligible_non_governor_vassal_base_trigger = yes
    }
    ucm_governor_is_mediocre_by_branch_trigger = yes
    highest_held_title_tier = tier_county
  }
}

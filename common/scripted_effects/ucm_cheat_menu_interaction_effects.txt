# 切换玩家：在 actor 上弹出确认事件；目标以互动 recipient 为准（见 ucm_cheat_menu.0099）。
# 选项须带 add_internal_flag = special（同 EP3 冒险者/迁徙等原版合法换角），避免引擎按「ESC 换角/控制台 play」处理而永久关闭成就。

ucm_switch_playable_character_effect = {
  if = {
    limit = {
      exists = scope:ucm_switch_target
      scope:ucm_switch_target = {
        is_alive = yes
        is_playable_character = yes
        is_ai = yes
        NOT = { this = root }
      }
    }
    set_player_character = scope:ucm_switch_target
  }
}

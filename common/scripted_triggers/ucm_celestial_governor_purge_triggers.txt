# Tiered dismissal: primary title at county/duchy/kingdom/empire (one tier, or all four at once through the event's all-tiers option); incompetence by the three civil or two military skills
# The target can be a governor or a landed non-governor vassal in the realm (without the governor system, the same thresholds and tiers apply)

# Become Ming: provincial-administration and similar systems use generate (court nomination) succession and often have no primary_heir / any_heir; check the primary title's heir queue or the nomination law
ucm_purge_has_succession_candidate_trigger = {
  OR = {
    exists = primary_heir
    exists = player_heir
    any_heir = {
      is_alive = yes
    }
    any_held_title = {
      is_noble_family_title = no
      any_title_heir = {
        is_alive = yes
      }
    }
  }
}

ucm_purge_eligible_governor_base_trigger = {
  is_ai = yes
  is_adult = yes
  NOT = { has_trait = incapable }
  is_imprisoned = no
  NOT = { is_at_war_with_liege = yes }
  NOR = {
    AND = {
      has_tgp_dlc_trigger = yes
      ucm_is_tgp_minister_trigger = yes
    }
  }
  ucm_is_not_h_china_emperor_title_holder_trigger = yes
  is_governor = yes
  ucm_purge_has_succession_candidate_trigger = yes
}

ucm_purge_eligible_non_governor_vassal_base_trigger = {
  is_ai = yes
  is_adult = yes
  NOT = { has_trait = incapable }
  is_imprisoned = no
  NOT = { is_at_war_with_liege = yes }
  NOR = {
    AND = {
      has_tgp_dlc_trigger = yes
      ucm_is_tgp_minister_trigger = yes
    }
  }
  ucm_is_not_h_china_emperor_title_holder_trigger = yes
  is_landed = yes
  is_governor = no
  ucm_purge_has_succession_candidate_trigger = yes
}

# Military-appointment contracts: martial + prowess; otherwise civil: diplomacy + stewardship + learning

ucm_governor_is_mediocre_by_branch_trigger = {
  OR = {
    AND = {
      OR = {
        vassal_contract_has_flag = celestial_military_appointment
        vassal_contract_has_flag = meritocratic_military_appointment
      }
      ucm_military_skill_sum_for_purge_value < ucm_military_mediocre_threshold_value
    }
    AND = {
      NOR = {
        vassal_contract_has_flag = celestial_military_appointment
        vassal_contract_has_flag = meritocratic_military_appointment
      }
      ucm_civil_skill_sum_for_purge_value < ucm_civil_mediocre_threshold_value
    }
  }
}

ucm_purge_target_trigger = {
  OR = { ucm_purge_eligible_governor_base_trigger = yes ucm_purge_eligible_non_governor_vassal_base_trigger = yes }
  ucm_governor_is_mediocre_by_branch_trigger = yes
}

ucm_has_purge_target_tier_trigger = { any_vassal_or_below = { ucm_purge_target_trigger = yes highest_held_title_tier = $TIER$ } }
ucm_has_purge_target_county_tier_trigger = { ucm_has_purge_target_tier_trigger = { TIER = tier_county } }
ucm_has_purge_target_duchy_tier_trigger = { ucm_has_purge_target_tier_trigger = { TIER = tier_duchy } }
ucm_has_purge_target_kingdom_tier_trigger = { ucm_has_purge_target_tier_trigger = { TIER = tier_kingdom } }
ucm_has_purge_target_empire_tier_trigger = { ucm_has_purge_target_tier_trigger = { TIER = tier_empire } }
ucm_has_anyone_to_purge_trigger = { any_vassal_or_below = { ucm_purge_target_trigger = yes highest_held_title_tier >= tier_county } }

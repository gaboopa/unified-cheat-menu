# AGOT 删除了部分原版兵种；GetMaA 取空后悬停 MAA 提示会崩溃，故在 AGOT 下隐藏这些按钮
ucm_sand_show_non_agot_maa = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes 
    NOT = { ucm_compat_agot_loaded_trigger = yes }
  }
  is_valid = { always = yes }
}

ucm_sand_window_init = {
	is_shown = { ucm_single_player_trigger = yes }
  scope = character
  is_valid = { always = yes }
  effect = { ucm_sand_maa_populate_catalog_effect = yes }
}

ucm_sand_soldiers_visible = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes  has_character_flag = ucm_sand_soldiers_open }
  is_valid = { always = yes }
}

ucm_sand_soldiers_close = {
	is_shown = { ucm_single_player_trigger = yes }
  scope = character
  is_valid = { always = yes }
  effect = { ucm_close_sand_soldiers_effect = yes }
}

ucm_sand_has_maa_pick = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes  has_character_flag = ucm_sand_has_maa_pick }
  is_valid = { always = yes }
}

ucm_sand_is_picked_maa = {
  scope = character
  saved_scopes = { maa_flag }
  is_shown = { ucm_single_player_trigger = yes 
    has_variable = ucm_sand_maa_pick
    var:ucm_sand_maa_pick = scope:maa_flag
  }
  is_valid = { always = yes }
}

ucm_sand_pick_maa = {
	is_shown = { ucm_single_player_trigger = yes }
  scope = character
  saved_scopes = { maa_flag }
  is_valid = { always = yes }
  effect = { ucm_sand_pick_maa_effect = yes }
}

ucm_sand_spawn_btn = {
	is_shown = { ucm_single_player_trigger = yes }
  scope = character
  is_valid = {
    has_character_flag = ucm_sand_has_maa_pick
    has_variable = ucm_sand_maa_pick
    var:ucm_sand_maa_scale >= 1
  }
  effect = { ucm_sand_spawn_current_effect = yes }
}

ucm_sand_scale_inc = {
	is_shown = { ucm_single_player_trigger = yes }
  scope = character
  is_valid = { var:ucm_sand_maa_scale < 100 }
  effect = { ucm_sand_scale_inc_effect = yes }
}

ucm_sand_scale_dec = {
	is_shown = { ucm_single_player_trigger = yes }
  scope = character
  is_valid = { var:ucm_sand_maa_scale > 1 }
  effect = { ucm_sand_scale_dec_effect = yes }
}

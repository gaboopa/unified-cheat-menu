# Monthly per-player maintenance, fired once per player per month by on_action ucm_monthly_tick
# (common/on_action/ucm_monthly_tick.txt). root = the player. Every step is a no-op unless the
# player switched the matching feature on.

namespace = ucm_monthly_tick

ucm_monthly_tick.0001 = {
	type = character_event
	hidden = yes

	trigger = {
		is_ai = no
		ucm_single_player_trigger = yes
	}

	immediate = {
		# Fidelity shield: end seduce/romance/abduct/elope schemes against and from the harem.
		if = {
			limit = { has_character_flag = ucm_seduce_shield_active }
			ucm_fidelity_affair_sweep_effect = yes
		}

		# Marriage ban: dissolve betrothals that involve a banned character.
		ucm_marriage_ban_betrothal_sweep_effect = yes

		# Government lock: maintain only for players already locked by a menu action.
		if = {
			limit = { has_character_flag = ucm_player_government_locked }
			ucm_player_government_lock_maintain_effect = yes
		}
	}
}

# Chamber Watch: based on the head of household (spouse or concubinist) having ucm_seduce_shield_active; covers their spouses and concubines (is_concubine)
# Checks is_ai = no, not is_playable_character, which let landless spouses through. The AI never has this flag.

# This character is a spouse or concubine under the watch: any spouse, or their concubinist, has the flag
ucm_seduce_shield_active_for_target_trigger = {
  OR = {
    any_spouse = { has_character_flag = ucm_seduce_shield_active }
    AND = {
      is_concubine = yes
      exists = concubinist
      concubinist = { has_character_flag = ucm_seduce_shield_active }
    }
  }
}

# this = a scheme the fidelity sweep ends.
ucm_fidelity_scheme_type_trigger = {
  OR = {
    scheme_type = seduce
    scheme_type = courting
    scheme_type = abduct
    scheme_type = elope
  }
}

# While any human has Chamber Watch on, enable the global AI no-lust/no-adultery rule (effectively changes a game rule)
ucm_seduce_shield_global_ai_rules_active_trigger = {
  any_player = {
    is_ai = no
    has_character_flag = ucm_seduce_shield_active
  }
}

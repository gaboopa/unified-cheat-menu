# Puts every vassal in the realm under the nearest holder above them in the de jure chain.
# Scope and root = the ruler. Only liege links change: no title moves.
ucm_one_click_transfer_direct_vassals_to_de_jure_liege_effect = {
	ucm_de_jure_transfer_sync_full_effect = yes
	send_interface_toast = {
		type = event_toast_effect_good
		title = ucm_de_jure_transfer_batch_complete_toast_title
		desc = ucm_de_jure_transfer_batch_complete_toast_desc
	}
}

ucm_de_jure_transfer_sync_full_effect = {
	while = {
		count = 12
		limit = { ucm_has_de_jure_transferable_direct_vassals_trigger = yes }
		# Resolved each pass: liege changes only apply on resolve, and the next pass needs to see them.
		create_title_and_vassal_change = {
			type = granted
			save_scope_as = ucm_transfer_vassal_change
		}
		every_vassal_or_below = {
			limit = { ucm_vassal_needs_de_jure_transfer_trigger = yes }
			save_scope_as = ucm_dj_vassal
			primary_title = {
				ordered_this_title_or_de_jure_above = {
					limit = {
						exists = holder
						holder = { ucm_de_jure_transfer_holder_trigger = yes }
					}
					order_by = ucm_de_jure_transfer_bottom_up_order_value
					scope:ucm_dj_vassal = {
						change_liege = { liege = prev.holder change = scope:ucm_transfer_vassal_change }
					}
				}
			}
		}
		resolve_title_and_vassal_change = scope:ucm_transfer_vassal_change
	}
}

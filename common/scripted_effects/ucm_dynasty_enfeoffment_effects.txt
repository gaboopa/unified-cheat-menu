# Dynasty titles: same granting chain as the surplus-title grant, but candidates are only adults of root's dynasty still in the realm (the ucm_grant_dynasty_only variable narrows the pool)

ucm_dynasty_one_click_grant_realm_titles_effect = {
  root = {
    set_variable = {
      name = ucm_grant_dynasty_only
      value = 1
    }
  }
  ucm_one_click_grant_realm_titles_effect = yes
  root = {
    if = {
      limit = {
        has_variable = ucm_grant_dynasty_only
      }
      remove_variable = ucm_grant_dynasty_only
    }
  }
}

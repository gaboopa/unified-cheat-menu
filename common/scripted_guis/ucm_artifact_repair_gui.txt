# 宝物修复独立窗口（gui/ucm_artifact_repair.gui）
# 须同时写 is_valid，否则部分版本下 Execute 不生效。

ucm_acm_repair_gui_open_visible = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes 
    has_character_flag = ucm_acm_repair_gui_open
    NOT = { has_character_flag = ucm_acm_gui_text_edit_active }
  }
  is_valid = { always = yes }
}

# gui/ucm_artifact_repair.gui 里用于判断是否已选中待修复宝物（GUI 数据函数没有直接的
# "HasValue(character variable)"，因此用 scripted_gui 包一层真正的脚本触发器 has_variable）
ucm_acm_repair_has_artifact_var = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes  has_variable = ucm_acm_repair_artifact_var }
  is_valid = { always = yes }
}

ucm_acm_repair_gui_close = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes  always = yes }
  is_valid = { always = yes }
  effect = {
    remove_character_flag = ucm_acm_repair_gui_open
    remove_character_flag = ucm_acm_edit_from_repair_gui
    remove_character_flag = ucm_acm_edit_from_craft_gui
    remove_character_flag = ucm_acm_gui_text_edit_active
    if = {
      limit = { has_global_variable = ucm_acm_repair_target }
      remove_global_variable = ucm_acm_repair_target
    }
    if = {
      limit = { has_variable = ucm_acm_repair_artifact_var }
      remove_variable = ucm_acm_repair_artifact_var
    }
  }
}

ucm_acm_repair_gui_save = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes  always = yes }
  is_valid = {
    has_global_variable = ucm_acm_repair_target
  }
  effect = {
    trigger_event = {
      id = ucm_artifact_repair.9001
      days = 0
    }
  }
}

ucm_acm_repair_gui_edit_name = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes  always = yes }
  is_valid = { always = yes }
  effect = {
    ucm_acm_open_artifact_name_editor_effect = yes
  }
}

ucm_acm_repair_gui_edit_desc = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes  always = yes }
  is_valid = { always = yes }
  effect = {
    ucm_acm_open_artifact_desc_editor_effect = yes
  }
}

ucm_acm_repair_gui_repick = {
  scope = character
  is_shown = { ucm_single_player_trigger = yes  always = yes }
  is_valid = { always = yes }
  effect = {
    remove_character_flag = ucm_acm_repair_gui_open
    if = {
      limit = { has_global_variable = ucm_acm_repair_target }
      remove_global_variable = ucm_acm_repair_target
    }
    if = {
      limit = { has_variable = ucm_acm_repair_artifact_var }
      remove_variable = ucm_acm_repair_artifact_var
    }
    ucm_acm_gui_open_repair_picker_effect = yes
  }
}

# The menu target lives in a global variable: an interaction's saved scopes don't reach the events and windows it opens.

ucm_cheat_menu_lock_recipient_effect = {
	if = {
		limit = { exists = scope:recipient }
		set_global_variable = { name = ucm_cheat_menu_recipient value = scope:recipient }
	}
	else_if = {
		limit = { exists = scope:ucm_cheat_menu_target }
		set_global_variable = { name = ucm_cheat_menu_recipient value = scope:ucm_cheat_menu_target }
	}
	else_if = {
		limit = { exists = scope:actor }
		set_global_variable = { name = ucm_cheat_menu_recipient value = scope:actor }
	}
}

ucm_cheat_menu_sync_target_effect = {
	if = {
		limit = { exists = global_var:ucm_cheat_menu_recipient }
		global_var:ucm_cheat_menu_recipient = { save_scope_as = ucm_cheat_menu_target }
	}
	else_if = {
		limit = { exists = scope:recipient }
		scope:recipient = { save_scope_as = ucm_cheat_menu_target }
	}
	else_if = {
		limit = { exists = scope:actor }
		scope:actor = { save_scope_as = ucm_cheat_menu_target }
	}
}

ucm_cheat_menu_sync_trait_edit_target_effect = {
	ucm_cheat_menu_sync_target_effect = yes
	scope:ucm_cheat_menu_target = { save_scope_as = ucm_trait_edit_target }
}

ucm_cheat_menu_sync_lifestyle_target_effect = {
	ucm_cheat_menu_sync_target_effect = yes
	scope:ucm_cheat_menu_target = { save_scope_as = ucm_lifestyle_target }
}

ucm_cheat_menu_open_sand_soldiers_effect = {
	ucm_cheat_menu_lock_recipient_effect = yes
	scope:actor = { ucm_army_spawner_open_effect = yes }
}

ucm_cheat_menu_open_new_traits_effect = {
	ucm_cheat_menu_lock_recipient_effect = yes
	scope:actor = { ucm_open_trait_editor_gui_effect = yes }
}

ucm_cheat_menu_open_base_game_traits_effect = {
	ucm_cheat_menu_lock_recipient_effect = yes
	scope:actor = { ucm_open_trait_editor_gui_effect = yes }
}

ucm_cheat_menu_open_lifestyle_effect = {
	ucm_cheat_menu_lock_recipient_effect = yes
	ucm_cheat_menu_sync_lifestyle_target_effect = yes
	scope:actor = { ucm_open_lifestyle_gui_effect = yes }
}

ucm_cheat_menu_open_steal_artifact_effect = {
	ucm_cheat_menu_lock_recipient_effect = yes
	scope:actor = {
		ucm_cheat_menu_sync_target_effect = yes
		add_character_flag = ucm_steal_picker_active
		set_variable = { name = ucm_steal_victim value = scope:ucm_cheat_menu_target }
		set_variable = { name = ucm_steal_page value = 0 }
		trigger_event = ucm_cheat_menu.3094
	}
}

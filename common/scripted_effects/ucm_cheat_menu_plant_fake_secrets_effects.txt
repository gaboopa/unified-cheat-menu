# Skonester「栽赃秘密」：在 scope:ucm_plant_menu_recipient 上安插秘密；scope:ucm_plant_menu_actor 为玩家。菜单见 ucm_cheat_menu.0097。

# Skonester「安插秘密」（单一互动内多选项）：在 scope:ucm_plant_menu_recipient 上安插莫须有秘密；scope:ucm_plant_menu_actor 为玩家。部分选项需先保存 scope:ucm_plant_secret_extra。

# 从目标或玩家宫廷选一名成年廷臣（非目标、非玩家、非目标主配偶），供「谋害第三人」「婚外情人」等使用。
ucm_plant_save_random_courtier_for_secret_effect = {
  scope:ucm_plant_menu_recipient = {
    if = {
      limit = {
        any_courtier = {
          is_adult = yes
          NOT = { this = scope:ucm_plant_menu_actor }
          NOT = { this = scope:ucm_plant_menu_recipient }
          NOR = {
            is_consort_of = scope:ucm_plant_menu_recipient
          }
        }
      }
      random_courtier = {
        limit = {
          is_adult = yes
          NOT = { this = scope:ucm_plant_menu_actor }
          NOT = { this = scope:ucm_plant_menu_recipient }
          NOR = {
            is_consort_of = scope:ucm_plant_menu_recipient
          }
        }
        save_scope_as = ucm_plant_secret_extra
      }
    }
    else = {
      scope:ucm_plant_menu_actor = {
        if = {
          limit = {
            any_courtier = {
              is_adult = yes
              NOT = { this = scope:ucm_plant_menu_actor }
              NOT = { this = scope:ucm_plant_menu_recipient }
              NOR = {
                is_consort_of = scope:ucm_plant_menu_recipient
              }
            }
          }
          random_courtier = {
            limit = {
              is_adult = yes
              NOT = { this = scope:ucm_plant_menu_actor }
              NOT = { this = scope:ucm_plant_menu_recipient }
              NOR = {
                is_consort_of = scope:ucm_plant_menu_recipient
              }
            }
            save_scope_as = ucm_plant_secret_extra
          }
        }
      }
    }
  }
}

# 婚外情人：与随机廷臣强制情人链 + 双方 secret_lover（逻辑同「与你通奸」，但第三者为 ucm_plant_secret_extra）。
ucm_cheat_menu_plant_lover_secret_via_saved_third_effect = {
  scope:ucm_plant_menu_recipient = {
    if = {
      limit = {
        exists = scope:ucm_plant_secret_extra
      }
      if = {
        limit = {
          has_relation_nemesis = scope:ucm_plant_secret_extra
        }
        remove_relation_nemesis = scope:ucm_plant_secret_extra
      }
      if = {
        limit = {
          has_relation_rival = scope:ucm_plant_secret_extra
        }
        remove_relation_rival = scope:ucm_plant_secret_extra
      }
      if = {
        limit = {
          has_relation_best_friend = scope:ucm_plant_secret_extra
        }
        remove_relation_best_friend = scope:ucm_plant_secret_extra
      }
      if = {
        limit = {
          has_relation_friend = scope:ucm_plant_secret_extra
        }
        remove_relation_friend = scope:ucm_plant_secret_extra
      }
      if = {
        limit = {
          has_relation_potential_lover = scope:ucm_plant_secret_extra
        }
        remove_relation_potential_lover = scope:ucm_plant_secret_extra
      }
      every_relation = {
        type = lover
        limit = {
          NOT = {
            this = scope:ucm_plant_secret_extra
          }
        }
        remove_relation_lover = this
      }
    }
  }
  scope:ucm_plant_secret_extra = {
    if = {
      limit = {
        exists = scope:ucm_plant_menu_recipient
      }
      if = {
        limit = {
          has_relation_nemesis = scope:ucm_plant_menu_recipient
        }
        remove_relation_nemesis = scope:ucm_plant_menu_recipient
      }
      if = {
        limit = {
          has_relation_rival = scope:ucm_plant_menu_recipient
        }
        remove_relation_rival = scope:ucm_plant_menu_recipient
      }
      if = {
        limit = {
          has_relation_best_friend = scope:ucm_plant_menu_recipient
        }
        remove_relation_best_friend = scope:ucm_plant_menu_recipient
      }
      if = {
        limit = {
          has_relation_friend = scope:ucm_plant_menu_recipient
        }
        remove_relation_friend = scope:ucm_plant_menu_recipient
      }
      if = {
        limit = {
          has_relation_potential_lover = scope:ucm_plant_menu_recipient
        }
        remove_relation_potential_lover = scope:ucm_plant_menu_recipient
      }
      every_relation = {
        type = lover
        limit = {
          NOT = {
            this = scope:ucm_plant_menu_recipient
          }
        }
        remove_relation_lover = this
      }
    }
  }
  scope:ucm_plant_menu_recipient = {
    if = {
      limit = {
        exists = scope:ucm_plant_secret_extra
        NOT = {
          has_relation_lover = scope:ucm_plant_secret_extra
        }
      }
      set_relation_lover = {
        target = scope:ucm_plant_secret_extra
        reason = lover_random_1
      }
    }
  }
  scope:ucm_plant_menu_recipient = {
    if = {
      limit = {
        has_relation_lover = scope:ucm_plant_secret_extra
        NOT = {
          is_consort_of = scope:ucm_plant_secret_extra
        }
      }
      add_secret = {
        type = secret_lover
        target = scope:ucm_plant_secret_extra
      }
    }
  }
  scope:ucm_plant_secret_extra = {
    if = {
      limit = {
        has_relation_lover = scope:ucm_plant_menu_recipient
        NOT = {
          is_consort_of = scope:ucm_plant_menu_recipient
        }
      }
      add_secret = {
        type = secret_lover
        target = scope:ucm_plant_menu_recipient
      }
    }
  }
}

# 栽赃菜单：互动结束后 interaction 的 actor/recipient 可能丢失，须用 save_scope；以下情人链对齐「关系确认」lover 效果，但仅用 menu_* 作用域。
ucm_plant_force_romance_plant_menu_effect = {
  scope:ucm_plant_menu_actor = {
    if = {
      limit = {
        is_adult = yes
        scope:ucm_plant_menu_recipient = {
          is_adult = yes
          NOT = { this = scope:ucm_plant_menu_actor }
        }
        NOT = { can_set_relation_lover_trigger = { CHARACTER = scope:ucm_plant_menu_recipient } }
      }
      hidden_effect = {
        set_sexuality = bisexual
        scope:ucm_plant_menu_recipient = { set_sexuality = bisexual }
      }
    }
  }
}

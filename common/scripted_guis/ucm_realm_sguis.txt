# Unified Cheat Menu: Titles, Faith & Culture and Realm tabs (spec 4.3-4.5). Root = player.
# Target: var:ucm_target. Second character: global_var:ucm_crozier_cheats_other.

# --- Titles, claims, vassals ----------------------------------------------------
ucm_target_claims_second_title = {
	scope = character
	is_shown = { ucm_single_player_trigger = yes }
	is_valid = { ucm_menu_pair_valid_trigger = yes global_var:ucm_crozier_cheats_other = { is_landed = yes } NOT = { var:ucm_target = { has_claim_on = global_var:ucm_crozier_cheats_other.primary_title } } }
	effect = { var:ucm_target = { add_pressed_claim = global_var:ucm_crozier_cheats_other.primary_title } }
}

ucm_target_independent = {
	scope = character
	is_shown = { ucm_single_player_trigger = yes }
	is_valid = { ucm_menu_target_valid_trigger = yes var:ucm_target = { is_landed = yes is_independent_ruler = no } }
	effect = { var:ucm_target = { create_title_and_vassal_change = { type = independency save_scope_as = ucm_change add_claim_on_loss = no } becomes_independent = { change = scope:ucm_change } resolve_title_and_vassal_change = scope:ucm_change } }
}

ucm_target_vassal_of_second = {
	scope = character
	is_shown = { ucm_single_player_trigger = yes }
	is_valid = { ucm_menu_pair_valid_trigger = yes var:ucm_target = { is_landed = yes } global_var:ucm_crozier_cheats_other = { is_landed = yes NOT = { any_liege_or_above = { this = var:ucm_target } } } NOT = { var:ucm_target.liege ?= global_var:ucm_crozier_cheats_other } }
	effect = { var:ucm_target = { create_title_and_vassal_change = { type = swear_fealty save_scope_as = ucm_change add_claim_on_loss = no } change_liege = { liege = global_var:ucm_crozier_cheats_other change = scope:ucm_change } resolve_title_and_vassal_change = scope:ucm_change } }
}

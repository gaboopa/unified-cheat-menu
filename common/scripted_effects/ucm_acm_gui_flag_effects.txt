  remove_character_flag = ucm_artifact_quality_masterwork

}

ucm_acm_toggle_give_recipient_effect = {
  if = {
    limit = { has_character_flag = ucm_craft_give_to_recipient }
    remove_character_flag = ucm_craft_give_to_recipient
  }
  else = {
    add_character_flag = ucm_craft_give_to_recipient
  }
}

ucm_acm_inc_craft_m1_cnt_effect = {
  if = {
    limit = { NOT = { var:ucm_craft_m1 = 0 } }
    ucm_acm_sync_craft_layer_quality_effect = yes
    if = {
      limit = { NOT = { has_variable = ucm_craft_m1_cnt } }
      set_variable = {
        name = ucm_craft_m1_cnt
        value = var:ucm_craft_quality_min_layers
      }
    }
    if = {
      limit = {
        var:ucm_craft_m1_cnt < var:ucm_craft_quality_min_layers
      }
      set_variable = {
        name = ucm_craft_m1_cnt
        value = var:ucm_craft_quality_min_layers
      }
    }
    change_variable = { name = ucm_craft_m1_cnt add = 1 }
    if = {
      limit = { var:ucm_craft_m1_cnt > 999 }
      set_variable = { name = ucm_craft_m1_cnt value = 999 }
    }
  }
}

ucm_acm_dec_craft_m1_cnt_effect = {

# 旧 GUI 旗帜 → ucm_acm_artifact_type（create 分支用）；须在 scope:actor 上调用
ucm_acm_sync_gui_kind_to_artifact_type_effect = {
  if = {
    limit = { has_character_flag = ucm_artifact_subtype_sword }
    set_variable = { name = ucm_acm_artifact_type value = flag:ucm_acm_sword }
  }
  else_if = {
    limit = { has_character_flag = ucm_artifact_subtype_axe }
    set_variable = { name = ucm_acm_artifact_type value = flag:ucm_acm_axe }
  }
  else_if = {
    limit = { has_character_flag = ucm_artifact_subtype_spear }
    set_variable = { name = ucm_acm_artifact_type value = flag:ucm_acm_spear }
  }
  else_if = {
    limit = { has_character_flag = ucm_artifact_kind_armor }
    set_variable = { name = ucm_acm_artifact_type value = flag:ucm_acm_armor }
  }
  else_if = {
    limit = { has_character_flag = ucm_artifact_kind_crown }
    set_variable = { name = ucm_acm_artifact_type value = flag:ucm_acm_crown }
  }
  else_if = {
    limit = { has_character_flag = ucm_artifact_kind_regalia }
    set_variable = { name = ucm_acm_artifact_type value = flag:ucm_acm_regalia }
  }
  else_if = {
    limit = { has_character_flag = ucm_artifact_kind_trinket }
    set_variable = { name = ucm_acm_artifact_type value = flag:ucm_acm_necklace }
  }
  else_if = {
    limit = { has_character_flag = ucm_artifact_kind_book }
    # advcm 类型表无 book，暂用 goblet 模板生成典籍外形宝物
    set_variable = { name = ucm_acm_artifact_type value = flag:ucm_acm_goblet }
  }
  else_if = {
    limit = { has_character_flag = ucm_artifact_kind_court }
    set_variable = { name = ucm_acm_artifact_type value = flag:ucm_acm_throne }
  }
  else_if = {
    limit = { has_character_flag = ucm_artifact_kind_weapon }
    set_variable = { name = ucm_acm_artifact_type value = flag:ucm_acm_sword }
  }
}

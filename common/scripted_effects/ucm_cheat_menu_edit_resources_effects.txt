# 官吏统辖「修改资源」：子事件链中 scope:recipient 会丢失，须从 global_var:ucm_cheat_menu_recipient 恢复。

ucm_cheat_menu_edit_resources_sync_recipient_effect = {
  ucm_cheat_menu_sync_target_effect = yes
  if = {
    limit = { exists = scope:ucm_cheat_menu_target }
    scope:ucm_cheat_menu_target = { save_scope_as = recipient }
  }
  else_if = {
    limit = { has_global_variable = ucm_cheat_menu_recipient }
    global_var:ucm_cheat_menu_recipient = { save_scope_as = recipient }
  }
}

# 直辖上限无上限：先清叠层再挂 +999（可重复点击刷新）
ucm_cheat_menu_recipient_apply_domain_unlimited_effect = {
  ucm_cheat_menu_recipient_clear_domain_limit_mods_effect = yes
  ucm_cheat_menu_recipient_add_domain_modifier_effect = {
    MODIFIER = ucm_edit_res_domain_unlimited
  }
}

ucm_cheat_menu_recipient_add_flat_dread_effect = {
  scope:recipient = {
    set_variable = {
      name = ucm_flat_dread_target
      value = {
        value = dread
        add = $DELTA$
        min = 0
      }
    }
    if = {
      limit = {
        NOT = { ucm_cheat_menu_dread_cap_150_trigger = yes }
      }
      if = {
        limit = { var:ucm_flat_dread_target > 100 }
        set_variable = {
          name = ucm_flat_dread_target
          value = 100
        }
      }
    }
    else = {
      if = {
        limit = { var:ucm_flat_dread_target > 150 }
        set_variable = {
          name = ucm_flat_dread_target
          value = 150
        }
      }
    }
    if = {
      limit = { var:ucm_flat_dread_target > dread }
      while = {
        count = 600
        limit = {
          dread < var:ucm_flat_dread_target
        }
        add_dread = 1
      }
    }
    else_if = {
      limit = { var:ucm_flat_dread_target < dread }
      while = {
        count = 600
        limit = {
          dread > var:ucm_flat_dread_target
        }
        add_dread = -1
      }
    }
    remove_variable = ucm_flat_dread_target
  }
}

      remove_short_term_gold = { value = gold }
      remove_long_term_gold = { value = gold }
      remove_reserved_gold = { value = gold }
    }
  }
  else_if = {
    limit = { exists = scope:recipient }
    scope:recipient = {
      remove_short_term_gold = { value = gold }
      remove_long_term_gold = { value = gold }
      remove_reserved_gold = { value = gold }
    }
  }
}

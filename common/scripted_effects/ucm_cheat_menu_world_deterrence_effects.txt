# Skonester菜单 · 威慑天下：恐怖值锁定 100；直属封臣畏服、禁派系；封臣好感冻结

ucm_cheat_menu_lock_dread_at_100_effect = {
  if = {
    limit = { dread < 100 }
    while = {
      count = 150
      limit = { dread < 100 }
      add_dread = 1
    }
  }
  if = {
    limit = { dread > 100 }
    while = {
      count = 150
      limit = { dread > 100 }
      add_dread = -1
    }
  }
}

ucm_cheat_menu_save_world_deterrence_opinion_effect = {
  if = {
    limit = { NOT = { has_variable = ucm_world_deterrence_opinion_saved } }
    set_variable = {
      name = ucm_world_deterrence_opinion_saved
      value = "opinion(scope:ucm_deterrence_liege)"
    }
  }
}

ucm_cheat_menu_restore_world_deterrence_opinion_effect = {
  if = {
    limit = { has_variable = ucm_world_deterrence_opinion_saved }
    set_variable = {
      name = ucm_world_deterrence_opinion_delta
      value = {
        value = var:ucm_world_deterrence_opinion_saved
        subtract = "opinion(scope:ucm_deterrence_liege)"
      }
    }
    if = {
      limit = { NOT = { var:ucm_world_deterrence_opinion_delta = 0 } }
      # add_opinion needs a modifier; generic vanilla modifiers by sign
      if = {
        limit = { var:ucm_world_deterrence_opinion_delta > 0 }
        add_opinion = { target = scope:ucm_deterrence_liege modifier = friendliness_opinion opinion = var:ucm_world_deterrence_opinion_delta }
      }
      else = {
        add_opinion = { target = scope:ucm_deterrence_liege modifier = contempt_opinion opinion = var:ucm_world_deterrence_opinion_delta }
      }
    }
    remove_variable = ucm_world_deterrence_opinion_delta
  }
}

ucm_cheat_menu_apply_world_deterrence_on_vassal_effect = {
  scope:ucm_deterrence_vassal = {
    remove_opinion = {
      target = scope:ucm_deterrence_liege
      modifier = ucm_world_deterrence_fear_opinion
    }
    ucm_cheat_menu_save_world_deterrence_opinion_effect = yes
    ucm_cheat_menu_strong_deterrence_leave_factions_vs_liege_effect = yes
    ucm_cheat_menu_strong_deterrence_end_wars_vs_liege_effect = yes
    if = {
      limit = { NOT = { has_character_flag = joining_faction_block } }
      add_character_flag = joining_faction_block
    }
    if = {
      limit = { NOT = { has_character_flag = ucm_world_deterrence_vassal } }
      add_character_flag = ucm_world_deterrence_vassal
    }
    set_variable = {
      name = ucm_world_deterrence_liege
      value = scope:ucm_deterrence_liege
    }
  }
  scope:ucm_deterrence_liege = {
    ucm_cheat_menu_strong_deterrence_give_liege_hook_effect = yes
  }
}

ucm_cheat_menu_apply_world_deterrence_direct_vassals_effect = {
  every_vassal = {
    limit = { is_alive = yes }
    save_scope_as = ucm_deterrence_vassal
    ucm_cheat_menu_apply_world_deterrence_on_vassal_effect = yes
  }
}

ucm_cheat_menu_apply_world_deterrence_effect = {
  scope:actor = {
    save_scope_as = ucm_deterrence_liege
    if = {
      limit = { NOT = { has_character_flag = ucm_world_deterrence_active } }
      add_character_flag = ucm_world_deterrence_active
    }
    if = {
      limit = { NOT = { has_character_modifier = ucm_world_deterrence_aura } }
      add_character_modifier = {
        modifier = ucm_world_deterrence_aura
      }
    }
    ucm_cheat_menu_lock_dread_at_100_effect = yes
    ucm_cheat_menu_apply_world_deterrence_direct_vassals_effect = yes
    send_interface_toast = {
      type = event_toast_effect_good
      title = ucm_world_deterrence_toast_title
      desc = ucm_world_deterrence_toast_desc
      left_icon = scope:actor
    }
  }
}

# 月度：维持恐怖值 100、为新直属封臣补效果、冻结好感
ucm_cheat_menu_world_deterrence_monthly_on_liege_effect = {
  if = {
    limit = { has_character_flag = ucm_world_deterrence_active }
    root = { save_scope_as = ucm_deterrence_liege }
    ucm_cheat_menu_lock_dread_at_100_effect = yes
    if = {
      limit = { NOT = { has_character_modifier = ucm_world_deterrence_aura } }
      add_character_modifier = {
        modifier = ucm_world_deterrence_aura
      }
    }
    every_vassal = {
      limit = {
        is_alive = yes
        OR = {
          NOT = { has_character_flag = ucm_world_deterrence_vassal }
          NOT = {
            var:ucm_world_deterrence_liege ?= scope:ucm_deterrence_liege
          }
        }
      }
      save_scope_as = ucm_deterrence_vassal
      ucm_cheat_menu_apply_world_deterrence_on_vassal_effect = yes
    }
    every_vassal = {
      limit = {
        is_alive = yes
        has_character_flag = ucm_world_deterrence_vassal
        var:ucm_world_deterrence_liege ?= scope:ucm_deterrence_liege
      }
      save_scope_as = ucm_deterrence_vassal
      ucm_cheat_menu_restore_world_deterrence_opinion_effect = yes
      ucm_cheat_menu_strong_deterrence_leave_factions_vs_liege_effect = yes
      ucm_cheat_menu_strong_deterrence_end_wars_vs_liege_effect = yes
    }
  }
}

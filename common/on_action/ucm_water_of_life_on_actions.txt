# Append scoped upkeep to the existing engine hooks; no vanilla overrides.
on_game_start = {
    on_actions = { ucm_water_of_life_restore_all }
}
on_game_start_after_lobby = {
    on_actions = { ucm_water_of_life_restore_all }
}
monthly_global_pulse = {
    on_actions = { ucm_water_of_life_restore_all }
}
ucm_water_of_life_restore_all = {
	trigger = { ucm_single_player_trigger = yes }
    effect = {
        every_living_character = {
            limit = { has_trait = water_of_life }
            ucm_water_of_life_restore_effect = yes
        }
    }
}

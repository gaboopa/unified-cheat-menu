ucm_de_jure_transfer_vassals_by_primary_chain_effect = {
  every_vassal_or_below = {
    limit = {
      is_landed = yes
      exists = primary_title
      NOT = { is_at_war_with = root }
      ucm_vassal_has_de_jure_transfer_target_in_realm_trigger = yes
    }
    ucm_de_jure_transfer_one_vassal_by_primary_chain_effect = yes
  }
}

ucm_de_jure_transfer_sync_full_effect = {
  create_title_and_vassal_change = {
    type = granted
    save_scope_as = ucm_transfer_vassal_change
  }
  while = {
    count = 12
    limit = {
      ucm_has_de_jure_transferable_direct_vassals_trigger = yes
    }
    ucm_de_jure_transfer_vassals_by_primary_chain_effect = yes
  }
  resolve_title_and_vassal_change = scope:ucm_transfer_vassal_change
}

ucm_de_jure_transfer_send_start_toast_effect = {
  send_interface_toast = {
    type = event_toast_effect_neutral
    title = ucm_de_jure_transfer_batch_start_toast_title
    desc = ucm_de_jure_transfer_batch_start_toast_desc
  }
}

ucm_de_jure_transfer_send_complete_toast_effect = {
  send_interface_toast = {
    type = event_toast_effect_good
    title = ucm_de_jure_transfer_batch_complete_toast_title
    desc = ucm_de_jure_transfer_batch_complete_toast_desc
  }
}

ucm_de_jure_transfer_batch_build_vassal_queue_effect = {
  clear_variable_list = ucm_de_jure_transfer_vassal_queue
  every_vassal_or_below = {
    limit = {
      is_landed = yes
      exists = primary_title
      NOT = { is_at_war_with = root }
      ucm_vassal_has_de_jure_transfer_target_in_realm_trigger = yes
    }
    save_scope_as = ucm_de_jure_transfer_queue_vassal
    root = {
      add_to_variable_list = {
        name = ucm_de_jure_transfer_vassal_queue
        target = scope:ucm_de_jure_transfer_queue_vassal
      }
    }
  }
}

ucm_de_jure_transfer_batch_finish_effect = {
  remove_character_flag = ucm_de_jure_transfer_batch_active
  clear_variable_list = ucm_de_jure_transfer_vassal_queue
  ucm_de_jure_transfer_send_complete_toast_effect = yes
}

ucm_de_jure_transfer_batch_step_effect = {
  create_title_and_vassal_change = {
    type = granted
    save_scope_as = ucm_transfer_vassal_change
  }
  while = {
    count = 12
    limit = {
      variable_list_size = {
        name = ucm_de_jure_transfer_vassal_queue
        value > 0
      }
    }
    ordered_in_list = {
      variable = ucm_de_jure_transfer_vassal_queue
      order_by = ucm_de_jure_transfer_bottom_up_order_value
      position = 0
      save_scope_as = ucm_de_jure_transfer_queue_vassal
    }
    remove_list_variable = {
      name = ucm_de_jure_transfer_vassal_queue
      target = scope:ucm_de_jure_transfer_queue_vassal
    }
    scope:ucm_de_jure_transfer_queue_vassal = {
      ucm_de_jure_transfer_one_vassal_by_primary_chain_effect = yes
    }
    clear_saved_scope = ucm_de_jure_transfer_queue_vassal
  }
  resolve_title_and_vassal_change = scope:ucm_transfer_vassal_change
}

ucm_de_jure_transfer_batch_start_effect = {
  root = {
    if = {
      limit = {
        has_character_flag = ucm_de_jure_transfer_batch_active
      }
      ucm_de_jure_transfer_batch_finish_effect = yes
    }
    add_character_flag = ucm_de_jure_transfer_batch_active
    ucm_de_jure_transfer_batch_build_vassal_queue_effect = yes
    ucm_de_jure_transfer_send_start_toast_effect = yes
    trigger_event = { id = ucm_officials.9130 days = 0 }
  }
}

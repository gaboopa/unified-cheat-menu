# 菜单已打开：旗标在右键发起者 scope:actor（玩家）上，必须显式 scope:actor 判定
# 否则 is_shown 在 scope:recipient 块之后会误查目标角色的旗标，导致打开菜单后选项反而消失
ucm_cheat_menu_menu_enabled_trigger = {
  OR = {
    scope:actor = { has_character_flag = ucm_cheat_menu_cheat_menu }
    AND = {
      NOT = { exists = scope:actor }
      has_character_flag = ucm_cheat_menu_cheat_menu
    }
  }
}

# 宝物窃取列表（翻页后勿因主菜单旗标丢失而整页选项消失）
ucm_steal_picker_enabled_trigger = {
  OR = {
    ucm_cheat_menu_menu_enabled_trigger = yes
    has_character_flag = ucm_steal_picker_active
  }
}

# 宝物制作/生成：仅人类玩家（禁止 AI 走 GUI、事件或 scripted_effect 造宝）
ucm_player_may_use_artifact_craft_trigger = {
  is_ai = no
}

# Skonester特质防护：月度全图 + 出生/每年补扫 + 开局清一次
# 每日全局分批 / AI 逐人脉冲 / 获封补刀 → zzz_ucm_trait_guard_pulses.txt
# 仅玩家Skonester菜单授予（ucm_traits_player_authorized）可保留；AI 亦同

on_birth_child = {
  on_actions = {
    ucm_trait_ai_guard_on_birth_pulse
  }
}

ucm_trait_ai_guard_on_birth_pulse = {
  effect = {
    if = {
      limit = {
        ucm_cheat_menu_must_strip_ucm_traits_trigger = yes
      }
      ucm_cheat_menu_strip_oob_ucm_traits_on_this_character_effect = yes
    }
  }
}

monthly_global_pulse = {
  on_actions = {
    ucm_trait_ai_guard_monthly_pulse
    ucm_trait_baidubuqin_monthly_upkeep_pulse
    ucm_trait_xinwuguai_monthly_upkeep_pulse
  }
}

ucm_trait_baidubuqin_monthly_upkeep_pulse = {
	trigger = { ucm_single_player_trigger = yes }
  effect = {
    every_living_character = {
      limit = {
        has_trait = ucm_trait_baidubuqin
      }
      ucm_trait_baidubuqin_upkeep_effect = yes
    }
  }
}

ucm_trait_xinwuguai_monthly_upkeep_pulse = {
	trigger = { ucm_single_player_trigger = yes }
  effect = {
    every_living_character = {
      limit = {
        has_trait = ucm_trait_xinwuguai
      }
      ucm_trait_xinwuguai_upkeep_effect = yes
    }
  }
}

random_yearly_everyone_pulse = {
  on_actions = {
    ucm_trait_ai_guard_yearly_spot_pulse
  }
}

ucm_trait_ai_guard_monthly_pulse = {
  effect = {
    ucm_cheat_menu_strip_all_unauthorized_ucm_traits_effect = yes
  }
}

# 每年对每名角色随机触发一次，兜底月度漏网

      }
      ucm_cheat_menu_strip_oob_ucm_traits_on_this_character_effect = yes
    }
  }
}

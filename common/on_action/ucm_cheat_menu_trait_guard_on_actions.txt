# UCM特质防护：出生 / 每年补扫 / 获封补刀 (zzz_ucm_trait_guard_pulses.txt)；开局全图清一次见 ucm_startup_sweep.txt
# 仅玩家UCM菜单授予（ucm_traits_player_authorized）可保留；AI 亦同
# (monthly_global_pulse / daily_global_pulse / ai_character_pulse are never fired by the engine; removed.)

on_birth_child = {
  on_actions = {
    ucm_trait_ai_guard_on_birth_pulse
  }
}

ucm_trait_ai_guard_on_birth_pulse = {
  trigger = {
    ucm_single_player_trigger = yes
    ucm_cheat_menu_must_strip_ucm_traits_trigger = yes
  }
  effect = {
    ucm_cheat_menu_strip_oob_ucm_traits_on_this_character_effect = yes
  }
}

ucm_monthly_tick = {
  on_actions = {
    ucm_trait_baidubuqin_monthly_upkeep_pulse
    ucm_trait_xinwuguai_monthly_upkeep_pulse
  }
}

ucm_trait_baidubuqin_monthly_upkeep_pulse = {
	trigger = { ucm_single_player_trigger = yes }
  effect = {
    # One-time migration for saves made before the holders list existed.
    if = {
      limit = { NOT = { has_global_variable = ucm_baidubuqin_list_built } }
      set_global_variable = { name = ucm_baidubuqin_list_built value = 1 }
      every_living_character = {
        limit = { has_trait = ucm_trait_baidubuqin }
        ucm_holder_list_add_effect = { LIST = ucm_baidubuqin_holders }
      }
    }
    every_in_global_list = {
      variable = ucm_baidubuqin_holders
      limit = {
        is_alive = yes
        has_trait = ucm_trait_baidubuqin
      }
      ucm_trait_baidubuqin_upkeep_effect = yes
    }
    # Collect stale entries first, remove after iterating (never edit a list while iterating it).
    every_in_global_list = {
      variable = ucm_baidubuqin_holders
      limit = {
        OR = {
          is_alive = no
          NOT = { has_trait = ucm_trait_baidubuqin }
        }
      }
      add_to_temporary_list = ucm_stale_ucm_baidubuqin_holders
    }
    every_in_list = {
      list = ucm_stale_ucm_baidubuqin_holders
      remove_list_global_variable = { name = ucm_baidubuqin_holders target = this }
    }
  }
}

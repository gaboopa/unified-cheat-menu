# A new titular title named and coloured after scope:target, held by scope:recipient.
ucm_create_titular_effect = {
	create_dynamic_title = { tier = $TIER$ name = UCM_TITULAR_NAME adjective = UCM_TITULAR_ADJECTIVE }
	scope:actor = {
		create_title_and_vassal_change = { type = created save_scope_as = ucm_titular_change add_claim_on_loss = no }
		scope:new_title = {
			set_color_from_title = scope:target
			set_coa = scope:target
			change_title_holder = { holder = scope:recipient change = scope:ucm_titular_change }
		}
		resolve_title_and_vassal_change = scope:ucm_titular_change
	}
}

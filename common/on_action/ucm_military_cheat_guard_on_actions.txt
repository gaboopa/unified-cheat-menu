  on_actions = {
    ucm_military_cheat_guard_monthly_pulse

    ucm_military_cheat_strip_all_unauthorized_effect = yes

    ucm_military_cheat_guard_yearly_spot_pulse

  effect = {

      limit = {
        ucm_has_any_military_cheat_modifier_trigger = yes

      ucm_military_cheat_strip_on_character_effect = yes

  }
}

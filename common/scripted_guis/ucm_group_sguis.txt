# Unified Cheat Menu: group-action picker (spec Section 5). Root = player.
# Pick a group (sets var:ucm_group_pick), then run an action on it through event ucm_group.1.

ucm_group_pick_courtiers = {
	scope = character
	is_shown = { ucm_single_player_trigger = yes }
	effect = { set_variable = { name = ucm_group_pick value = flag:courtiers } }
}

# Mythical legend presets for the menu's Legend section (tools/gen_ucm_extra.py lists every ucm_legend_master_add_*).
# Root = the player. scope:ucm_cheat_menu_target receives the legend and starts promoting it.

ucm_legend_master_promote_effect = {
  if = { limit = { exists = scope:ucm_new_legend } set_promoted_legend = scope:ucm_new_legend }
}

# CK3 has no effect that deletes a legend; the target just stops spreading the one they promote.
ucm_legend_master_clear_owned_legends_effect = {
  scope:ucm_cheat_menu_target ?= { stop_promoting_legend = yes }
}

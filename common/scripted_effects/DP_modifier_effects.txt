### Modifier Menu Customizer ###

dp_mod_plus_step_effect = {
	if = {
		limit = { has_character_flag = $MOD$_flag }
		if = {
			limit = { NOT= { has_character_modifier = $MOD$_neg } }
			add_character_modifier = { modifier = $MOD$ }
		}
		else = { remove_character_modifier = $MOD$_neg }
	}
}

dp_mod_minus_step_effect = {
	if = {
		limit = { has_character_flag = $MOD$_flag }
		if = {
			limit = { NOT= { has_character_modifier = $MOD$ } }
			add_character_modifier = { modifier = $MOD$_neg }
		}
		else = { remove_character_modifier = $MOD$ }
	}
}

dp_mod_clear_flag_effect = {
	if = { limit = { has_character_flag = $MOD$_flag } remove_character_flag = $MOD$_flag }
}

dp_mod_clear_instances_effect = {
	if = { limit = { has_character_flag = $MOD$_flag } remove_all_character_modifier_instances = $MOD$ remove_all_character_modifier_instances = $MOD$_neg }
}

dp_mod_reset_effect = {
	if = { limit = { has_character_modifier = $MOD$ } remove_all_character_modifier_instances = $MOD$ }
}

# Some holding type changes leave a province without its main building, so it shows no tax. Scope: province.

ucm_ensure_holding_base_effect = { if = { limit = { has_holding_type = $HOLDING$ NOT = { has_building_or_higher = $BUILDING$ } } add_building = $BUILDING$ } }
ucm_ensure_castle_holding_base_castle_building = { ucm_ensure_holding_base_effect = { HOLDING = castle_holding BUILDING = castle_01 } }
ucm_ensure_city_holding_base_city_building = { ucm_ensure_holding_base_effect = { HOLDING = city_holding BUILDING = city_01 } }
ucm_ensure_church_holding_base_temple_building = { ucm_ensure_holding_base_effect = { HOLDING = church_holding BUILDING = temple_01 } }
ucm_ensure_tribal_holding_base_tribe_building = { ucm_ensure_holding_base_effect = { HOLDING = tribal_holding BUILDING = tribe_01 } }

# Scope: province.
ucm_repair_province_holding_base_effect = { ucm_ensure_castle_holding_base_castle_building = yes ucm_ensure_city_holding_base_city_building = yes ucm_ensure_church_holding_base_temple_building = yes ucm_ensure_tribal_holding_base_tribe_building = yes }

# Scope: character. Their counties' provinces, plus baronies held outside a held county.
ucm_held_titles_repair_broken_holding_bases_effect = {
  every_held_county = { every_county_province = { ucm_repair_province_holding_base_effect = yes } }
  every_held_title = { limit = { tier = tier_barony } title_province = { ucm_repair_province_holding_base_effect = yes } }
}

ucm_realm_repair_broken_holding_bases_effect = {
  ucm_held_titles_repair_broken_holding_bases_effect = yes
  every_vassal_or_below = { limit = { is_landed = yes } ucm_held_titles_repair_broken_holding_bases_effect = yes }
}
